\section{Preliminares y Definiciones Geométricas}

Consideramos el espacio de polinomios $\Poly$ equipado con el siguiente producto interno de Sobolev discreto:
\begin{equation}
    \ip{p}{q} = \int_{S_R} p(z)\overline{q(z)} d\mu(z) + \sum_{k=1}^{N} p'(c_k)\overline{q'(c_k)},
\end{equation}
donde $\mu$ es la medida de Lebesgue soportada en la circunferencia $S_R = \{z \in \C : \abs{z} = R\}$ y $\mathcal{A} = \{c_1, \dots, c_N\} \subset \C$ es el conjunto finito de átomos en las derivadas.

\begin{definicion}[Espacio de Hilbert Sobolev]
    Denotamos por $\mathcal{H}$ a la completación del espacio de polinomios $\Poly$ respecto a la norma inducida $\norm{p} = \sqrt{\ip{p}{p}}$. Es decir, $\mathcal{H} = \overline{\Poly}^{\norm{\cdot}}$.
\end{definicion}

Para estudiar el operador de multiplicación sobre subespacios definidos mediante condiciones de anulación en los átomos, es necesario distinguir entre la estructura algebraica del espacio de polinomios y la topología del espacio de Hilbert. A tal efecto, se definen las evaluaciones puntuales en el espacio de polinomios y se considera su clausura topológica, siguiendo la metodología estándar en el análisis de operadores no acotados.

\begin{definicion}[Funcionales de Evaluación Algebraicos]
    Para cualquier $\xi \in \C$, definimos el funcional de evaluación lineal sobre el espacio de polinomios $\delta_\xi : \Poly \to \C$ como $\delta_\xi(p) = p(\xi)$. Análogamente, definimos la evaluación de la derivada como $\delta'_\xi(p) = p'(\xi)$.
\end{definicion}

Basándonos en la geometría de los átomos respecto al soporte continuo, clasificamos el conjunto de índices $\mathcal{I} = \{1, \dots, N\}$ en dos categorías:
\begin{itemize}
    \item \textbf{Átomos Interiores ($I_{in}$):} $\{k \in \mathcal{I} : \abs{c_k} < R\}$.
    \item \textbf{Átomos Exteriores ($I_{out}$):} $\{k \in \mathcal{I} : \abs{c_k} > R\}$. Sea $m = |I_{out}|$.
\end{itemize}

\begin{proposicion}[Acotación y Extensión Continua en el Interior]
    Para los átomos interiores ($k \in I_{in}$), el funcional de evaluación $\delta_{c_k}$ es continuo respecto a la norma de Sobolev $\norm{\cdot}$. En consecuencia, puede extenderse de forma única a un funcional lineal acotado en todo el espacio de Hilbert $\mathcal{H}$, y su núcleo es un subespacio cerrado.
\end{proposicion}

\begin{proof}
    Para cualquier polinomio $p \in \Poly$ y un átomo $c_k$ con $\abs{c_k} < R$, podemos aplicar la Fórmula Integral de Cauchy sobre la circunferencia $S_R$:
    $$ p(c_k) = \frac{1}{2\pi i} \int_{S_R} \frac{p(z)}{z - c_k} dz. $$
    Tomando valores absolutos y aplicando la desigualdad de Cauchy-Schwarz:
    $$ \abs{p(c_k)}^2 \leq \left( \frac{1}{2\pi} \int_{S_R} \frac{1}{\abs{z - c_k}^2} \abs{dz} \right) \left( \int_{S_R} \abs{p(z)}^2 \abs{dz} \right). $$
    Puesto que $\abs{c_k} < R$, la distancia al borde de la circunferencia está estrictamente acotada por abajo, luego la primera integral es una constante finita $M_{c_k} > 0$. Además, la segunda integral está dominada por la parte continua de la norma de Sobolev (salvo una constante de proporcionalidad dependiente de la medida de Lebesgue). Por tanto:
    $$ \abs{p(c_k)}^2 \leq \tilde{M}_{c_k} \|p\|_{L^2(\mu)}^2 \leq \tilde{M}_{c_k} \norm{p}^2. $$
    Esto demuestra que el funcional está acotado en $\Poly$. Por el Teorema de Extensión de Operadores Lineales Densos, se extiende de manera continua a $\mathcal{H}$.
\end{proof}

\begin{observacion}[Átomos Exteriores]
    Para los átomos exteriores ($k \in I_{out}$), el funcional de evaluación $\delta_{c_k}$ no es continuo respecto a la norma $L^2$ sobre $S_R$. En consecuencia, su núcleo algebraico no es necesariamente cerrado en $\mathcal{H}$. Por este motivo, los subespacios asociados a condiciones de anulación en átomos exteriores se definen de forma algebraica y se considera su clausura topológica. Esta clausura preserva las propiedades dimensionales requeridas sobre el dominio denso de polinomios.
\end{observacion}

\begin{definicion}[Subespacio Estable $W$]
    Definimos el subespacio estable algebraico $W_0$ como el conjunto de polinomios que se anulan en los átomos exteriores:
    \begin{equation}
        W_0 = \{p \in \Poly : p(c_k) = 0, \forall k \in I_{out}\}.
    \end{equation}
    El subespacio estable $W \subset \mathcal{H}$ se define como la clausura topológica de $W_0$ respecto a la norma de Sobolev: $W = \overline{W_0}^{\norm{\cdot}}$.
\end{definicion}

\begin{definicion}[Subespacio Totalmente Restringido $W'$]
    Definimos el subespacio algebraico $W'_0$ imponiendo restricciones nulas en la totalidad de los átomos:
    \begin{equation}
        W'_0 = \{p \in \Poly : p(c_k) = 0, \forall k \in \{1, \dots, N\}\}.
    \end{equation}
    Y definimos el subespacio de Hilbert correspondiente como su clausura: $W' = \overline{W'_0}^{\norm{\cdot}}$.
\end{definicion}

\begin{definicion}[Codimensión]
    Sea $M$ un subespacio vectorial de un espacio de Banach $X$. La codimensión de $M$ en $X$, denotada por $\operatorname{codim}(M)$, se define como la dimensión algebraica del espacio vectorial cociente $X/M$:
    $$ \operatorname{codim}(M) = \dim(X/M). $$
    En el caso particular de que $X$ sea un espacio de Hilbert y $M$ sea un subespacio cerrado, esta definición coincide con la dimensión del complemento ortogonal $M^\perp$.
\end{definicion}

\begin{proposicion}[Codimensión de Intersecciones de Núcleos]
    Sea $\mathcal{S} = \{z_1, \dots, z_k\} \subset \C$ un conjunto finito de $k$ puntos distintos. Definimos el subespacio $V_{\mathcal{S}} \subset \Poly$ como el conjunto de polinomios que se anulan en todos los puntos de $\mathcal{S}$:
    $$ V_{\mathcal{S}} = \{ p \in \Poly : p(z) = 0, \quad \forall z \in \mathcal{S} \} $$
    Entonces, el subespacio $V_{\mathcal{S}}$ tiene codimensión exacta $k$ en $\Poly$.
\end{proposicion}

\begin{proof}
    Asociamos a cada punto $z_i \in \mathcal{S}$ el funcional lineal de evaluación $\delta_{z_i} \in \Poly^*$. El subespacio $V_{\mathcal{S}}$ es, por definición, la intersección de los núcleos de estos funcionales: $V_{\mathcal{S}} = \bigcap_{i=1}^k \ker(\delta_{z_i})$.

    Para determinar la codimensión, primero demostramos que el conjunto de funcionales $\{\delta_{z_1}, \dots, \delta_{z_k}\}$ es linealmente independiente. Supongamos que existe una combinación lineal nula $\sum_{i=1}^k \alpha_i \delta_{z_i} = 0_{\Poly^*}$.
    Para cada índice $j \in \{1, \dots, k\}$, construimos explícitamente el polinomio de interpolación de Lagrange asociado al conjunto $\mathcal{S}$:
    \begin{equation}
        L_j(z) = \prod_{\substack{l=1 \\ l \neq j}}^{k} \frac{z - z_l}{z_j - z_l}
    \end{equation}
    Por construcción, estos polinomios satisfacen la propiedad de ortogonalidad puntual $L_j(z_i) = \delta_{ji}$ (Delta de Kronecker). Evaluando la combinación lineal supuesta en el polinomio $L_j$, obtenemos:
    $$ 0 = \sum_{i=1}^k \alpha_i \delta_{z_i}(L_j) = \sum_{i=1}^k \alpha_i L_j(z_i) = \alpha_j \cdot 1 $$
    Por tanto, $\alpha_j = 0$ para todo $j$, lo que demuestra la independencia lineal.

    Finalmente, definimos la aplicación lineal de evaluación conjunta $\Phi: \Poly \to \C^k$ dada por $\Phi(p) = (p(z_1), \dots, p(z_k))$.
    Dado que los funcionales componentes son independientes, la aplicación es sobreyectiva. Aplicando el Primer Teorema de Isomorfía:
    $$ \Poly / V_{\mathcal{S}} \cong \C^k \implies \operatorname{codim}(V_{\mathcal{S}}) = \dim(\C^k) = k $$
\end{proof}

\begin{corolario}
    Considerando los conjuntos de átomos definidos anteriormente:
    \begin{enumerate}
        \item El subespacio estable $W$, que anula los $m$ átomos inestables ($I_{out}$), tiene codimensión $m$.
        \item El subespacio totalmente restringido $W'$, que anula la totalidad de los $N$ átomos ($\mathcal{I}$), tiene codimensión $N$.
    \end{enumerate}
\end{corolario}

Dado que nuestra medida $\mu$ es la medida de Lebesgue en la circunferencia unidad $S_1$, la componente continua de nuestro espacio de Sobolev está ligada al espacio de Hardy\cite{duren1970theory}.

\begin{definicion}[Espacio de Hardy $H^2(\mathbb{D})$]
    El espacio de Hardy $H^2(\mathbb{D})$ se define como el espacio de funciones holomorfas $f$ en el disco unidad abierto $\mathbb{D} = \{z \in \C : \abs{z} < 1\}$ tales que sus medias cuadráticas sobre circunferencias concéntricas permanecen acotadas:
    $$ \sup_{0 \le r < 1} \frac{1}{2\pi} \int_{0}^{2\pi} \abs{f(re^{i\theta})}^2 d\theta < \infty. $$
\end{definicion}

Existe un isomorfismo isométrico natural entre $H^2(\mathbb{D})$ y el subespacio cerrado de $L^2(S_1)$ formado por funciones cuyos coeficientes de Fourier de índice negativo son nulos. En nuestro contexto de aproximación polinómica, podemos caracterizar $H^2$ de manera equivalente como la clausura del espacio de polinomios $\Poly$ bajo la norma $L^2$ estándar en la frontera:
$$ H^2 \cong \overline{\Poly}^{L^2(S_1)}. $$

Esta identificación es fundamental , ya que nos permite utilizar la estructura de \textit{Reproducing Kernel Hilbert Space} (RKHS) de $H^2$. En particular, para cualquier $c \in \mathbb{D}$, el núcleo reproductor de Szegő viene dado por $K_c(z) = (1-\bar{c}z)^{-1}$.

\section{Acotación del Operador Multiplicación}

\begin{teorema}[Acotación Condicionada]
    Sea $D: \Poly \to \Poly$ el operador de multiplicación definido por $D(p)(z) = z p(z)$. La restricción de este operador al subespacio estable $W$, denotada como $D|_W : W \to \mathcal{H}$, es un operador acotado. Es decir, existe una constante $K < \infty$ tal que:
    $$ \norm{Dp} \leq K \norm{p}, \quad \forall p \in W. $$
\end{teorema}

\begin{proof}
    Sea $p \in W$. Examinamos el cuadrado de la norma de la imagen $\norm{Dp}^2$ descomponiéndola en su parte integral y su parte discreta:
    $$ \norm{Dp}^2 = \int_{S_R} \abs{z p(z)}^2 d\mu(z) + \sum_{k=1}^{N} \abs{(zp)'(c_k)}^2. $$

    \textbf{1. Acotación de la parte integral:}
    Dado que el soporte de la medida $\mu$ es la circunferencia $S_R$, tenemos $\abs{z} = R$ en todo el dominio de integración. Por tanto:
    $$ \int_{S_R} \abs{z p(z)}^2 d\mu(z) = R^2 \int_{S_R} \abs{p(z)}^2 d\mu(z) \leq R^2 \norm{p}^2. $$

    \textbf{2. Acotación de la parte discreta:}
    Utilizamos la regla de la derivada del producto $(zp)'(z) = p(z) + z p'(z)$. Dividimos la suma de átomos en los conjuntos $I_{in}$ (interiores) e $I_{out}$ (exteriores).

    \textit{Caso A: Átomos Exteriores ($k \in I_{out}$)}.
    Por definición del subespacio estable $W$, si $p \in W$, entonces $p(c_k) = 0$ para todo $k \in I_{out}$. La expresión evaluada se reduce a:
    $$ \abs{p(c_k) + c_k p'(c_k)}^2 = \abs{0 + c_k p'(c_k)}^2 = \abs{c_k}^2 \abs{p'(c_k)}^2. $$

    \textit{Caso B: Átomos Interiores ($k \in I_{in}$)}.
    En este caso no podemos asumir que $p(c_k)$ se anule. Utilizamos la desigualdad triangular $(a+b)^2 \leq 2a^2 + 2b^2$:
    $$ \abs{p(c_k) + c_k p'(c_k)}^2 \leq 2\abs{p(c_k)}^2 + 2\abs{c_k}^2 \abs{p'(c_k)}^2. $$
    Por la Proposición de Evaluación Puntual (ver Preliminares), sabemos que la evaluación es un funcional acotado. Existe una constante $M_k > 0$ tal que $\abs{p(c_k)}^2 \leq M_k \norm{p}^2$.

    \textbf{3. Combinación de cotas:}
    Sustituyendo las cotas obtenidas en la expresión de la norma total:
    \begin{align*}
        \norm{Dp}^2 & \leq R^2 \norm{p}^2 + \sum_{k \in I_{out}} \abs{c_k}^2 \abs{p'(c_k)}^2 + \sum_{k \in I_{in}} \left( 2 M_k \norm{p}^2 + 2 \abs{c_k}^2 \abs{p'(c_k)}^2 \right) \\
                    & \leq \left( R^2 + \max_{k \in I_{out}}\abs{c_k}^2 + 2\sum_{k \in I_{in}} M_k + 2\max_{k \in I_{in}}\abs{c_k}^2 \right) \norm{p}^2.
    \end{align*}
    Nótese que hemos usado que $\abs{p'(c_k)}^2 \leq \norm{p}^2$ por definición de la norma Sobolev. Si definimos $K$ como la raíz cuadrada del término entre paréntesis (que es una constante finita dependiente solo de $R$ y los átomos), concluimos que $\norm{Dp} \leq K \norm{p}$.
\end{proof}

\begin{corolario}[Acotación del Espectro Remanente]
    Como consecuencia directa de la acotación del operador en $W$, el $(m+1)$-ésimo valor singular asintótico del operador de multiplicación, denotado como $\sigma_{m+1}$, está acotado superiormente por la norma de la restricción:
    \begin{equation}
        \sigma_{m+1} \le \|\D|_W\| < \infty
    \end{equation}
\end{corolario}

\begin{proof}
    Utilizamos la caracterización variacional de los valores singulares dada por el Principio Min-Max. El $(m+1)$-ésimo valor singular se define como el ínfimo de la norma del operador sobre todos los subespacios posibles de codimensión $m$:
    $$ \sigma_{m+1} = \inf \{ \|\D|_V\| : V \subset \mathcal{H}, \, \operatorname{codim}(V) = m \} $$
    En la Definición del subespacio estable, establecimos que $W$ tiene precisamente codimensión $m$ (al ser intersección de núcleos de funcionales linealmente independientes en un espacio de Hilbert). Por lo tanto, $W$ pertenece al conjunto de subespacios sobre los que se toma el ínfimo.
    Por la propiedad fundamental del ínfimo, el valor evaluado en un elemento específico del conjunto ($W$) siempre es una cota superior del ínfimo global:
    $$ \sigma_{m+1} \le \|\D|_W\| $$
    Dado que el Teorema de Acotación Condicionada demuestra que $\|\D|_W\|$ es finito, concluimos que $\sigma_{m+1}$ es finito.
\end{proof}

\section{Divergencia de los Valores Singulares Inestables}

Los átomos situados fuera del disco soporte de la medida continua generan un comportamiento singular en el operador. A diferencia del caso acotado, estos puntos inducen valores singulares infinitos.

\subsection{Persistencia de la Codimensión}

El siguiente resultado establece que la imposición de restricciones lineales continuas sobre el subespacio denso $\Poly$ preserva la codimensión tras tomar la clausura en el espacio de Hilbert.

\begin{proposicion}[Persistencia de la Codimensión bajo Clausura]
    Sea $\mathcal{H}$ un espacio de Hilbert y $\Poly$ un subespacio denso. Sean $\ell_1, \dots, \ell_N$ funcionales lineales linealmente independientes definidos en $\Poly$ que son continuos respecto a la norma $\norm{\cdot}$.
    Consideremos el subespacio algebraico:
    $$ W_0 = \{ p \in \Poly : \ell_k(p) = 0, \, \forall k=1,\dots,N \}. $$
    Si definimos $W = \overline{W_0}$ como la clausura de $W_0$ en $\mathcal{H}$, entonces $W$ es un subespacio cerrado y su codimensión es exactamente $N$:
    $$ \operatorname{codim}(W) = N. $$
\end{proposicion}

\begin{proof}
    Dado que cada funcional $\ell_k$ es continuo y lineal en el subespacio denso $\Poly$, por el Teorema de Extensión de Operadores Lineales Acotados (BLT Theorem), existe una única extensión continua $\tilde{\ell}_k \in \mathcal{H}^*$ para cada $k$.
    La propiedad de continuidad implica que la anulación del funcional se preserva en el límite, por lo que la clausura del núcleo algebraico coincide con el núcleo de la extensión en todo el espacio:
    $$ W = \overline{\bigcap_{k=1}^N \ker(\ell_k)} = \bigcap_{k=1}^N \ker(\tilde{\ell}_k). $$
    Definimos el operador lineal acotado $\Phi: \mathcal{H} \to \C^N$ mediante $\Phi(f) = (\tilde{\ell}_1(f), \dots, \tilde{\ell}_N(f))$. Es evidente que $\ker(\Phi) = W$.
    Por el Primer Teorema de Isomorfía para espacios vectoriales, tenemos la identificación algebraica:
    $$ \mathcal{H}/W = \mathcal{H}/\ker(\Phi) \cong \operatorname{Im}(\Phi). $$
    Puesto que los funcionales originales eran linealmente independientes, sus extensiones también lo son, lo que implica que el operador $\Phi$ es sobreyectivo sobre $\C^N$ (es decir, $\operatorname{Im}(\Phi) = \C^N$).
    Por tanto, $\dim(\mathcal{H}/W) = \dim(\C^N) = N$, lo que concluye la demostración.
\end{proof}

\begin{corolario}
    Si los átomos $\{c_k\}_{k=1}^N$ son puntos de evaluación acotada (BPE), el subespacio $W = \overline{\{p \in \Poly : (zp)'(c_k)=0\}}$ tiene codimensión exactamente $N$ en $\mathcal{H}$.
\end{corolario}

\begin{proposicion}[Divergencia de $s$-números]
    \cite{gohberg1969introduction} Sea $D$ el operador de multiplicación por $z$. Sea $m = |I_{out}|$ el número de átomos inestables. Entonces, los primeros $m$ valores singulares (números de aproximación) del operador son infinitos:
    \begin{equation}
        \sigma_1 = \sigma_2 = \dots = \sigma_m = \infty.
    \end{equation}
\end{proposicion}

\begin{proof}
    Utilizamos la caracterización variacional de los valores singulares (Principio Min-Max). El $m$-ésimo valor singular está dado por:
    $$ \sigma_m = \inf \{ \norm{D|_V} : V \subset \mathcal{H}, \operatorname{codim}(V) = m-1 \}. $$
    Para demostrar que $\sigma_m = \infty$, probaremos que para cualquier subespacio $V$ de codimensión $m-1$, el operador es no acotado sobre $V$.

    Sea $V \subset \mathcal{H}$ un subespacio arbitrario con $\operatorname{codim}(V) = m-1$.
    Consideremos el subespacio estable $W$ definido en la sección anterior, el cual anula los $m$ átomos inestables. Sabemos que $\operatorname{codim}(W) = m$. Por razones de dimensionalidad (propiedades de los complementos ortogonales en espacios de Hilbert), es imposible que $V$ esté contenido en $W$, ya que $\operatorname{codim}(V) < \operatorname{codim}(W)$.

    Esto implica que existe al menos un átomo inestable $c^* \in I_{out}$ tal que el funcional de evaluación $\delta_{c^*}$ no se anula idénticamente sobre $V$. En consecuencia, el operador actúa sobre $V$ de forma no acotada, ya que el módulo del átomo $c^*$ es estrictamente mayor que $R$.

    Para formalizar la divergencia, construimos la sucesión de polinomios prueba $\{Q_n\}_{n \geq 1}$ definida por:
    $$ Q_n(z) = (n+1)c^* z^n - n z^{n+1}. $$
    Esta sucesión está diseñada para anular la derivada en el punto crítico, $Q_n'(c^*) = 0$, minimizando la norma de entrada, mientras maximiza el valor de la función $Q_n(c^*) = (c^*)^{n+1}$.

    Analizamos el cociente de Rayleigh asintótico $\frac{\norm{DQ_n}}{\norm{Q_n}}$:

    \textbf{1. Norma de entrada ($\norm{Q_n}$):}
    Dado que $Q_n'(c^*) = 0$, la parte discreta en $c^*$ se anula. La norma está dominada por la parte integral ($L^2$):
    $$ \norm{Q_n}^2 \approx \int_{S_R} \abs{(n+1)c^* z^n - n z^{n+1}}^2 d\mu \approx C \cdot n^2 R^{2n}. $$

    \textbf{2. Norma de salida ($\norm{DQ_n}$):}
    La derivada de la imagen es $(zQ_n)'(c^*) = Q_n(c^*) + c^* Q_n'(c^*) = (c^*)^{n+1}$.
    La contribución de este término atómico a la norma es:
    $$ \abs{(zQ_n)'(c^*)}^2 = \abs{c^*}^{2n+2}. $$

    \textbf{3. Divergencia del cociente:}
    Comparando los órdenes de magnitud cuando $n \to \infty$:
    $$ \frac{\norm{DQ_n}^2}{\norm{Q_n}^2} \approx \frac{\abs{c^*}^{2n+2}}{n^2 R^{2n}} = \frac{\abs{c^*}^2}{n^2} \left( \frac{\abs{c^*}}{R} \right)^{2n}. $$
    Puesto que $c^* \in I_{out}$, tenemos $\abs{c^*} > R$. El término exponencial $(\abs{c^*}/R)^{2n}$ domina sobre el factor polinomial $n^{-2}$, haciendo que el cociente tienda a infinito.

    Dado que la restricción $\delta_{c^*}|_V$ es no trivial, existen elementos en $V$ cuyo comportamiento bajo $D$ reproduce la singularidad en $c^*$. En particular, los polinomios $\{Q_n\}$ pueden aproximarse mediante elementos de $V$, de donde se concluye que $\sup_{p \in V} \frac{\norm{Dp}}{\norm{p}} = \infty$. Por tanto, $\sigma_m = \infty$.
\end{proof}

\begin{proposicion}[Dominancia del Soporte]
    Sea $W'$ el subespacio totalmente restringido, definido como la clausura en $\mathcal{H}$ del conjunto de polinomios que se anulan en todos los átomos $\{c_1, \dots, c_N\}$ (véase la Sección~1).
    Sea $\D$ el operador de multiplicación por z, definido como $\D(p) = z p(z)$. La norma de la restricción de este operador al subespacio $W'$, denotada como $\D|_{W'} : W' \to \mathcal{H}$, es:
    \begin{equation}
        \norm{\D p} \le \max \left( R, \max_{k=1,\dots,N} |c_k| \right) \norm{p} \quad \forall p \in W'
    \end{equation}
\end{proposicion}

\begin{proof}
    Sea $p \in W'$. Evaluamos la norma al cuadrado de la imagen $\norm{zp}^2$. Para la parte continua tenemos:
    $$\int_{|z|=R} |z p(z)|^2 d\mu \le R^2 \int_{|z|=R} |p(z)|^2 d\mu = R^2 \|p\|_{L^2}^2.$$
    La condición $p \in W'$ implica $p(c_k)=0$ para todo $k$ (por construcción de la clausura y continuidad de la evaluación en los átomos). Al evaluar la derivada de la imagen $(zp)'$ en los átomos, el término libre desaparece:
    $$ (zp)'(c_k) = p(c_k) + c_k p'(c_k) = c_k p'(c_k) $$
    Al sumar sobre todos los átomos, se acota la contribución utilizando el módulo del átomo de mayor módulo:
    $$ \sum_{k \in \mathcal{I}}  |c_k|^2 |p'(c_k)|^2 \le \left( \max_{k \in \mathcal{I}} |c_k|^2 \right) \sum_{k \in \mathcal{I}}  |p'(c_k)|^2. $$
    Dado que la suma parcial de las derivadas ponderadas es menor o igual que la norma total $\norm{p}^2$, este término queda acotado.
    Definimos la constante $K$ como:
    $$ K = \max \left( R, \max_{k \in \mathcal{I}} |c_k| \right). $$
    Sustituyendo esta cota en el numerador, observamos que tanto el coeficiente de la parte integral ($R^2$) como el coeficiente de la parte atómica ($\max |c_k|^2$) son menores o iguales a $K^2$:
    \begin{align*}
        \frac{\|\D p\|_S^2}{\|p\|_S^2} & \le \frac{K^2 \|p\|_{L^2}^2 + K^2 \sum_{k \in \mathcal{I}}  |p'(c_k)|^2}{\|p\|_{S}^2}          \\
                                       & = \frac{K^2 \left( \|p\|_{L^2}^2 + \sum_{k \in \mathcal{I}}  |p'(c_k)|^2 \right)}{\|p\|_{S}^2} \\
                                       & = \frac{K^2 \|p\|_S^2}{\|p\|_S^2} = K^2.
    \end{align*}
    Tomando la raíz cuadrada, concluimos que la norma del operador restringido satisface la desigualdad:
    $$\|\D|_{W'}\| \le K = \max \left( R, \max_{k=1,\dots,N} |c_k| \right).$$
    Esto demuestra que el operador restringido es acotado, con cota determinada por el máximo entre el radio del soporte continuo y el módulo del átomo más alejado del origen.
\end{proof}

\begin{corolario}[Acotación del Espectro Restringido]
    Como consecuencia directa de la acotación del operador en $W'$, el $(N+1)$-ésimo valor singular asintótico del operador de multiplicación, denotado como $\sigma_{N+1}$, está acotado superiormente por la norma de la restricción:
    \begin{equation}
        \sigma_{N+1} \le \|\D|_W'\| \le \max \left( R, \max_{k=1,\dots,N} |c_k| \right) < \infty
    \end{equation}
\end{corolario}

\section{Análisis Espectral: El Caso de un Único Átomo}

Con el fin de analizar el impacto que las derivadas discretas tienen sobre la estructura espectral del operador de multiplicación $D(p) = zp$, se estudia en primer lugar el caso particular de un único átomo en la derivada. Específicamente, se considera la perturbación de la medida continua, soportada en la circunferencia unidad $S_1$, mediante la adición de un único término discreto en la derivada, evaluado en un punto $c_1 \in \C$.

En el marco de la teoría de ideales de operadores, los números de Gelfand constituyen la herramienta natural para cuantificar el comportamiento espectral direccional de un operador acotado. Para un operador $T \in \mathcal{L}(\mathcal{H})$, el $m$-ésimo número de Gelfand se define como:
\begin{equation}
    c_m(T) = \inf \{ \norm{T|_V} : V \subset \mathcal{H}, \, \operatorname{codim}(V) < m \}
\end{equation}

A lo largo de esta sección, se asume que el átomo $c_1$ constituye una \textit{Bounded Point Evaluation} (BPE). Como se estableció en la sección de preliminares, esta hipótesis es necesaria para garantizar que el operador $D$ sea globalmente acotado en $\mathcal{H}$.

\subsection{El Primer Número de Gelfand: La Norma Global}

Por definición, el primer número de Gelfand, $c_1(D)$, corresponde al ínfimo sobre subespacios de codimensión estrictamente menor que 1. El único subespacio que cumple esta condición es el espacio total $\mathcal{H}$. Por consiguiente:
$$ c_1(D) = \norm{D} $$

Este primer número refleja el comportamiento global del operador. Analíticamente, se verifica que $c_1(D)$ depende de la posición del átomo $c_1$: cuanto mayor es la contribución del término discreto a la norma de Sobolev, mayor es la dilatación de $c_1(D)$ respecto al radio de la circunferencia.

\subsection{Un ejemplo ilustrativo: El átomo en el origen}

Consideremos el espacio de polinomios $\Poly$ dotado de la medida de Lebesgue normalizada en la circunferencia unidad $S_1$ y un único átomo en la derivada situado en el origen ($c_1 = 0$). El producto interno de Sobolev toma la forma:
\begin{equation}
    \ip{p}{q} = \int_{S_1} p(z)\overline{q(z)} \frac{\abs{dz}}{2\pi} + p'(0)\overline{q'(0)}.
\end{equation}

Sea $p(z) = \sum_{k=0}^n a_k z^k$ un polinomio arbitrario. Gracias a la ortonormalidad de la base de monomios $\{z^k\}$ respecto a la medida de Lebesgue, la parte continua de la norma es simplemente la suma de los módulos al cuadrado de sus coeficientes. Por otro lado, la derivada evaluada en el origen es exactamente el coeficiente de grado uno: $p'(0) = a_1$.
Por lo tanto, el cuadrado de la norma de Sobolev de entrada es:
\begin{equation} \label{eq:norma_in_0}
    \norm{p}^2 = \sum_{k=0}^n \abs{a_k}^2 + \abs{p'(0)}^2 = \sum_{k=0}^n \abs{a_k}^2 + \abs{a_1}^2 = \abs{a_0}^2 + 2\abs{a_1}^2 + \sum_{k=2}^n \abs{a_k}^2.
\end{equation}

Veamos ahora el efecto del operador de multiplicación $\D(p)(z) = z p(z)$. El polinomio resultante es $\D p(z) = \sum_{k=0}^n a_k z^{k+1}$. Su derivada en el origen, por inspección directa del coeficiente de grado uno del polinomio resultante, es $(\D p)'(0) = p(0) = a_0$.
El cuadrado de la norma de salida resulta ser:
\begin{equation} \label{eq:norma_out_0}
    \norm{\D p}^2 = \sum_{k=0}^n \abs{a_k}^2 + \abs{(\D p)'(0)}^2 = \sum_{k=0}^n \abs{a_k}^2 + \abs{a_0}^2 = 2\abs{a_0}^2 + \abs{a_1}^2 + \sum_{k=2}^n \abs{a_k}^2.
\end{equation}

En la teoría de ideales de operadores, la norma global del operador coincide con su primer número de Gelfand, $c_1(\D)$, definido como el supremo del cociente de Rayleigh. Utilizando \eqref{eq:norma_in_0} y \eqref{eq:norma_out_0}:
\begin{equation}
    c_1(\D)^2 = \sup_{p \neq 0} \frac{\norm{\D p}^2}{\norm{p}^2} = \sup_{a_k} \frac{2\abs{a_0}^2 + \abs{a_1}^2 + \sum_{k \ge 2} \abs{a_k}^2}{\abs{a_0}^2 + 2\abs{a_1}^2 + \sum_{k \ge 2} \abs{a_k}^2}.
\end{equation}

Para determinar el supremo de este cociente, se reescribe el numerador en función del denominador. Denotando $S = \norm{p}^2 = \abs{a_0}^2 + 2\abs{a_1}^2 + \sum_{k \ge 2} \abs{a_k}^2$, se expresa el numerador como:
\begin{align*}
    2\abs{a_0}^2 + \abs{a_1}^2 + \sum_{k \ge 2} \abs{a_k}^2 & = 2\left( \abs{a_0}^2 + 2\abs{a_1}^2 + \sum_{k \ge 2} \abs{a_k}^2 \right) - 3\abs{a_1}^2 - \sum_{k \ge 2} \abs{a_k}^2 \\
                                                            & = 2S - 3\abs{a_1}^2 - \sum_{k \ge 2} \abs{a_k}^2.
\end{align*}

Sustituyendo esta expresión algebraica en el cociente de Rayleigh, obtenemos:
\begin{equation}
    \frac{\norm{\D p}^2}{\norm{p}^2} = \frac{2S - 3\abs{a_1}^2 - \sum_{k \ge 2} \abs{a_k}^2}{S} = 2 - \frac{3\abs{a_1}^2 + \sum_{k \ge 2} \abs{a_k}^2}{\norm{p}^2}.
\end{equation}

Dado que los módulos al cuadrado son siempre cantidades no negativas y $\norm{p}^2 > 0$ (para cualquier $p \neq 0$), el término fraccionario que se resta al 2 es siempre mayor o igual a cero. Esto establece la siguiente cota superior para el cociente:
$$ \frac{\norm{\D p}^2}{\norm{p}^2} \le 2. $$

El supremo (y en este caso, el máximo) se alcanza si y solo si el término sustractivo es exactamente nulo. Esto exige matemáticamente que $a_1 = 0$ y $a_k = 0$ para todo $k \ge 2$. La única variable que queda libre es $a_0 \neq 0$, lo que implica que el subespacio de polinomios que maximizan la acción del operador está formado exclusivamente por las funciones constantes, como por ejemplo $p(z) = 1$.

Verificamos la igualdad evaluando el cociente en esta función constante:
$$ \frac{\norm{\D(1)}^2}{\norm{1}^2} = 2. $$
Concluimos formalmente que la norma global del operador es $c_1(\D) = \sqrt{2}$.

\subsection{Dependencia de la Posición: El caso de un átomo arbitrario}

El caso $c=0$ produce una norma global $c_1(\D) = \sqrt{2}$, alcanzada por los polinomios constantes. A continuación se analiza el caso de un átomo interior arbitrario $c \neq 0$. Este análisis requiere consideraciones adicionales, ya que los núcleos reproductores del espacio de Hardy $H^2$ no coinciden con la base de monomios cuando el punto de evaluación no es el origen.

Sea $c \in \C$ un átomo interior con $0 < \abs{c} < 1$. Para aplicar la caracterización matricial del primer número de Gelfand, necesitamos los representantes de Riesz de la evaluación funcional y de la derivada en el punto $c$. Para la medida de Lebesgue en la circunferencia, estos son los conocidos núcleos de Szegő:
\begin{equation}
    K_0(z) = \frac{1}{1 - \bar{c}z}, \quad K_1(z) = \frac{\partial}{\partial \bar{c}} K_0(z) = \frac{z}{(1 - \bar{c}z)^2}.
\end{equation}

El cálculo exacto de $c_1(\D)$ se reduce a resolver el problema de autovalores generalizados para las matrices de Gram de estos núcleos. No obstante, es posible obtener cotas inferiores evaluando el cociente de Rayleigh en polinomios específicos.

Evaluamos el cociente de Rayleigh en dos polinomios naturales: la función constante y la función identidad.

\textbf{1. El candidato constante ($p_0(z) = 1$):}
Como se demostró anteriormente, $\norm{1}^2 = 1$ y $\norm{\D(1)}^2 = 2$. El cociente de Rayleigh permanece invariante respecto a la posición del átomo:
$$ \mathcal{R}(1) = \frac{\norm{\D(1)}^2}{\norm{1}^2} = 2. $$

\textbf{2. El candidato lineal ($p_1(z) = z$):}
Calculamos las normas involucradas para este polinomio cuando $c \neq 0$:
\begin{itemize}
    \item Derivada en el átomo: $p_1'(c) = 1$.
    \item Derivada de la imagen: $(\D p_1)'(c) = (z^2)'(c) = 2c$.
    \item Norma de entrada: $\norm{z}^2 = \|z\|_{L^2}^2 + \abs{1}^2 = 1 + 1 = 2$.
    \item Norma de salida: $\norm{z^2}^2 = \|z^2\|_{L^2}^2 + \abs{2c}^2 = 1 + 4\abs{c}^2$.
\end{itemize}
El cociente de Rayleigh para el polinomio lineal es:
$$ \mathcal{R}(z) = \frac{1 + 4\abs{c}^2}{2} = \frac{1}{2} + 2\abs{c}^2. $$

La comparación de ambos candidatos revela un cambio cualitativo en el comportamiento del supremo.
\begin{itemize}
    \item Si $\abs{c}$ es pequeño, el término $\frac{1}{2} + 2\abs{c}^2$ es menor que 2, por lo que el supremo se alcanza en el polinomio constante.
    \item Sin embargo, existe un punto crítico cuando $\frac{1}{2} + 2\abs{c}^2 = 2$, es decir, cuando $\abs{c} = \frac{\sqrt{3}}{2} \approx 0.866$.
\end{itemize}

Para átomos situados cerca de la frontera ($|c| > \sqrt{3}/2$), el polinomio $p(z)=z$ sufre una dilatación mayor que el polinomio constante. Esto demuestra rigurosamente que:
\begin{equation}
    c_1(\D) \ge \max \left( \sqrt{2}, \, \sqrt{\frac{1}{2} + 2\abs{c}^2} \right).
\end{equation}

Este resultado tiene la siguiente interpretación geométrica: a medida que el átomo se acerca a la frontera ($|c| \to 1$), el primer número de Gelfand crece, y la dirección que alcanza la norma pasa de las funciones constantes a funciones con mayor dependencia en $z$, alineándose con la singularidad del núcleo $K_1(z)$.

\subsection{Caracterización Exacta del Primer Número de Gelfand}

Hemos establecido que $c_1(D) = \norm{D}$. Si bien el cálculo de la norma de un operador en un espacio de dimensión infinita es, en general, un problema de difícil resolución, la estructura del producto de Sobolev y la hipótesis BPE permiten reducir este cálculo a un problema de optimización en dimensión finita.

El cálculo de la norma global equivale a encontrar el supremo del cociente de Rayleigh:
\begin{equation} \label{eq:rayleigh_c1}
    c_1(D)^2 = \sup_{p \in \Poly, p \neq 0} \frac{\norm{Dp}^2}{\norm{p}^2} = \sup_{p \neq 0} \frac{\|p\|_{L^2(\mu)}^2 + \abs{p(c_1) + c_1 p'(c_1)}^2}{\|p\|_{L^2(\mu)}^2 + \abs{p'(c_1)}^2}.
\end{equation}
Nótese que hemos utilizado el hecho de que, sobre la circunferencia $S_1$, la norma $L^2$ es invariante bajo la multiplicación por $z$, es decir, $\|zp\|_{L^2(\mu)} = \|p\|_{L^2(\mu)}$.

Dado que $c_1$ es un átomo BPE, tanto la evaluación de la función como la de su derivada son funcionales continuos en la completación de $\Poly$ respecto a la norma $L^2(\mu)$ (el espacio de Hardy $H^2(\mu)$). Por el Teorema de Representación de Riesz, existen dos núcleos reproductores $K_0, K_1 \in H^2(\mu)$ tales que, para todo polinomio $p$:
\begin{align*}
    p(c_1)  & = \langle p, K_0 \rangle_{L^2(\mu)}, \\
    p'(c_1) & = \langle p, K_1 \rangle_{L^2(\mu)}.
\end{align*}

Definimos el subespacio bidimensional generado por estos representadores, $V_K = \operatorname{span}\{K_0, K_1\}$. Por el Teorema de Proyección Ortogonal, cualquier polinomio $p$ se puede descomponer de forma única como:
$$ p(z) = \underbrace{\alpha K_0(z) + \beta K_1(z)}_{p_K \in V_K} + \underbrace{p_{\perp}(z)}_{p_{\perp} \perp V_K}, $$
donde $\alpha, \beta \in \C$. La ortogonalidad de $p_{\perp}$ respecto a $K_0$ y $K_1$ implica que se anula al ser evaluado: $p_{\perp}(c_1) = 0$ y $p_{\perp}'(c_1) = 0$.

Sustituyendo esta descomposición en el cociente de Rayleigh \eqref{eq:rayleigh_c1}, observamos que la parte ortogonal $p_{\perp}$ no contribuye en absoluto a la parte discreta (evaluaciones), pero contribuye a la norma $L^2$:
$$ \frac{\norm{Dp}^2}{\norm{p}^2} = \frac{\|p_K\|_{L^2(\mu)}^2 + \abs{p_K(c_1) + c_1 p_K'(c_1)}^2 + \|p_{\perp}\|_{L^2(\mu)}^2}{\|p_K\|_{L^2(\mu)}^2 + \abs{p_K'(c_1)}^2 + \|p_{\perp}\|_{L^2(\mu)}^2}. $$

Sea $N_0$ el numerador evaluado en $p_K$ y $D_0$ el denominador evaluado en $p_K$. Si denotamos $x = \|p_{\perp}\|_{L^2(\mu)}^2 \geq 0$, estamos estudiando la función $f(x) = \frac{N_0 + x}{D_0 + x}$.
Sabemos que el operador es expansivo para el primer valor singular, es decir, el supremo del cociente es estrictamente mayor que 1 (lo cual implica que existe $p_K$ donde $N_0 > D_0$). Bajo esta condición, la derivada $f'(x) = \frac{D_0 - N_0}{(D_0 + x)^2}$ es estrictamente negativa. Esto significa que para maximizar el cociente, debemos minimizar $x$. El máximo absoluto se alcanza necesariamente cuando $x = 0$, es decir, cuando $p_{\perp} = 0$.

Por lo tanto, la búsqueda en dimensión infinita colapsa a una optimización en el subespacio de dimensión dos $V_K$. Identificando $p_K$ con el vector de coeficientes $v = (\alpha, \beta)^T \in \C^2$, las normas se expresan como formas cuadráticas:
\begin{align*}
    \norm{p_K}^2   & = v^* A v, \\
    \norm{D p_K}^2 & = v^* B v,
\end{align*}
donde $A$ y $B$ son matrices hermíticas de orden $2 \times 2$ cuyas entradas dependen exclusivamente de los productos escalares $\langle K_i, K_j \rangle_{L^2}$ y las evaluaciones de los núcleos en el átomo $c_1$.

El cálculo de $c_1(D)^2$ se reduce entonces a encontrar el mayor autovalor generalizado del haz de matrices $(B, A)$, es decir, la mayor raíz $\lambda$ de la ecuación característica bidimensional:
\begin{equation}
    \det(B - \lambda A) = 0.
\end{equation}
Esta reducción justifica rigurosamente por qué el primer número de Gelfand es computable y finito bajo la hipótesis BPE, dependiendo únicamente de la geometría local en el átomo y de la función de Christoffel-Darboux (o núcleo de Szegő) asociada a la medida continua.

Cabe preguntarse si, al neutralizar esta fuente de perturbación mediante la restricción a un subespacio adecuado, es posible determinar el valor del número de Gelfand residual.

\subsection{Cálculo Exacto: La Solución Matricial General}

Para obtener un valor explícito del primer número de Gelfand $c_1(\D)$ para cualquier posición del átomo $c \in \mathbb{D}$, debemos resolver el problema de autovalores generalizados asociado al subespacio óptimo.

Como el supremo se alcanza en el subespacio generado por los núcleos reproductores del espacio de Hardy $H^2$, denotemos la base por $\mathcal{B} = \{K_0, K_1\}$, donde:
\begin{equation}
    K_0(z) = \frac{1}{1-\bar{c}z}, \quad K_1(z) = \frac{z}{(1-\bar{c}z)^2}.
\end{equation}
Cualquier polinomio candidato se escribe como $p = u K_0 + v K_1$. El problema de maximización de la norma $\norm{\D p}^2 / \norm{p}^2$ equivale a encontrar la mayor raíz $\lambda$ de la ecuación característica $\det(\mathbf{B} - \lambda \mathbf{A}) = 0$.

Las entradas de estas matrices se calculan utilizando las propiedades de reproducción $\langle f, K_0 \rangle_{L^2} = f(c)$ y $\langle f, K_1 \rangle_{L^2} = f'(c)$.

\textbf{1. Matriz de Norma de Entrada ($\mathbf{A}$):}
La norma de entrada es $\norm{p}^2 = \|p\|_{L^2}^2 + \abs{p'(c)}^2$. Los elementos de la matriz $\mathbf{A} = (A_{ij})$ son:
$$ A_{ij} = \langle K_j, K_i \rangle_{L^2} + K_j'(c)\overline{K_i'(c)}. $$
Usando la notación $k_{ij} = K_i^{(j)}(c)$ para las derivadas de los núcleos evaluadas en $c$:
\begin{equation}
    \mathbf{A} = \begin{pmatrix}
        k_{00} + \abs{k_{01}}^2          & \overline{k_{01}} + k_{11}\overline{k_{01}} \\
        k_{01} + \overline{k_{11}}k_{01} & k_{11} + \abs{k_{11}}^2
    \end{pmatrix}.
\end{equation}

\textbf{2. Matriz de Norma de Salida ($\mathbf{B}$):}
La norma de salida es $\norm{\D p}^2 = \|p\|_{L^2}^2 + \abs{(\D p)'(c)}^2$. Observamos que $(\D p)'(c) = p(c) + c p'(c)$. Por tanto, la matriz $\mathbf{B}$ es una perturbación de rango 1 de la parte continua:
$$ B_{ij} = \langle K_j, K_i \rangle_{L^2} + [K_j(c) + c K_j'(c)] \overline{[K_i(c) + c K_i'(c)]}. $$

\subsubsection{Ejemplo Numérico: Átomo en $c = 0.5$}

Particularicemos para un átomo real en $c=1/2$. Los valores de los núcleos y sus derivadas son:
\begin{align*}
    k_{00} & = K_0(c) = \frac{1}{1-c^2} = \frac{4}{3},            \\
    k_{01} & = K_0'(c) = \frac{c}{(1-c^2)^2} = \frac{8}{9},       \\
    k_{11} & = K_1'(c) = \frac{1+c^2}{(1-c^2)^3} = \frac{80}{27}.
\end{align*}

Sustituyendo estos valores en la matriz $\mathbf{A}$ (y simplificando fracciones):
$$ \mathbf{A} \approx \begin{pmatrix} 2.123 & 3.522 \\ 3.522 & 11.742 \end{pmatrix}. $$
Para la matriz $\mathbf{B}$, calculamos el término de derivada de la imagen. Sea $w_0$ y $w_1$ los valores de $(\D K_0)'(c)$ y $(\D K_1)'(c)$ respectivamente:
\begin{align*}
    w_0 & = k_{00} + c k_{01} = \frac{4}{3} + \frac{1}{2}\left(\frac{8}{9}\right) = \frac{16}{9},    \\
    w_1 & = k_{10} + c k_{11} = \frac{8}{9} + \frac{1}{2}\left(\frac{80}{27}\right) = \frac{64}{27}.
\end{align*}
(Nótese que $K_1(c) = K_0'(c)$ en el caso real). Así construimos $\mathbf{B}$:
$$ B_{ij} = k_{ji} + w_j w_i \implies \mathbf{B} \approx \begin{pmatrix} 4.493 & 5.101 \\ 5.101 & 8.581 \end{pmatrix}. $$

El primer número de Gelfand es la raíz cuadrada de la mayor solución $\lambda$ de $\det(\mathbf{B} - \lambda \mathbf{A}) = 0$:
$$ (4.493 - 2.123\lambda)(8.581 - 11.742\lambda) - (5.101 - 3.522\lambda)^2 = 0. $$
Resolviendo la ecuación cuadrática resultante obtenemos $\lambda_{max} \approx 4.55$.

Por lo tanto, la norma del operador con un átomo en $c=0.5$ es:
\begin{equation}
    c_1(\D) = \sqrt{\lambda_{max}} \approx 2.13.
\end{equation}

Este valor es estrictamente mayor que $\sqrt{2} \approx 1.41$, obtenido para el átomo en el origen. Esto confirma cuantitativamente que el desplazamiento del átomo hacia la frontera incrementa la norma del operador.

\subsection{El Segundo Número de Gelfand: Recuperando la Circunferencia}

A continuación se calcula $c_2(D)$, el ínfimo de la norma del operador sobre todos los subespacios de codimensión 1.

El siguiente teorema establece que, independientemente de la posición exacta del átomo BPE, al restringir el espacio una sola dimensión, la norma del operador coincide exactamente con el radio de la circunferencia.

\begin{teorema}[Valor exacto de $c_2$ para un átomo]
    Sea $\mathcal{H}$ el espacio de Sobolev con medida continua soportada en $S_1$ y un único átomo BPE $c_1$ en la derivada. El segundo número de Gelfand del operador de multiplicación $D(p) = zp$ es exactamente el radio de la circunferencia:
    $$ c_2(D) = 1 $$
\end{teorema}

\begin{proof}
    La demostración se estructura en dos partes: en primer lugar, se exhibe un subespacio de codimensión 1 donde la norma del operador restringido es igual a 1 (cota superior); en segundo lugar, se demuestra que ningún subespacio de codimensión 1 permite reducir la norma del operador por debajo de este valor (cota inferior).

    \textbf{Parte I: Cota superior ($c_2(D) \le 1$)}

    Buscamos un subespacio $V$ de codimensión 1 que neutralice el efecto expansivo de la derivada. Motivados por la expresión de la norma de la imagen, definimos el subespacio óptimo $W_{opt}$ imponiendo que la derivada de la imagen se anule en el átomo:
    $$ W_{opt} = \{ p \in \mathcal{H} : (zp)'(c_1) = 0 \}. $$
    Esta condición es el núcleo del funcional lineal $\Psi(p) = p(c_1) + c_1 p'(c_1)$. Gracias a nuestra hipótesis BPE, las evaluaciones $p(c_1)$ y $p'(c_1)$ son continuas, lo que hace que $\Psi \in \mathcal{H}^*$. En consecuencia, $W_{opt} = \ker(\Psi)$ es un subespacio cerrado de codimensión 1 (o 0, si $\Psi$ fuera trivial, lo cual también es admisible).

    Si tomamos cualquier polinomio $p \in W_{opt}$, su norma de salida se reduce a:
    \begin{align*}
        \norm{Dp}^2 & = \int_{S_1} \abs{z p(z)}^2 d\mu + \underbrace{\abs{(zp)'(c_1)}^2}_{=0 \text{ en } W_{opt}} \\
                    & = \int_{S_1} 1 \cdot \abs{p(z)}^2 d\mu = \|p\|_{L^2(\mu)}^2.
    \end{align*}
    Dado que en nuestro producto de Sobolev siempre se cumple que la parte integral está acotada por la norma total ($\|p\|_{L^2(\mu)}^2 \le \norm{p}^2$), deducimos que $\norm{Dp}^2 \le \norm{p}^2$.
    Por tanto, $\norm{D|_{W_{opt}}} \le 1$. Al ser $W_{opt}$ un subespacio válido para el ínfimo, garantizamos que $c_2(D) \le 1$.

    \textbf{Parte II: Cota inferior ($c_2(D) \ge 1$)}

    Procedemos por reducción al absurdo. Supóngase que $c_2(D) < 1$. Esto implicaría la existencia de un subespacio cerrado $V \subset \mathcal{H}$ de codimensión 1 (o 0), y un cierto $\epsilon > 0$, donde el operador fuera una contracción estricta:
    $$ \norm{Dp}^2 \le (1-\epsilon) \norm{p}^2, \quad \forall p \in V. $$
    Desglosando esta hipótesis en sus componentes continua y discreta:
    $$ \|p\|_{L^2(\mu)}^2 + \abs{(zp)'(c_1)}^2 \le (1-\epsilon) \left( \|p\|_{L^2(\mu)}^2 + \abs{p'(c_1)}^2 \right). $$
    El término $\abs{(zp)'(c_1)}^2$ es siempre no negativo, por lo que si lo eliminamos del lado izquierdo, la desigualdad debe seguir siendo cierta:
    $$ \|p\|_{L^2(\mu)}^2 \le (1-\epsilon) \|p\|_{L^2(\mu)}^2 + (1-\epsilon) \abs{p'(c_1)}^2. $$
    Reagrupando la norma $L^2$, llegamos a la siguiente condición ineludible para todo $p \in V$:
    \begin{equation} \label{eq:absurdo_n1}
        \epsilon \|p\|_{L^2(\mu)}^2 \le (1-\epsilon) \abs{p'(c_1)}^2.
    \end{equation}

    Consideremos el funcional de evaluación de la derivada $E(p) = p'(c_1)$, restringido a nuestro subespacio $V$. Como $E$ es un funcional lineal a valores escalares sobre un espacio de dimensión infinita ($V$), su núcleo, $\ker(E|_V)$, no puede ser trivial; de hecho, es de dimensión infinita.

    Elegimos entonces un polinomio no nulo $q \in \ker(E|_V)$. Para este elemento particular, se cumple $q'(c_1) = 0$. Al introducir $q$ en nuestra desigualdad \eqref{eq:absurdo_n1}, la parte discreta se desvanece:
    $$ \epsilon \|q\|_{L^2(\mu)}^2 \le 0. $$
    Puesto que $\epsilon > 0$, la única solución matemática es que $\|q\|_{L^2(\mu)} = 0$. Pero la medida $\mu$ reside en la circunferencia $S_1$, la cual contiene infinitos puntos. Un polinomio que se anula en la circunferencia completa es idénticamente nulo ($q=0$), lo que contradice la elección de $q \neq 0$.

    La contradicción se obtiene al suponer que el operador podía ser acotado por debajo de 1 restringiendo una única dimensión. Por consiguiente, $c_2(D) \ge 1$.

    Combinando las Partes I y II, se concluye que $c_2(D) = 1$.
\end{proof}

\section{Cálculo Exacto del Número de Gelfand Asintótico (Caso Acotado)}

Aplicando la definición de los números de Gelfand introducida en la sección anterior, se extiende el análisis al caso general de $N$ átomos.

Para poder aplicar esta teoría globalmente a nuestro operador de multiplicación $D(p) = zp$, es condición necesaria que $D$ sea un operador acotado en $\mathcal{H}$. Como hemos demostrado en secciones anteriores, esto ocurre si y solo si las evaluaciones en los átomos no generan divergencias en la norma.

\begin{lema}[Acotación Global bajo Hipótesis BPE]
    Supongamos que los $N$ átomos discretos $\{c_1, \dots, c_N\}$ asociados a las derivadas son \textit{Bounded Point Evaluations} (BPE) en $\mathcal{H}$, y están situados en el disco unidad abierto ($|c_k| < 1$). Entonces, el operador $D$ es globalmente acotado en $\mathcal{H}$, y sus números de Gelfand $c_m(D)$ están bien definidos para todo $m \ge 1$.
\end{lema}

Bajo esta premisa de acotación global, se demuestra que, independientemente de la posición exacta de los $N$ átomos BPE dentro del disco, el $(N+1)$-ésimo número de Gelfand está determinado por la parte continua de la medida.

\begin{teorema}[Valor exacto de $c_{N+1}$]
    Sea $\mathcal{H}$ el espacio de Sobolev con medida continua soportada en la circunferencia unidad $S_1$ ($R=1$) y $N$ átomos BPE en la derivada. Entonces, el $(N+1)$-ésimo número de Gelfand del operador de multiplicación es exactamente el radio de la circunferencia:
    $$ c_{N+1}(D) = 1 $$
\end{teorema}

\begin{proof}
    Para demostrar esta igualdad, estableceremos independientemente una cota superior y una cota inferior para el ínfimo que define a $c_{N+1}(D)$, el cual evalúa subespacios de codimensión menor o igual a $N$.

    \textbf{Paso 1: Cota Superior ($c_{N+1}(D) \le 1$)}

    Definimos el subespacio $W_{opt}$ anulando la derivada de la imagen en los $N$ átomos:
    $$ W_{opt} = \{ p \in \mathcal{H} : (zp)'(c_k) = 0, \quad \forall k=1,\dots,N \}. $$
    Podemos expresar esta restricción mediante los funcionales $\Psi_k(p) = p(c_k) + c_k p'(c_k)$. Al ser los átomos BPE por hipótesis, las evaluaciones de la función y de su derivada son funcionales lineales continuos. Por tanto, $\Psi_k \in \mathcal{H}^*$.
    El subespacio $W_{opt} = \bigcap_{k=1}^N \ker(\Psi_k)$ es la intersección de $N$ núcleos de funcionales continuos, lo que garantiza que es un subespacio cerrado con $\operatorname{codim}(W_{opt}) \le N$.

    Evaluamos la norma de la imagen para cualquier $p \in W_{opt}$:
    \begin{align*}
        \norm{Dp}^2 & = \int_{S_1} \abs{z p(z)}^2 d\mu + \sum_{k=1}^N \underbrace{\abs{(zp)'(c_k)}^2}_{=0 \text{ en } W_{opt}} \\
                    & = \int_{S_1} 1 \cdot \abs{p(z)}^2 d\mu = \|p\|_{L^2(\mu)}^2.
    \end{align*}
    Por la definición del producto interno de Sobolev, es evidente que $\|p\|_{L^2(\mu)}^2 \le \norm{p}^2$. En consecuencia, $\norm{Dp} \le \norm{p}$ para todo $p \in W_{opt}$, lo que significa que $\norm{D|_{W_{opt}}} \le 1$.
    Como $W_{opt}$ es un candidato válido para el ínfimo (su codimensión no excede $N$), se sigue que:
    $$ c_{N+1}(D) \le \norm{D|_{W_{opt}}} \le 1. $$

    \textbf{Paso 2: Cota Inferior ($c_{N+1}(D) \ge 1$)}

    Procederemos por reducción al absurdo. Supongamos que $c_{N+1}(D) < 1$. Por la definición de ínfimo, existiría un subespacio cerrado $V \subset \mathcal{H}$ con $\operatorname{codim}(V) \le N$ y un $\epsilon > 0$ tal que el operador se contrae estrictamente sobre $V$:
    $$ \norm{Dp}^2 \le (1-\epsilon) \norm{p}^2, \quad \forall p \in V. $$
    Desarrollando las normas de Sobolev:
    $$ \|p\|_{L^2(\mu)}^2 + \sum_{k=1}^N \abs{(zp)'(c_k)}^2 \le (1-\epsilon) \left( \|p\|_{L^2(\mu)}^2 + \sum_{k=1}^N \abs{p'(c_k)}^2 \right). $$
    Dado que el sumatorio de la izquierda es siempre no negativo, podemos omitirlo para obtener una desigualdad que debe seguir siendo cierta:
    $$ \|p\|_{L^2(\mu)}^2 \le (1-\epsilon) \|p\|_{L^2(\mu)}^2 + (1-\epsilon) \sum_{k=1}^N \abs{p'(c_k)}^2. $$
    Reagrupando términos:
    \begin{equation} \label{eq:absurdo_gelfand}
        \epsilon \|p\|_{L^2(\mu)}^2 \le (1-\epsilon) \sum_{k=1}^N \abs{p'(c_k)}^2, \quad \forall p \in V.
    \end{equation}

    Consideremos la aplicación de evaluación de derivadas $E: V \to \C^N$ definida por $E(p) = (p'(c_1), \dots, p'(c_N))$.
    Por nuestra hipótesis, los átomos son \textit{Bounded Point Evaluations} (BPE), lo que implica por definición que cada funcional de evaluación es continuo respecto a la norma de Sobolev. En consecuencia, el operador lineal $E$ es continuo.

    En Análisis Funcional, el núcleo de un operador lineal continuo es siempre un subespacio cerrado. Por tanto, $\ker(E|_V)$ es un subespacio cerrado de $V$. Además, dado que la imagen $\operatorname{Im}(E|_V)$ está contenida en $\C^N$, es de dimensión finita (y por tanto cerrada).

    Aplicando el Primer Teorema de Isomorfía Topológica para espacios de Banach (véase Conway \cite{conway1990course}), tenemos el siguiente isomorfismo algebraico y topológico:
    $$ V / \ker(E|_V) \cong \operatorname{Im}(E|_V) \subseteq \C^N $$

    Dado que el espacio $V$ es de dimensión infinita (por tener codimensión finita en $\mathcal{H}$) y el espacio cociente $V / \ker(E|_V)$ es de dimensión finita, se deduce que el núcleo $\ker(E|_V)$ debe ser un subespacio cerrado de dimensión infinita.

    Tomemos cualquier polinomio no nulo $q \in \ker(E|_V)$ (cuya existencia está garantizada al ser de dimensión infinita). Para este polinomio se cumple $q'(c_k) = 0$ para todo $k=1, \dots, N$. Sustituyendo $q$ en la desigualdad \eqref{eq:absurdo_gelfand} obtenemos:
    $$ \epsilon \|q\|_{L^2(\mu)}^2 \le 0. $$
    Como $\epsilon > 0$, concluimos que $\|q\|_{L^2(\mu)} = 0$. Al estar la medida $\mu$ soportada en la circunferencia $S_1$ (un conjunto con infinitos puntos), el único polinomio con norma $L^2$ nula es el polinomio idénticamente cero, lo cual contradice la elección de $q \neq 0$.

    Esta contradicción demuestra que no existe ningún subespacio $V$ de codimensión $\le N$ capaz de reducir la norma del operador por debajo de 1. Por consiguiente:
    $$ c_{N+1}(D) \ge 1. $$

    La combinación de la cota superior y la inferior concluye la demostración: $c_{N+1}(D) = 1$.
\end{proof}

\section{Convergencia Asintótica de las Subdiagonales}

\begin{proposicion}[Convergencia Asintótica de la Matriz de Hessenberg]
    \cite{lopez2001sobolev} Sea $\mathbf{D} = (d_{n,k})_{n,k=0}^\infty$ la matriz de representación del operador de multiplicación por $z$ en la base ortonormal de Sobolev $\{\psi_n\}_{n=0}^\infty$.
    Los elementos de las diagonales principales de $\mathbf{D}$ convergen a los parámetros geométricos del soporte continuo:
    \begin{align}
        \lim_{n \to \infty} d_{n, n}   & = C \quad (\text{Diagonal Principal})   \\
        \lim_{n \to \infty} d_{n+1, n} & = R \quad (\text{Subdiagonal Inferior})
    \end{align}
\end{proposicion}

\begin{proof}
    La demostración se fundamenta en la teoría de asintótica de razones (\textit{ratio asymptotics}) para polinomios ortogonales. Para garantizar la validez de estos resultados, primero verificamos que la medida de ortogonalidad pertenece a la clase de Szegő.

    La medida $\mu$ asociada a nuestro producto de Sobolev se compone de una parte absolutamente continua soportada en la circunferencia $S_R$ y una parte discreta finita. La parte continua corresponde a la medida de Lebesgue, cuyo peso es constante $w(\theta) = c > 0$ sobre el soporte. La condición de Szegő requiere la integrabilidad logarítmica del peso:
    $$ \int_{0}^{2\pi} \log w(\theta) \, d\theta > -\infty $$
    Dado que el peso es constante y estrictamente positivo, $\log(c)$ es finito y la integral converge.

    Bajo esta premisa, los trabajos de Berriochoa y Cachafeiro \cite{berriochoa2000bernstein} y López Lagomasino y Pijeira Cabrera \cite{lopez1999zero} establecen que la adición de masas discretas en la derivada no altera el comportamiento asintótico de la razón de los polinomios fuera del soporte esencial (el disco cerrado). Por consiguiente, se cumple el teorema de comparación asintótica fuerte con respecto a los polinomios ortogonales estándar de la circunferencia $\{\varphi_n\}$:
    \begin{equation}
        \lim_{n \to \infty} \frac{\psi_{n+1}(z)}{\psi_n(z)} = \lim_{n \to \infty} \frac{\varphi_{n+1}(z)}{\varphi_n(z)} = \frac{z - C}{R}, \quad \forall |z| > R.
    \end{equation}

    Los elementos de la matriz $\mathbf{D}$ son precisamente los coeficientes de la relación de recurrencia que satisface la base ortonormal:
    \begin{equation} \label{eq:recurrencia}
        z \psi_n(z) = d_{n+1, n} \psi_{n+1}(z) + d_{n, n} \psi_n(z) + \sum_{k=0}^{n-1} d_{k, n} \psi_k(z).
    \end{equation}
    Dividiendo la ecuación (\ref{eq:recurrencia}) por $\psi_n(z)$ y tomando el límite cuando $n \to \infty$ para un $z$ fijo con módulo suficientemente grande, los términos de la suma decaen asintóticamente. La ecuación se estabiliza a:
    $$ z \approx d_{n+1, n} \left( \frac{z - C}{R} \right) + d_{n, n}. $$
    Para que esta igualdad se mantenga para todo $z$:
    \begin{itemize}
        \item Comparando el coeficiente de $z^1$:
              $$ 1 = \frac{d_{n+1, n}}{R} \implies d_{n+1, n} \to R. $$
        \item Comparando el término independiente ($z^0$):
              $$ 0 = d_{n+1, n} \left( \frac{-C}{R} \right) + d_{n, n}. $$
              Sustituyendo el límite $d_{n+1, n} \to R$:
              $$ 0 = R \left( \frac{-C}{R} \right) + d_{n, n} \implies d_{n, n} \to C. $$
    \end{itemize}
\end{proof}


\section{Comportamiento Asintótico y Perturbaciones Espectrales}

En esta sección se analiza el comportamiento límite de los valores singulares de las matrices de Hessenberg $H_n$ asociadas al operador, utilizando resultados clásicos de teoría de perturbaciones.

La estructura del producto interno de Sobolev puede interpretarse como una perturbación de rango finito del producto interno estándar en $L^2(\mu)$. El siguiente resultado establece la estabilidad del espectro esencial bajo tales perturbaciones.

\begin{proposicion}[Invariancia del Espectro Esencial]
    \cite{frymark2021spectral} Sea $H_n$ la matriz de representación del operador de multiplicación $D$ en la base ortonormal truncada de dimensión $n$.
    Consideremos la descomposición de los valores singulares de $H_n$, denotados por $\sigma_k^{(n)}$, ordenados de forma decreciente.
    Al eliminar los efectos de los $m$ átomos inestables (ya tratados en la Proposición 2), el límite de la norma del operador restringido recupera el radio del soporte continuo.
\end{proposicion}

\begin{proof}
    El operador de multiplicación por $z$, denotado por $M_z$, actuando sobre el espacio $L^2$ de la circunferencia de radio $R$, tiene como espectro esencial el conjunto de valores que alcanza la función multiplicadora: $\sigma_{ess}(M_z) = \{z \in \C : \abs{z} = R\}$. Por tanto, el radio espectral esencial es $R$.

    El producto interno de Sobolev $\ip{\cdot}{\cdot}$ difiere del producto escalar en $L^2$ únicamente en una suma finita de términos (las $N$ evaluaciones de derivadas). En el contexto de operadores, esto implica que la diferencia entre el operador en el espacio Sobolev y el operador en $L^2$ es un operador de rango finito (a lo sumo rango $N$).

    Invocamos el \textbf{Teorema de Weyl sobre el Espectro Esencial}, el cual establece que el espectro esencial de un operador autoadjunto (o normal/acotado en contextos más generales) es invariante bajo perturbaciones compactas. Dado que una perturbación de rango finito es compacta, deducimos que:
    $$ \sigma_{ess}(D|_W) = \sigma_{ess}(M_z) = R. $$
    En términos de los valores singulares de las secciones finitas $H_n$, esto implica que, salvo un número finito de valores atípicos (outliers) generados por la perturbación, la sucesión converge al valor del espectro esencial:
    $$ \lim_{n \to \infty} \sigma_{k}^{(n)} = R, \quad \text{para } k \text{ suficientemente grande}. $$
\end{proof}

Basándonos en los resultados anteriores, se formaliza la estratificación de los valores singulares.

\begin{proposicion}[Clasificación Espectral Asintótica]
    Sea $\{\sigma_j^{(n)}\}_{j=1}^n$ la secuencia de valores singulares de la matriz $H_n$. En el límite asintótico $n \to \infty$, los valores singulares se segregan en tres conjuntos disjuntos según su comportamiento:

    \begin{enumerate}
        \item \textbf{Valores Singulares Divergentes ($I_{out}$):}
              Para los índices $j = 1, \dots, m$ (correspondientes a los átomos con $\abs{c_k} > R$), se cumple:
              $$ \lim_{n \to \infty} \sigma_j^{(n)} = \infty. $$
              Esto es consecuencia directa de la Proposición 2 (No acotación del operador en subespacios de codimensión menor a $m$).

        \item \textbf{Valores Singulares Aislados ($I_{in}$):}
              Para los átomos interiores con $\abs{c_k} < R$ que introducen una perturbación en la derivada, aparecen valores singulares que convergen a constantes estrictamente mayores que el radio esencial:
              $$ \lim_{n \to \infty} \sigma_j^{(n)} = \lambda_j > R, \quad \text{para } j = m+1, \dots, m+k. $$
              Estos valores corresponden a autovalores aislados del operador perturbado $D|_W$, cuya existencia está garantizada por la teoría de perturbaciones de rango finito (valores singulares atípicos o \textit{outliers} espectrales).

        \item \textbf{Parte Esencial (Convergencia a $R$):}
              Para el resto de índices $j > m+k$, los valores singulares colapsan al radio espectral esencial:
              $$ \lim_{n \to \infty} \sigma_j^{(n)} = R. $$
              Esto confirma que el comportamiento espectral del operador está determinado asintóticamente por la medida de Lebesgue en la circunferencia.
    \end{enumerate}
\end{proposicion}

\subsection{Rapidez de Convergencia}

\begin{observacion}[Velocidad de Convergencia de la Matriz de Hessenberg]
    Sean $\{d_{n,n}(a)\}_{n=0}^\infty$ y $\{d_{n+1,n}(a)\}_{n=0}^\infty$ las sucesiones correspondientes a los elementos de la diagonal principal y la subdiagonal inferior de la matriz $\mathbf{D}$, respectivamente.
    Para cualquier valor fijo del átomo $a$, se cumple:
    \begin{align}
        \lim_{n \to \infty} d_{n,n}(a)   & = 0 \quad (\text{Centro de la Circunferencia}) \\
        \lim_{n \to \infty} d_{n+1,n}(a) & = 1 \quad (\text{Radio de la Circunferencia})
    \end{align}

    La demostración se realiza mediante el análisis asintótico de los coeficientes dominantes de las expresiones racionales que definen cada término de las sucesiones.

    \textbf{1. Convergencia de la Diagonal Principal ($d_{n,n} \to 0$):}
    Los términos de la sucesión $\{d_{n,n}(a)\}$ son funciones racionales de la forma $\frac{P_n(a)}{Q_n(a)}$. Al examinar los términos de orden superior ($n=8, 9, \dots$), se verifica que el cociente entre los coeficientes líderes del numerador y del denominador sigue una regla de decaimiento cuadrático.
    Específicamente, para el $k$-ésimo término avanzado, el coeficiente dominante $C_k$ satisface la relación de recurrencia aproximada:
    $$ C_k \approx \frac{1}{k(k+1)} $$
    (Por ejemplo, para los últimos términos calculados, los coeficientes son $\frac{1}{42}, \frac{1}{56}, \frac{1}{72}$, correspondientes a $k=6, 7, 8$).
    Tomando el límite cuando $k \to \infty$:
    $$ \lim_{n \to \infty} d_{n,n}(a) = \lim_{k \to \infty} \frac{1}{k^2+k} = 0 $$
    Esto confirma que la diagonal principal converge al centro $C=0$.

    \textbf{2. Convergencia de la Subdiagonal ($d_{n+1,n} \to 1$):}
    Los términos de la sucesión $\{d_{n+1,n}(a)\}$ involucran radicales en el denominador. El análisis de los coeficientes de los términos de mayor grado en $a$ revela que la razón entre el numerador y el denominador satisface:
    $$ L_k = 1 - \frac{1}{k^2} $$
    Donde los términos calculados verifican razones de $\frac{48}{49}$ ($1 - 1/7^2$) y $\frac{63}{64}$ ($1 - 1/8^2$).
    Calculando el límite asintótico de esta sucesión:
    $$ \lim_{n \to \infty} d_{n+1,n}(a) = \lim_{k \to \infty} \left( 1 - \frac{1}{k^2} \right) = 1 $$
    Esto confirma que la subdiagonal converge al radio $R=1$, independientemente del valor del parámetro $a$.
\end{observacion}
