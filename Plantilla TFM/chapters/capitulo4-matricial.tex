\chapter{Análisis Asintótico y Secciones Matriciales}\label{sec:matricial}

\section{Matriz de Hessenberg y secciones truncadas}

Sea $\mathcal H=\overline{\Poly}$ la compleción hilbertiana de $(\Poly,\langle\cdot,\cdot\rangle_S)$ y sea $\{\phi_n\}_{n\ge0}$ la base ortonormal de polinomios asociada. En el caso clásico, si $T\in\mathcal L(\mathcal H)$ es un operador acotado, admite una representación matricial respecto de una base ortonormal $\{e_n\}_{n\ge0}$ cuyas entradas son $\langle Te_j,e_i\rangle$ \cite{}. Esta representación matricial puede hacerse también a operadores de $T:\Poly \longrightarrow \Poly$ no necesariamente acotados, basta que $T$ esté definido en el subespacio denso $\Poly$. Por ello, aunque $\D$ pueda no extenderse como operador acotado sobre la compleción, su acción sobre la base de $\Poly$ sigue determinando una matriz infinita. Esta representación permite pasar del operador abstracto a una familia de objetos finito-dimensionales sobre los que pueden estudiarse espectro y valores singulares de manera directa. Esta perspectiva se sitúa dentro del enfoque matricial desarrollado recientemente para polinomios ortogonales y cantidades espectrales asociadas; véanse \cite{EGT23,EG25-2}.

\begin{defn}\label{def:hessenberg}
	Sea $(\Poly, \langle \cdot,\cdot\rangle_S)$ y sea $\{\phi_n\}_{n\ge0}$ la sucesión de polinomios ortonormales asociada al producto de Sobolev. La representación matricial de $\D$ es
	\[
		\mathbf D=(d_{i,j})_{i,j\ge0},
		\qquad
		d_{i,j}=\langle z\phi_j,\phi_i\rangle_S.
	\]
\end{defn}

\begin{prop}\label{prop:hessenberg}
	Sea $(\Poly, \langle \cdot,\cdot\rangle_S)$ y sea $\{\phi_n\}_{n\ge0}$ la sucesión de polinomios ortonormales asociada al producto de Sobolev.
	Entonces la matriz $\mathbf D$ es de Hessenberg, es decir,
	\[
		d_{i,j}=0,
		\qquad i>j+1.
	\]
\end{prop}

\begin{proof}
	La expansión de $z\phi_j$ en la base ortonormal solo puede involucrar polinomios de grado a lo sumo $j+1$. Por ortogonalidad, sus coeficientes frente a $\phi_i$ se anulan cuando $i>j+1$.
\end{proof}

\begin{defn}\label{def:seccion_truncada}
	Para cada $n\in\mathbb N$, sea
	\(
		\mathbb P_n:=[\phi_0,\dots,\phi_n],
	\)
	y sea $\pi_n:\mathcal H\to\mathbb P_n$ la proyección ortogonal. Definimos la aplicación lineal
	\[
		\mathcal D_n:\mathbb P_n\to\mathbb P_n,
		\qquad
		\mathcal D_n(p):=\pi_n(\D p).
	\]
	La matriz de $\mathcal D_n$ respecto de la base $\{\phi_0,\dots,\phi_n\}$ se llama la \emph{sección principal} de orden $n+1$ de $\mathbf D$ y viene dada por
	\[
		\mathbf D_n:=(d_{i,j})_{0\le i,j\le n}\in\mathcal M_{n+1}(\mathbb C).
	\]
\end{defn}

\begin{rmk}
	Como $\mathbb P_n\subset\Poly$ y $\D(\mathbb P_n)\subset\Poly$, la aplicación $\mathcal D_n$ está bien definida.
\end{rmk}

\begin{prop}\label{prop:charpoly_truncada}
	Sea $\{\phi_n\}_{n\ge0}$ la sucesión de polinomios ortonormales asociada al producto de Sobolev en $(\Poly, \langle \cdot,\cdot\rangle_S)$. Para todo \(n\ge0\), sea
	\(\mathbf D_n\) la sección principal de \(\D\) en la base \(\{\phi_0,\dots,\phi_n\}\).
	Entonces existe una constante
	\(
		\kappa_n\in\C\setminus\{0\}
	\)
	tal que
	\[
		\det(z\mathbf I_{n+1}-\mathbf D_n)
		=
		\det(z\mathbf I_{n+1}-\mathbf D_n^t)
		=
		\kappa_n\,\phi_{n+1}(z).
	\]
\end{prop}

\begin{proof}
	Para cada $j=0,\dots,n$, la expansión de $z\phi_j$ en la base ortonormal viene dada por
	\[
		z\phi_j(z)=\sum_{i=0}^{j+1} d_{i,j}\phi_i(z).
	\]
	Sea ahora $z_0\in\C$ un cero de $\phi_{n+1}$. Evaluando en $z_0$, obtenemos para cada
	$j=0,\dots,n$
	\[
		z_0\phi_j(z_0)=\sum_{i=0}^{n} d_{i,j}\phi_i(z_0),
	\]
	pues el término con $\phi_{n+1}(z_0)$ desaparece. Si definimos
	\[
		v(z_0):=(\phi_0(z_0),\dots,\phi_n(z_0))^t\in\C^{n+1},
	\]
	entonces $v(z_0)\neq 0$, ya que $\phi_0$ es constante no nula, y la identidad anterior se escribe como
	\[
		\mathbf D_n^t v(z_0)=z_0 v(z_0).
	\]
	Por tanto, todo cero de $\phi_{n+1}$ es un autovalor de $\mathbf D_n^t$, es decir,
	\[
		\phi_{n+1}(z)=0
		\quad\Longrightarrow\quad
		\det(z\mathbf I_{n+1}-\mathbf D_n^t)=0.
	\]
	Además, $\det(z\mathbf I_{n+1}-\mathbf D_n^t)$ y $\phi_{n+1}(z)$ son polinomios de
	grado $n+1$. Como todos los ceros del segundo son también ceros del primero, se sigue que existe una constante
	$\kappa_n\in\C\setminus\{0\}$ tal que
	\[
		\det(z\mathbf I_{n+1}-\mathbf D_n^t)=\kappa_n\phi_{n+1}(z).
	\]
	Además, para toda matriz cuadrada $A$ se tiene
	\(\det(z\mathbf I-A^t)=\det\bigl((z\mathbf I-A)^t\bigr)=\det(z\mathbf I-A)\).
	Aplicando esto a $A=\mathbf D_n$, concluimos que
	\[
		\det(z\mathbf I_{n+1}-\mathbf D_n)=\det(z\mathbf I_{n+1}-\mathbf D_n^t)=\kappa_n\phi_{n+1}(z).
	\]
\end{proof}

\begin{cor}
	Sea $\{\phi_n\}_{n\ge0}$ la sucesión de polinomios ortonormales asociada al producto de Sobolev en $(\Poly,\langle \cdot,\cdot\rangle_S)$ y, para cada $n\ge0$, sea $\mathbf D_n$ la sección principal de $\D$ en la base $\{\phi_0,\dots,\phi_n\}$. Entonces
	\[
		\sigma(\mathbf D_n)
		=
		\sigma(\mathbf D_n^t)
		=
		\{z\in\C:\phi_{n+1}(z)=0\},
	\]
	contando multiplicidades algebraicas.
\end{cor}

\begin{proof}
	Es una consecuencia inmediata de la Proposición~\ref{prop:charpoly_truncada}.
\end{proof}

\begin{rmk}
	La Proposición~\ref{prop:charpoly_truncada} identifica a $\mathbf D_n$ como una representación matricial finita cuya información espectral queda determinada por $\phi_{n+1}$. En particular, los autovalores de $\mathbf D_n$ y de $\mathbf D_n^t$ coinciden con los ceros, luego el estudio de los ceros de $\phi_{n+1}$ pueden abordarse mediante el análisis de $\mathbf D_n$.
\end{rmk}

\section{Valores singulares de \texorpdfstring{$\mathbf D_n$}{Dn} y relación con el índice \texorpdfstring{$Q_k(\D)$}{Qk(D)}}

Siguiendo con la filosofía de los capítulos anteriores, introduciremos ahora los valores singulares de estas representaciones matriciales y veremos cómo se relacionan con $Q_k(\D)$.

\begin{defn}\label{def:sigma_kn}
	Para $n\ge0$, sea \(\mathbf D_n\in\mathcal M_{n+1}(\C)\) la sección principal de \(\mathbf D\) en la base \(\{\phi_0,\dots,\phi_n\}\).
	Denotamos por
	\[
		\sigma_{0,n}\ge\sigma_{1,n}\ge\cdots\ge\sigma_{n,n}\ge0
	\]
	los valores singulares de \(\mathbf D_n\), ordenados de forma decreciente.
\end{defn}

\begin{rmk}
	Se tiene $\sigma_{0,n}=\|\mathbf D_n\|_2$, donde la norma $\|\cdot\|_2$ es la norma del operador inducido por la matriz $\mathbf D_n \in \mathcal M_{n+1}(\C)$, considerando la norma euclídea. La indexación desde $0$ sitúa en $\sigma_{0,n}$ el mayor valor singular de la truncación.
\end{rmk}

De manera análoga al desarrollo hecho con operadores usando los números de Gelfand y el índice $Q_k(\D)$, necesitaremos una herramienta variacional para las representaciones matriciales. El estudio de estas truncaciones mediante valores singulares encaja además en la línea de trabajo matricial reciente sobre polinomios ortogonales, desigualdades espectrales y localización de ceros; cf. \cite{EG25-2}. Para ello usaremos la caracterización min--max de Courant--Fischer para valores singulares; véase \cite[Section~3.1]{horn2012matrix}.

\begin{lem}
	\label{lem:minmax_sigma_Dn}
	Sea \(n\ge0\) y sea \(\mathbf D_n\in\mathcal M_{n+1}(\C)\) la sección principal de \(\mathbf D\) en la base \(\{\phi_0,\dots,\phi_n\}\). Entonces, para todo \(0\le k\le n\),
	\[
		\sigma_{k,n}
		=
		\min_{\substack{S\subset\C^{n+1}\\ \dim S=n+1-k}}
		\max_{\substack{x\in S\\ x\neq0}}
		\frac{\|\mathbf D_nx\|_2}{\|x\|_2}.
	\]
	Equivalentemente,
	\[
		\sigma_{k,n}
		=
		\max_{\substack{S\subset\C^{n+1}\\ \dim S=k+1}}
		\min_{\substack{x\in S\\ x\neq0}}
		\frac{\|\mathbf D_nx\|_2}{\|x\|_2}.
	\]
\end{lem}

\begin{lem}\label{lem:bloque_gram_truncadas}
	Sea \(\mathbf D=(d_{i,j})_{i,j\ge0}\) una matriz de Hessenberg superior infinita y, para cada \(n\ge0\), sea
	\(
		\mathbf D_n=(d_{i,j})_{0\le i,j\le n}
	\)
	su sección principal de orden \(n+1\). Entonces
	\[
		(\mathbf D_{n+1}^*\mathbf D_{n+1})_n
		=
		\mathbf D_n^*\mathbf D_n+H_n,
	\]
	donde
	\[
		H_n=
		\begin{pmatrix}
			0      & \cdots & 0      & 0             \\
			\vdots & \ddots & \vdots & \vdots        \\
			0      & \cdots & 0      & 0             \\
			0      & \cdots & 0      & |d_{n+1,n}|^2
		\end{pmatrix}
		\in\mathcal M_{n+1}(\C).
	\]
	En particular, \(H_n\) es hermitiana semidefinida positiva (HPSD).
\end{lem}

\begin{proof}
	Por la estructura de Hessenberg,
	\[
		\mathbf D_{n+1}=
		\begin{pmatrix}
			\mathbf D_n & u_n         \\
			w_n         & d_{n+1,n+1}
		\end{pmatrix}
	\]
	para ciertos vectores \(u_n\in\C^{n+1}\) y \(w_n\in(\C^{n+1})^*\), donde
	\[
		w_n=(0,\dots,0,d_{n+1,n}).
	\]
	Al multiplicar por bloques se obtiene que el bloque principal superior izquierdo de \(\mathbf D_{n+1}^*\mathbf D_{n+1}\) es
	\[
		\mathbf D_n^*\mathbf D_n+
		\begin{pmatrix}
			0      & \cdots & 0      & 0             \\
			\vdots & \ddots & \vdots & \vdots        \\
			0      & \cdots & 0      & 0             \\
			0      & \cdots & 0      & |d_{n+1,n}|^2
		\end{pmatrix}.
	\]
	La última matriz es hermitiana semidefinida positiva (HPSD).
\end{proof}

\begin{prop}\label{prop:monotonia_sigma_kn}
	Sea \(\mathbf D\) una matriz de Hessenberg superior infinita y, para cada \(n\ge0\), sean
	\[
		\sigma_{0,n}\ge\sigma_{1,n}\ge\cdots\ge\sigma_{n,n}\ge0
	\]
	los valores singulares de \(\mathbf D_n\) ordenados de forma decreciente. Entonces, para todo \(k\ge0\) fijo, la sucesión
	\(
		\{\sigma_{k,n}\}_{n\ge k}
	\)
	es creciente. En particular,
	\[
		\lim_{n\to\infty}\sigma_{k,n}\in[0,+\infty]
	\]
	existe para todo \(k\ge0\).
\end{prop}

\begin{proof}
	Fijado \(k\ge0\), para cada \(n\ge k\) denotemos por
	\[
		\lambda_{0,n}\ge\lambda_{1,n}\ge\cdots\ge\lambda_{n,n}\ge0
	\]
	los autovalores de \(\mathbf D_n^*\mathbf D_n\). Entonces
	\[
		\sigma_{k,n}^2=\lambda_{k,n}.
	\]
	Por el Lema~\ref{lem:bloque_gram_truncadas},
	\[
		(\mathbf D_{n+1}^*\mathbf D_{n+1})_n
		=
		\mathbf D_n^*\mathbf D_n+H_n,
		\qquad H_n\succeq0.
	\]
	La monotonía de Weyl para perturbaciones hermitianas semidefinidas positivas y el teorema de entrelazamiento de Cauchy para submatrices principales dan, en la forma estándar, las desigualdades siguientes; véase \cite[Chapter~4]{horn2012matrix}.
	Por tanto,
	\[
		\lambda_{k,n}
		\le
		\lambda_k\bigl((\mathbf D_{n+1}^*\mathbf D_{n+1})_n\bigr),
	\]
	y además
	\[
		\lambda_k\bigl((\mathbf D_{n+1}^*\mathbf D_{n+1})_n\bigr)
		\le
		\lambda_{k,n+1}.
	\]
	Por tanto,
	\[
		\lambda_{k,n}\le\lambda_{k,n+1},
		\qquad n\ge k,
	\]
	y, en consecuencia,
	\[
		\sigma_{k,n}\le\sigma_{k,n+1},
		\qquad n\ge k.
	\]
	La existencia del límite se sigue de la monotonía.
\end{proof}

\begin{defn}\label{def:sigma_operador}
	Para cada \(k\ge0\), donde \(\sigma_{k,n}\) está definido en la Definición~\ref{def:sigma_kn}, definimos como valores singulares de la representación matricial \(\mathbf D\) los números
	\[
		\sigma_k(\mathbf D):=\lim_{n\to\infty}\sigma_{k,n}\in[0,+\infty].
	\]
\end{defn}

\begin{prop}\label{prop:sigma_qk}
	Para todo \(n\ge0\), sea \(\mathbf D_n\in\mathcal M_{n+1}(\C)\) la sección principal de \(\mathbf D\) en la base \(\{\phi_0,\dots,\phi_n\}\) y sean \(\sigma_{0,n}\ge\cdots\ge\sigma_{n,n}\ge0\) sus valores singulares. Entonces, para todo \(0\le k\le n\),
	\[
		\sigma_{k,n}\le Q_k(\D).
	\]
\end{prop}

\begin{proof}
	Fijemos $n\ge0$ y $0\le k\le n$. Sea $V\subset\Poly$ un subespacio con
	\(
		\operatorname{codim}_{\Poly}(V)<k+1.
	\)
	Consideremos \(V_n:=V\cap\mathbb P_n\).
	La aplicación natural
	\[
		\mathbb P_n/(V\cap\mathbb P_n)\longrightarrow \Poly/V,
		\qquad
		p+(V\cap\mathbb P_n)\longmapsto p+V,
	\]
	es inyectiva. En efecto, si $p\in\mathbb P_n$ verifica $p+V=V$, entonces $p\in V$, y por tanto $p\in V\cap\mathbb P_n$.
	En consecuencia,
	\[
		\operatorname{codim}_{\mathbb P_n}(V\cap\mathbb P_n)
		\le
		\operatorname{codim}_{\Poly}(V)
		<k+1.
	\]
	Al pasar a coordenadas en la base $\{\phi_0,\dots,\phi_n\}$ obtenemos un subespacio $S_n\subset\mathbb C^{n+1}$ con codimensión menor que $k+1$, o equivalentemente con dimensión al menos $n+1-k$. Por Courant--Fischer,
	\[
		\sigma_{k,n}
		\le
		\max_{\substack{x\in S_n\\x\neq0}}\frac{\|\mathbf D_nx\|_2}{\|x\|_2}.
	\]
	Si $x$ corresponde al polinomio $p\in V_n$, entonces
	\[
		\|\mathbf D_nx\|_2=\|\pi_n(\D p)\|_S\le \|\D p\|_S\le \|\D|_V\|_S\,\|p\|_S=\|\D|_V\|_S\,\|x\|_2.
	\]
	Por tanto, \(\sigma_{k,n}\le \|\D|_V\|_S\).
	Tomando ínfimo sobre todos los subespacios $V$ con codimensión $<k+1$, obtenemos
	\[
		\sigma_{k,n}\le Q_k(\D).
	\]
\end{proof}

\begin{cor}\label{cor:sigma_operador_qk}
	Para el operador \(\D\) sobre \((\Poly,\langle\cdot,\cdot\rangle_S)\) y su representación matricial \(\mathbf D\), para todo \(k\ge0\) se tiene
	\[
		\sigma_k(\mathbf D)\le Q_k(\D).
	\]
\end{cor}

\begin{proof}
	Para cada $n\ge k$, la Proposición~\ref{prop:sigma_qk} da \(\sigma_{k,n}\le Q_k(\D)\).
	Pasando al límite cuando $n\to\infty$, obtenemos
	\[
		\sigma_k(\mathbf D)\le Q_k(\D).
	\]
\end{proof}

\begin{thm}\label{thm:igualdad_sigma_qk_general}
	Para el operador \(\D\) sobre \((\Poly,\langle\cdot,\cdot\rangle_S)\) y su representación matricial \(\mathbf D\), para todo \(k\ge0\) se tiene
	\[
		\sigma_k(\mathbf D)=Q_k(\D).
	\]
\end{thm}

\begin{proof}
	La desigualdad \(\sigma_k(\mathbf D)\le Q_k(\D)\) es el contenido del Corolario~\ref{cor:sigma_operador_qk}. Probemos la desigualdad contraria.

    Sea $0 <a<Q_k(\D)$, donde puede ocurrir que $Q_k(\D)=+\infty$. Por la caracterización max--min del Teorema~\ref{thm:maxmin_Qk}, existe un subespacio $E\subset\Poly$ con
    \(\dim E=k+1\)
    tal que
    \[
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|\D p\|_S}{\|p\|_S}>a.
    \]
    Como $E$ es finito-dimensional y $E \subset \Poly$, existe $N\ge0$ tal que \(E\subset\mathbb P_N\). Entonces \(\D(E)\subset\mathbb P_{N+1}\).
    Por tanto, para todo $n\ge N+1$ y todo $p\in E$,
    \[
        \mathcal D_n p=\pi_n(\D p)=\D p.
    \]

    Aplicando la forma max--min del Lema~\ref{lem:minmax_sigma_Dn} a $\mathbf D_n$, para $n\ge N+1$ obtenemos
    \[
        \sigma_{k,n}
        \ge
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|\mathcal D_n p\|_S}{\|p\|_S}
        =
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|\D p\|_S}{\|p\|_S}
        >a.
    \]
    Pasando al límite, \(\sigma_k(\mathbf D)\ge a\).
    Como esto se verifica para $0 <a<Q_k(\D)$ se concluye que
    \[
        \sigma_k(\mathbf D)\ge Q_k(\D).
    \]
    Este resultado unido con la desigualdad contraria, esto prueba la igualdad.
\end{proof}

En el caso de operadores acotados, el Teorema~\ref{thm:equivalencia_Qk_Gelfand} identifica el índice $Q_k$ con el número de Gelfand del operador. Combinando ese hecho con el resultado anterior obtenemos el siguiente corolario.

\begin{cor}\label{cor:caso_acotado_ck}
	Si el operador de multiplicación \(\D\) se extiende como elemento de \(\mathcal L(\mathcal H)\), entonces para todo \(k\ge0\),
	\[
		\sigma_k(\mathbf D)=c_k(\D).
	\]
\end{cor}

\begin{proof}
	Por el Teorema~\ref{thm:igualdad_sigma_qk_general},
	\(\sigma_k(\mathbf D)=Q_k(\D)\).
	Como \(\D\in\mathcal L(\mathcal H)\), el Teorema~\ref{thm:equivalencia_Qk_Gelfand} da
	\(Q_k(\D)=c_k(\D)\).
	Se concluye que
	\[
		\sigma_k(\mathbf D)=c_k(\D).
	\]
\end{proof}

\section{Análisis asintótico espectral de \texorpdfstring{$\mathbf D$}{D}}

Una vez identificados los valores singulares límite $\sigma_k(\mathbf D)$ de la representación matricial con los índices $Q_k(\D)$, podemos aplicar los resultados del Teorema~\ref{thm:estabilizacion} para obtener información sobre la estabilización de los valores singulares de las secciones principales.

\begin{thm}\label{thm:igualdad_sigma_qN_general}
	Sean $R>0$ y $F\subset\C$ un conjunto finito de puntos, consideremos sobre $\Poly$ el producto escalar
	\[
		\langle p,q\rangle_S
		=
		\int_{\mathbb{S}_R}p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
		+
		\sum_{c\in F} p'(c)\overline{q'(c)},
		\qquad p,q\in\Poly.
	\]
	Sea $\mathbf D$ la representación matricial del operador $\D$. Si $N:=|F|$, entonces, para todo $k\ge N$,
	\[
		\sigma_k(\mathbf D)=Q_k(\D)=R.
	\]
\end{thm}

\begin{proof}
	Por el Teorema~\ref{thm:qk_vale_R_desde_N}, \(Q_k(\D)=R\) para todo \(k\ge N\). Por el Teorema~\ref{thm:igualdad_sigma_qk_general}, \(\sigma_k(\mathbf D)=Q_k(\D)\).
	Se concluye que
	\[
		\sigma_k(\mathbf D)=Q_k(\D)=R,
		\qquad k\ge N.
	\]
\end{proof}

\begin{cor}\label{cor:igualdad_sigma_qk_exterior_puro}
	Sea $R>0$ y sea el producto escalar de Sobolev sobre \(\Poly\) dado por
	\[
		\langle p,q\rangle_S
		=
		\int_{\mathbb{S}_R}p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
		+
		\sum_{c\in F} p'(c)\overline{q'(c)},
		\qquad p,q\in\Poly,
	\]
	donde $F\subset\C$ es un conjunto finito de puntos y satisface $|c|\ge R$ para todo $c\in F$. Sea $m:=|F|$ y sea $\mathbf D$ la representación matricial del operador $\D$. Entonces, para todo $k\ge 0$ se tiene
	\[
		\sigma_k(\mathbf D)
		=
		\begin{cases}
			+\infty, & 0\le k<m, \\
			R,       & k\ge m.
		\end{cases}
	\]
\end{cor}

\begin{proof}
	Como en este caso $I_{\mathrm{out}}=F$, se tiene $m=N=|F|$. Por el Teorema~\ref{thm:descripcion_global_qk},
	\[
		Q_k(\D)=
		\begin{cases}
			+\infty, & 0\le k<m, \\
			R,       & k\ge m.
		\end{cases}
	\]
	Aplicando ahora el Teorema~\ref{thm:igualdad_sigma_qk_general}, se concluye que
	\[
		\sigma_k(\mathbf D)=Q_k(\D)
		=
		\begin{cases}
			+\infty, & 0\le k<m, \\
			R,       & k\ge m.
		\end{cases}
	\]
\end{proof}

\begin{cor}\label{cor:estabilizacion_singular_umbral}
	Sea $R>0$, sea $F\subset\C$ finito y consideremos sobre $\Poly$ el producto escalar
	\[
		\langle p,q\rangle_S
		=
		\int_{\mathbb{S}_R}p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
		+
		\sum_{c\in F} p'(c)\overline{q'(c)},
		\qquad p,q\in\Poly.
	\]
	Sea
	\[
		I_{\mathrm{out}}:=\{c\in F:|c|\ge R\},
		\qquad
		m:=|I_{\mathrm{out}}|.
	\]
	Sea además \(\mathbf D\) la representación matricial del operador \(\D\).
	Si $m\ge1$, entonces
	\[
		\sigma_m(\mathbf D)=Q_m(\D)<\infty.
	\]
\end{cor}

\begin{proof}
	Por el Teorema~\ref{thm:igualdad_sigma_qk_general},
	\[
		\sigma_m(\mathbf D)=Q_m(\D).
	\]
	Como $Q_m(\D)<\infty$ por el Teorema~\ref{thm:estabilizacion}, se concluye el resultado.
\end{proof}

\begin{rmk}
	En el caso general con $N=|F|$,
	\[
		\sigma_k(\mathbf D)=Q_k(\D)=R,
		\qquad k\ge N.
	\]
	En el caso exterior con $I_{\mathrm{in}}=\varnothing$ y $m=|I_{\mathrm{out}}|$,
	\[
		\sigma_j(\mathbf D)=Q_j(\D)=+\infty,
		\qquad 0\le j<m,
	\]
	y
	\[
		\sigma_k(\mathbf D)=Q_k(\D)=R,
		\qquad k\ge m.
	\]
	En el caso mixto se obtiene la estabilización matricial en el umbral,
	\[
		\sigma_m(\mathbf D)=Q_m(\D)<\infty,
	\]
	 aunque el valor exacto de este índice no queda determinado por el Teorema~\ref{thm:estabilizacion}.
\end{rmk}

\subsection{Ejemplos}

En los ejemplos siguientes fijamos $R=1$ y dejamos indicados los índices singulares relevantes para contrastar los resultados teóricos con los cálculos numéricos de las truncaciones. En cada caso puede sustituirse el recuadro por la imagen generada computacionalmente del soporte del producto de Sobolev.

\begin{exm}
	Consideremos el producto escalar
	\[
		\langle p,q\rangle_S
		=
		\int_{\mathbb S_1}p(z)\overline{q(z)}\,d\mathbf m_1(z)
		+
		p'(0)\overline{q'(0)}
		+
		p'\!\left(\tfrac12\right)\overline{q'\!\left(\tfrac12\right)}
		+
		p'\!\left(-\tfrac13\right)\overline{q'\!\left(-\tfrac13\right)}
		+
		p'\!\left(\tfrac{i}{2}\right)\overline{q'\!\left(\tfrac{i}{2}\right)},
		\qquad p,q\in\Poly.
	\]
	En este caso
	\[
		F=\left\{0,\tfrac12,-\tfrac13,\tfrac{i}{2}\right\},
		\qquad
		I_{\mathrm{out}}=\varnothing,
		\qquad
		m=0,
		\qquad
		N=4.
	\]
	Por la Proposición~\ref{prop:atomos_interiores_acotado}, el operador $\D$ es acotado; por tanto no aparece ningún índice con valor singular infinito. Además, por el Teorema~\ref{thm:igualdad_sigma_qN_general},
	\[
		\sigma_k(\mathbf D)=Q_k(\D)=1,
		\qquad k\ge 4.
	\]

	\begin{figure}[H]
		\centering
		\fbox{\parbox[c][4cm][c]{0.78\linewidth}{\centering
			Sustituir por la imagen del soporte del producto:\\[2mm]
			circunferencia $\mathbb S_1$ y átomos en $0$, $\tfrac12$, $-\tfrac13$ y $\tfrac{i}{2}$.}}
		\caption{Soporte del producto de Sobolev en el ejemplo con átomos interiores.}
		\label{fig:ejemplo_interior_puro}
	\end{figure}

	Los índices relevantes para completar numéricamente pueden resumirse en la tabla siguiente:
	\begin{center}
		\begin{tabular}{|c|c|}
			\hline
			Índice & Valor \\
			\hline
			$\sigma_0(\mathbf D)$ & $\boxed{\text{valor numérico a insertar}}<\infty$ \\
			\hline
			$\sigma_4(\mathbf D)$ & $1$ \\
			\hline
		\end{tabular}
	\end{center}
	En este ejemplo no aparece ningún índice con valor infinito. El interés del cuadro está en separar el primer valor finito, $\sigma_0(\mathbf D)$, del primer índice cuyo valor queda forzado por la teoría general, que es $k=N=4$.
\end{exm}

\begin{exm}
	Consideremos ahora
	\[
		\langle p,q\rangle_S
		=
		\int_{\mathbb S_1}p(z)\overline{q(z)}\,d\mathbf m_1(z)
		+
		p'(2)\overline{q'(2)}
		+
		p'(-2)\overline{q'(-2)}
		+
		p'\!\left(\tfrac{3i}{2}\right)\overline{q'\!\left(\tfrac{3i}{2}\right)}
		+
		p'\!\left(-\tfrac{3i}{2}\right)\overline{q'\!\left(-\tfrac{3i}{2}\right)},
		\qquad p,q\in\Poly.
	\]
	Entonces
	\[
		F=\left\{2,-2,\tfrac{3i}{2},-\tfrac{3i}{2}\right\},
		\qquad
		I_{\mathrm{out}}=F,
		\qquad
		m=4,
		\qquad
		N=4.
	\]
	Por el Corolario~\ref{cor:igualdad_sigma_qk_exterior_puro},
	\[
		\sigma_k(\mathbf D)=Q_k(\D)
		=
		\begin{cases}
			+\infty, & 0\le k<4, \\
			1,       & k\ge 4.
		\end{cases}
	\]

	\begin{figure}[H]
		\centering
		\fbox{\parbox[c][4cm][c]{0.78\linewidth}{\centering
			Sustituir por la imagen del soporte del producto:\\[2mm]
			circunferencia $\mathbb S_1$ y átomos en $2$, $-2$, $\tfrac{3i}{2}$ y $-\tfrac{3i}{2}$.}}
		\caption{Soporte del producto de Sobolev en el ejemplo con átomos exteriores.}
		\label{fig:ejemplo_exterior_puro}
	\end{figure}

	Los índices relevantes pueden organizarse así:
	\begin{center}
		\begin{tabular}{|c|c|}
			\hline
			Índice & Valor \\
			\hline
			$\sigma_3(\mathbf D)$ & $+\infty$ \\
			\hline
			$\sigma_4(\mathbf D)$ & $1$ \\
			\hline
		\end{tabular}
	\end{center}
	Aquí el contraste es inmediato: $k=3$ es el último índice con valor infinito, mientras que $k=4$ ya es finito y además coincide con el valor general $R=1$.
\end{exm}

\begin{exm}
	Finalmente, consideremos el caso mixto
	\[
		\langle p,q\rangle_S
		=
		\int_{\mathbb S_1}p(z)\overline{q(z)}\,d\mathbf m_1(z)
		+
		p'(2)\overline{q'(2)}
		+
		p'(-2)\overline{q'(-2)}
		+
		p'\!\left(\tfrac{3i}{2}\right)\overline{q'\!\left(\tfrac{3i}{2}\right)}
		+
		p'(0)\overline{q'(0)}
		+
		p'\!\left(\tfrac12\right)\overline{q'\!\left(\tfrac12\right)},
		\qquad p,q\in\Poly.
	\]
	Ahora
	\[
		F=\left\{2,-2,\tfrac{3i}{2},0,\tfrac12\right\},
		\qquad
		I_{\mathrm{out}}=\left\{2,-2,\tfrac{3i}{2}\right\},
		\qquad
		I_{\mathrm{in}}=\left\{0,\tfrac12\right\},
		\qquad
		m=3,
		\qquad
		N=5.
	\]
	Por el Teorema~\ref{thm:estabilizacion},
	\[
		\sigma_0(\mathbf D)=\sigma_1(\mathbf D)=\sigma_2(\mathbf D)=+\infty,
		\qquad
		\sigma_3(\mathbf D)<\infty.
	\]
	Además, por el Teorema~\ref{thm:igualdad_sigma_qN_general},
	\[
		\sigma_k(\mathbf D)=Q_k(\D)=1,
		\qquad k\ge 5.
	\]

	\begin{figure}[H]
		\centering
		\fbox{\parbox[c][4cm][c]{0.78\linewidth}{\centering
			Sustituir por la imagen del soporte del producto:\\[2mm]
			circunferencia $\mathbb S_1$, átomos exteriores en $2$, $-2$ y $\tfrac{3i}{2}$,\\
			y átomos interiores en $0$ y $\tfrac12$.}}
		\caption{Soporte del producto de Sobolev en el ejemplo mixto.}
		\label{fig:ejemplo_mixto}
	\end{figure}

	Los índices destacados quedan resumidos en la tabla:
	\begin{center}
		\begin{tabular}{|c|c|}
			\hline
			Índice & Valor \\
			\hline
			$\sigma_2(\mathbf D)$ & $+\infty$ \\
			\hline
			$\sigma_3(\mathbf D)$ & $\boxed{\text{valor numérico a insertar}}<\infty$ \\
			\hline
			$\sigma_5(\mathbf D)$ & $1$ \\
			\hline
		\end{tabular}
	\end{center}
	Este ejemplo muestra con claridad la diferencia entre el umbral de finitud $m=3$ y el índice $N=5$: en $k=3$ ya se alcanza la finitud, pero el valor singular no queda fijado por la cota general $R=1$ hasta llegar a $k=5$.
\end{exm}
