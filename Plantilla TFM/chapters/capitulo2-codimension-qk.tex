\chapter{Codimensión, Dicotomía topológica e índice \texorpdfstring{$Q_k$}{Qk}}\label{sec:codimension_qk}

\section{Marco hilbertiano y codimensión topológica}

Partimos de un producto interno $\langle\cdot, \cdot\rangle$. Denotamos por $\mathcal H$ la compleción de $\Poly$ respecto de la norma inducida por ese producto. En particular,
\[
    \mathcal H=\overline{\Poly},
\]
y $\Poly$ es un subespacio denso de $\mathcal H$.

La noción clásica de codimensión en este contexto es la siguiente.

\begin{defn}[Codimensión topológica en Hilbert \cite{halmos1982hilbert}]\label{def:codim_top_hilbert}
    Sea $\mathcal H$ un espacio de Hilbert y sea $M\subset \mathcal H$ un subespacio cerrado. Definimos la codimensión de $M$ en $\mathcal H$:
    \[
        \operatorname{codim}_{\mathcal H}(M):=\dim(\mathcal H/M).
    \]
    Equivalentemente,
    \[
        \operatorname{codim}_{\mathcal H}(M)=\dim(M^\perp).
    \]
\end{defn}

\section{Límite de la codimensión topológica}

El punto delicado aparece cuando un subespacio de $\Poly$ está definido por anulación de funcionales lineales y se lo mira dentro de la compleción hilbertiana $\mathcal H$. En ese paso, la condición lineal puede sobrevivir o desaparecer tras tomar clausura, según el funcional sea acotado o no respecto de la norma de $\mathcal H$. La proposición siguiente describe exactamente esta dicotomía.

\begin{prop}[Dicotomía topológica para un funcional]\label{thm:dicotomia_1}
    Sea $(\mathcal H,\langle\cdot,\cdot\rangle)$ un espacio de Hilbert tal que $\mathcal H=\overline{\Poly}$, y sea $L:\Poly\to\mathbb C$ un funcional lineal no nulo. Entonces ocurre exactamente una de las dos posibilidades siguientes:
    \begin{enumerate}
        \item si $L$ no es acotado respecto de la norma de $\mathcal H$, entonces
              \[
                  \overline{\ker L}=\mathcal H,
                  \qquad
                  \operatorname{codim}_{\mathcal H}(\overline{\ker L})=0;
              \]
        \item si $L$ es acotado respecto de la norma de $\mathcal H$, entonces
              \[
                  \operatorname{codim}_{\mathcal H}(\overline{\ker L})=1.
              \]
    \end{enumerate}
\end{prop}

\begin{proof}
    Supongamos que $L$ no es acotado. Fijado $p\in\Poly$, existe una sucesión $w_n\in\Poly$ tal que
    \[
        |L(w_n)|>n\|w_n\|.
    \]
    Definimos
    \[
        u_n:=\frac{w_n}{L(w_n)}.
    \]
    Entonces
    \[
        L(u_n)=1,
        \qquad
        \|u_n\|=\frac{\|w_n\|}{|L(w_n)|}<\frac1n,
    \]
    de modo que $\|u_n\|\to0$. Sea
    \[
        v_n:=p-L(p)u_n\in\ker L,
        \qquad
        \|v_n-p\|=|L(p)|\,\|u_n\|\to 0.
    \]
    Como $\Poly$ es denso en $\mathcal H$, se sigue que $\overline{\ker L}=\mathcal H$.

    Supongamos que $L$ es acotado. Entonces se extiende de manera única a un funcional lineal acotado $\varphi$ sobre $\mathcal H$. Como $\ker\varphi$ es cerrado y contiene a $\ker L$,
    \[
        \overline{\ker L}^{\,\mathcal H}=\ker\varphi.
    \]
    Tomemos $q\in\Poly$ con $L(q)\neq 0$. Para todo $x\in\mathcal H$,
    \[
        x=\left(x-\frac{\varphi(x)}{L(q)}q\right)+\frac{\varphi(x)}{L(q)}q,
    \]
    y el primer sumando pertenece a $\ker\varphi$. Por tanto,
    \[
        \mathcal H=\ker\varphi+[q].
    \]
    Como $q\notin\ker\varphi$, esa suma es directa. En consecuencia,
    \[
        \operatorname{codim}_{\mathcal H}(\overline{\ker L})=\operatorname{codim}_{\mathcal H}(\ker\varphi)=1.
    \]
\end{proof}

\begin{prop}
    Sea $\mathcal H$ un espacio de Hilbert y sean $\varphi_1,\dots,\varphi_N\in\mathcal H^*$ linealmente independientes. Si
    \[
        M:=\bigcap_{i=1}^N \ker(\varphi_i),
    \]
    entonces
    \[
        \operatorname{codim}_{\mathcal H}(M)=N.
    \]
\end{prop}

\begin{proof}
    Por el teorema de Riesz, existen $f_1,\dots,f_N\in\mathcal H$ linealmente independientes tales que
    \[
        \varphi_i(x)=\langle x,f_i\rangle_{\mathcal H},
        \qquad x\in\mathcal H,
    \]
    para $i=1,\dots,N$. Sea
    \[
        F=[f_1, \dots, f_N].
    \]
    Entonces
    \[
        x\in M
        \iff \varphi_i(x)=0\ \text{para todo }i
        \iff \langle x,f_i\rangle_{\mathcal H}=0\ \text{para todo }i
        \iff x\perp F,
    \]
    de modo que $M=F^{\perp}$. Como los vectores $f_1,\dots,f_N$ son linealmente independientes, se tiene $\dim(F)=N$, y por la descomposición ortogonal $\mathcal H=F^{\perp}\oplus F$ concluimos que
    \[
        \operatorname{codim}_{\mathcal{H}}(M)=N.
    \]
\end{proof}

\section{Codimensión algebraica y restricciones lineales finitas}

La definición que introducimos ahora no responde a una noción esencialmente distinta de codimensión, sino a la misma construcción aplicada en otro espacio ambiente. Si trabajamos en $\Poly$, la cantidad
\[
    \dim(\Poly/V)
\]
cuenta exactamente el número de restricciones lineales impuestas por $V$, sin que intervenga todavía ningún proceso de clausura. Esto permite conservar la información lineal que puede perderse al pasar a $\mathcal H$.

\begin{defn}\label{def:codim_alg}
    Sea $V\subset \Poly$ un subespacio vectorial. Definimos
    \[
        \operatorname{codim}_{\Poly}(V):=\dim(\Poly/V)\in\mathbb N\cup\{\infty\},
    \]
    donde la dimensión es algebraica, es decir, la cardinalidad de una base de Hamel (una base vectorial en el sentido puramente algebraico) \cite{hormander1989linearfunctionalanalysis,Aizpuru2009Apuntes}.
\end{defn}

La primera caracterización es la que usaremos de manera sistemática.

\begin{prop}\label{prop:finite-codim-kernels}
    Para un subespacio vectorial $V\subset \Poly$, son equivalentes:
    \begin{enumerate}
        \item $\operatorname{codim}_{\Poly}(V)<\infty$;
        \item existen $m\in\mathbb N$ y $L_1,\dots,L_m\in\Poly^*$ tales que
              \[
                   V=\bigcap_{j=1}^m \ker L_j.
               \]
    \end{enumerate}
    Además,
    \[
        \operatorname{codim}_{\Poly}(V)
        =
        \min\Bigl\{m\in\mathbb N: \exists\,L_1,\dots,L_m\in\Poly^*\ \text{tales que}\ V=\bigcap_{j=1}^m\ker L_j\Bigr\},
    \]
    y más precisamente, si $V=\bigcap_{j=1}^m\ker L_j$, entonces
    \[
        \operatorname{codim}_{\Poly}(V)=\dim[L_1,\dots,L_m].
    \]
\end{prop}

\begin{proof}
    Supongamos que $\operatorname{codim}_{\Poly}(V)=n<\infty$. Entonces existe un subespacio complementario $F\subset\Poly$ tal que
    \[
        \Poly=V\oplus F,
        \qquad
        \dim(F)=n.
    \]
    Tomemos una base $u_1,\dots,u_n$ de $F$ y sea $u_1^*,\dots,u_n^*$ la base dual de $F^*$. Para cada $p\in\Poly$, escribiendo de manera única $p=v+f$ con $v\in V$ y $f\in F$, definimos
    \[
        L_j(p):=u_j^*(f),\qquad j=1,\dots,n.
    \]
    Entonces $L_j\in\Poly^*$ y
    \[
        V=\bigcap_{j=1}^n\ker L_j,
    \]
    pues $L_j(p)=0$ para todo $j$ si y solo si la componente $f$ es nula.

    Supongamos ahora que
    \[
        V=\bigcap_{j=1}^m\ker L_j.
    \]
    Sea
    \[
        r:=\dim[L_1,\dots,L_m].
    \]
    Reordenando si hace falta, podemos suponer que $L_1,\dots,L_r$ forman una base de ese subespacio. Entonces
    \[
        V=\bigcap_{i=1}^r\ker L_i.
    \]
    Consideremos la aplicación lineal
    \[
        \Psi:\Poly\to\mathbb C^r,
        \qquad
        \Psi(p)=(L_1(p),\dots,L_r(p)).
    \]
    Como $L_1,\dots,L_r$ son linealmente independientes, $\Psi$ es sobreyectiva. Además,
    \[
        \ker\Psi=V.
    \]
    Por el primer teorema de isomorfía,
    \[
        \Poly/V\cong\mathbb C^r,
    \]
    y por tanto
    \[
        \operatorname{codim}_{\Poly}(V)=r=
        \dim[L_1,\dots,L_m].
    \]
    La fórmula del mínimo se deduce de inmediato de esta identidad.
\end{proof}

A continuación damos una caracterización de la codimensión algebraica atendiendo a la ortogonalidad dada por el producto escalar.

\begin{prop}\label{prop:equiv_ortho_interseccion}
    Sea $\langle\cdot,\cdot\rangle$ un producto interno en $\Poly$, sean $f_1,\dots,f_m\in\Poly$ y sea $F=[f_1,\dots,f_m]$. Para $j=1,\dots,m$, definimos
    \[
        L_j(p):=\langle p,f_j\rangle,
        \qquad p\in\Poly.
    \]
    Entonces
    \[
        F^{\perp}_{\Poly}=\{p\in\Poly:\langle p,f_j\rangle=0,\ j=1,\dots,m\}=\bigcap_{j=1}^m\ker(L_j).
    \]
    Además,
    \[
        \operatorname{codim}_{\Poly}(F^{\perp}_{\Poly})
        =
        \dim F.
    \]
    En particular, si $f_1,\dots,f_m$ son linealmente independientes, entonces
    \[
        \operatorname{codim}_{\Poly}(F^{\perp}_{\Poly})=m.
    \]
\end{prop}

\begin{proof}
    Se tiene
    \[
        F^{\perp}_{\Poly}=\bigcap_{j=1}^m\ker(L_j)
    \]
    por definición.

    Sea ahora
    \[
        r:=\dim F.
    \]
    Reordenando si hace falta, podemos suponer que $f_1,\dots,f_r$ forman una base de ese subespacio. Entonces, para cada $j>r$, el vector $f_j$ es combinación lineal de $f_1,\dots,f_r$, y por sesquilinealidad del producto interno se sigue que $L_j$ es combinación lineal de $L_1,\dots,L_r$ y
    \[
        \bigcap_{i=1}^r \operatorname{ker}(L_i) \subset \operatorname{ker}(L_j).
    \]
    Por tanto,
    \[
        F^{\perp}_{\Poly}=\bigcap_{i=1}^r\ker(L_i).
    \]

    Si $\sum_{i=1}^r\alpha_iL_i=0$, tomando
    \(
    q=\sum_{i=1}^r\overline{\alpha_i}f_i
    \)
    se obtiene
    \[
        0=\sum_{i=1}^r\alpha_iL_i(q)=\langle q,q\rangle.
    \]
    Luego $q=0$, y como $f_1,\dots,f_r$ son linealmente independientes, $\alpha_1=\cdots=\alpha_r=0$. Por tanto, $L_1,\dots,L_r$ son linealmente independientes.

    Aplicando la Proposición~\ref{prop:finite-codim-kernels} a $L_1,\dots,L_r$, concluimos que
    \[
        \operatorname{codim}_{\Poly}(F^{\perp}_{\Poly})=r.
    \]
\end{proof}

En este trabajo son relevantes los siguientes funcionales de evaluación y evaluación de la derivada en un punto.

\begin{defn}\label{def:funcionales_restringidos}
    Sea $c\in\C$. Definimos los funcionales lineales
    \[
        \delta_c:\Poly\to\C,
        \qquad
        \delta_c(p)=p(c),
    \]
    y
    \[
        \delta'_c:\Poly\to\C,
        \qquad
        \delta'_c(p)=p'(c).
    \]
\end{defn}

\begin{lem}
    Sean $z_1,\dots,z_N\in\C$ distintos. Entonces $\delta_{z_1},\dots,\delta_{z_N}$ son linealmente independientes en $\Poly^*$.
\end{lem}

\begin{proof}
    Supongamos que existe una combinación lineal nula $\sum_{i=1}^N \alpha_i \delta_{z_i} = 0$.
    Para cada $j\in\{1,\dots,N\}$, sea
    \[
        L_j(z) = \prod_{\substack{l=1 \\ l \neq j}}^{N} \frac{z - z_l}{z_j - z_l}
    \]
    Entonces $L_j(z_i)=\delta_{ji}$. Evaluando en $L_j$,
    \[
        0 = \sum_{i=1}^N \alpha_i \delta_{z_i}(L_j) = \sum_{i=1}^N \alpha_i L_j(z_i) = \alpha_j \cdot 1
    \]
    para todo $j$. Luego $\alpha_j=0$ para todo $j$.
\end{proof}

\begin{cor}
    Sean $z_1,\dots,z_N\in\C$ distintos y sea
    \[
        V:=\bigcap_{i=1}^N \ker(\delta_{z_i}).
    \]
    Entonces
    \[
        \operatorname{codim}_{\Poly}(V)=N.
    \]
\end{cor}

\begin{proof}
    Por el lema anterior, $\delta_{z_1},\dots,\delta_{z_N}$ son linealmente independientes. Aplicando la Proposición~\ref{prop:finite-codim-kernels},
    \[
        \operatorname{codim}_{\Poly}(V) = \dim[\delta_{z_1}, \dots, \delta_{z_N}] = N.
    \]
\end{proof}

\section{Relación entre codimensión algebraica y topológica}

En esta sección estudiaremos la relación entre las codimensiones algebraicas y topológicas introducidas en las secciones anteriores.

La proposición siguiente precisa cómo se relacionan ambas codimensiones.

\begin{prop}\label{prop:relacion_codimensiones}
    Sea $\mathcal H=\overline{\Poly}$ con respecto a un producto interno.
    \begin{enumerate}
        \item Si $V\subset\Poly$ satisface $\operatorname{codim}_{\Poly}(V)=m<\infty$, entonces
              \[
                  \operatorname{codim}_{\mathcal H}(\overline V)\le m.
              \]
        \item Si $M\subset\mathcal H$ es cerrado y $\operatorname{codim}_{\mathcal H}(M)=m<\infty$, entonces
              \[
                  \overline{M\cap\Poly}=M
                  \qquad\text{y}\qquad
                  \operatorname{codim}_{\Poly}(M\cap\Poly)=m.
              \]
    \end{enumerate}
\end{prop}

\begin{proof}
    Para $(1)$, tomemos un subespacio complementario $F\subset\Poly$ con
    \[
        \Poly=V\oplus F,
        \qquad
        \dim(F)=m.
    \]
    Como $F$ es finito-dimensional, es cerrado en $\mathcal H$. Además, $\overline V+F$ es cerrado: si $x_n=v_n+f_n\in \overline V+F$ y $x_n\to x$, entonces $(x_n)$ es acotada; como $F$ es finito-dimensional, de $(f_n)$ extraemos una subsucesión $(f_{n_k})$ convergente a cierto $f\in F$, y por tanto
    \[
        v_{n_k}=x_{n_k}-f_{n_k}\to x-f.
    \]
    Como $\overline V$ es cerrado, $x-f\in\overline V$, luego $x\in\overline V+F$. Por otra parte,
    \[
        V+F\subset \overline V+F,
    \]
    así que
    \[
        \overline{V+F}\subset \overline V+F.
    \]
    Recíprocamente, si $x=v+f\in \overline V+F$ con $v\in\overline V$ y $f\in F$, tomando $v_n\in V$ con $v_n\to v$ se tiene
    \[
        v_n+f\in V+F,
        \qquad
        v_n+f\to x,
    \]
    y por tanto $x\in\overline{V+F}$. Luego
    \[
        \overline{V+F}=\overline V+F.
    \]
    Como $\Poly=V\oplus F$, se sigue que
    \[
        \mathcal H=\overline{\Poly}=\overline{V+F}=\overline V+F.
    \]
    La aplicación lineal
    \[
        \pi_F:F\to \mathcal H/\overline V,
        \qquad
        \pi_F(f)=f+\overline V,
    \]
    es sobreyectiva, pues dado $x+\overline V\in \mathcal H/\overline V$, como
    \[
        \mathcal H=\overline V+F,
    \]
    existen $v\in \overline V$ y $f\in F$ tales que $x=v+f$. Por tanto,
    \[
        x+\overline V=f+\overline V=\pi_F(f).
    \]
    Luego
    \[
        \dim(\mathcal H/\overline V)\le \dim(F)=m.
    \]

    Para $(2)$, como $M$ es cerrado y de codimensión $m$, existen $\varphi_1,\dots,\varphi_m\in\mathcal H^*$ linealmente independientes tales que
    \[
        M=\bigcap_{j=1}^m \ker \varphi_j,
    \]
    y, restringiendo a $\Poly$,
    \[
        M\cap\Poly=\bigcap_{j=1}^m \ker L_j,
    \]
    donde $L_j:=\varphi_j|_{\Poly}$ para $j=1,\dots,m$. Si una combinación lineal de los $L_j$ se anula en $\Poly$, también se anula en $\mathcal H$ por densidad, pues los $\varphi_j$ son continuos. Luego $L_1,\dots,L_m$ son linealmente independientes. Por la Proposición~\ref{prop:finite-codim-kernels},
    \[
        \operatorname{codim}_{\Poly}(M\cap\Poly)=m.
    \]
    Fijado $x\in M$, tomamos una sucesión $p_n\in\Poly$ tal que $p_n\to x$ y definimos
    \[
        a_n:=(\varphi_1(p_n),\dots,\varphi_m(p_n))\in\mathbb C^m.
    \]
    Como $x\in M$, se tiene $a_n\to 0$. Consideremos la aplicación lineal
    \[
        S:\Poly\to\mathbb C^m,
        \qquad
        S(p)=(L_1(p),\dots,L_m(p)).
    \]
    Como $L_1,\dots,L_m$ son linealmente independientes, $S$ es sobreyectiva. Elegimos $u_1,\dots,u_m\in\Poly$ tales que $S(u_i)=e_i$, donde $e_i$ es la base canónica de $\mathbb C^m$, y definimos
    \[
        R:\mathbb C^m\to\Poly,
        \qquad
        R(b_1,\dots,b_m):=\sum_{i=1}^m b_i u_i.
    \]
    Entonces $S\circ R=\mathrm{Id}_{\mathbb C^m}$. Para cada $n$, ponemos
    \[
        v_n:=p_n-R(a_n)\in\Poly.
    \]
    Como $S(v_n)=0$, se tiene $v_n\in M\cap\Poly$. Además,
    \[
        \|v_n-x\|\le \|p_n-x\|+\|R(a_n)\|\xrightarrow[n\to\infty]{}0,
    \]
    porque $p_n\to x$, $a_n\to0$ y $R$ es continua sobre el espacio finito-dimensional $\mathbb C^m$. Luego $x\in\overline{M\cap\Poly}$, y por tanto $\overline{M\cap\Poly}=M$.
\end{proof}

\begin{rmk}
    La desigualdad de la parte $(1)$ puede ser estricta. No hay aquí dos nociones incompatibles de codimensión, sino dos formas de medir un mismo subespacio en ambientes distintos. Si $V\subset\Poly$, la cantidad $\operatorname{codim}_{\Poly}(V)$ cuenta las restricciones lineales antes de clausurar; en cambio, $\operatorname{codim}_{\mathcal H}(\overline V)$ mide cuántas de esas restricciones sobreviven al pasar a la compleción hilbertiana. La Proposición~\ref{thm:dicotomia_1} muestra precisamente que una condición lineal puede colapsar por completo: un subespacio de codimensión uno en $\Poly$ puede tener clausura total en $\mathcal H$ y, por tanto, codimensión topológica cero.
\end{rmk}

\section{Índice \texorpdfstring{$Q_k$}{Qk}}

Recordemos que, dado $T\in\mathcal L(\mathcal H)$ y un subespacio vectorial $M\subset\mathcal H$, la restricción $T|_M$ satisface
\[
    \|T|_M\|=\sup_{\substack{p\in M\\ p\ne0}}\frac{\|Tp\|}{\|p\|}\le \|T\|<+\infty.
\]

\begin{defn}\label{def:numeros_gelfand}
    Sea $T\in\mathcal L(\mathcal H)$. Para cada $k\ge 0$, definimos el número de Gelfand de orden $k$ de $T$ como
    \begin{equation}\label{eq:gelfand_numbers}
        c_k(T)=\inf\{\|T|_M\|:M\subset \mathcal H,\ M\text{ cerrado},\ \operatorname{codim}_{\mathcal H}(M)<k+1\}.
    \end{equation}
    En particular, $c_0(T)=\|T\|$. \cite{Ullrich2024}
\end{defn}
\begin{rmk}
    Nuestra definición del número de Gelfand $c_0(T)$ corresponde al número $c_1(T)$, y en general $c_{k-1}(T)$ corresponde a $c_k(T)$. La usamos así por coherencia de notación al trabajar en espacios de polinomios.
\end{rmk}

El índice de Gelfand se define para operadores acotados, mientras que en este trabajo aparecerán operadores no acotados. La discusión anterior muestra que, en ese régimen, medir solo la codimensión de la clausura en $\mathcal H$ no siempre retiene las restricciones lineales relevantes. Esto justifica introducir un nuevo índice que extienda la noción de los números de Gelfand a operadores no acotados sobre $\Poly$.
\begin{defn}\label{def:indice_Qk}
    Sea $T:\Poly\to\Poly$ un operador lineal no necesariamente acotado definido sobre $(\Poly,\|\cdot\|)$, donde la norma está inducida por un producto interno. Para cada $k\ge0$, definimos
    \[
        Q_k(T)
        :=\inf\bigl\{\|T|_V\|:V\subset\Poly\ \ \operatorname{codim}_{\Poly}(V)<k+1\bigr\},
    \]
    donde $Q_k(T)\in [0,+\infty]$.
\end{defn}

\begin{rmk}
    En particular, para $k=0$ la condición
    \(
    \operatorname{codim}_{\Poly}(V)<1
    \)
    fuerza $V=\Poly$, y por tanto
    \[
        Q_0(T)=\|T\|,
    \]
    admitiendo el valor $+\infty$ cuando $T$ no es acotado.
\end{rmk}

\begin{prop}\label{prop:propiedades_Qk}
    Sea $T:\Poly\to\Poly$ un operador lineal. Entonces
    \[
        \|T\|=Q_0(T)\ge Q_1(T)\ge \cdots \ge Q_k(T)\ge Q_{k+1}(T)\ge \cdots \ge 0,
    \]
    admitiendo el valor $+\infty$ cuando corresponda.
\end{prop}

\begin{proof}
    La identidad $Q_0(T)=\|T\|$ es inmediata de la definición.

    Para cada $k\ge0$, sea
    \[
        \mathcal{F}_k = \{ V \subset \Poly : \operatorname{codim}_{\Poly}(V) < k+1 \}.
    \]
    Entonces $\mathcal F_k\subset\mathcal F_{k+1}$, y por tanto
    \[
        Q_{k+1}(T) \le Q_k(T).
    \]
    La no negatividad es trivial.
\end{proof}

\begin{defn}
    Sea $T:\Poly\to\Poly$ un operador lineal no necesariamente acotado. Para $k\ge1$, diremos que la cadena 
    \[
        Q_0(T)\ge Q_1(T)\ge \cdots \ge Q_k(T)\ge Q_{k+1}(T)\ge \cdots \ge 0,
    \]
    estabiliza en $k$ si $Q_{k-1}(T)=\infty$ y $Q_k(T)<\infty$.
\end{defn}

A continuación fijamos un modelo elemental en el que la cadena se rompe por primera vez en $k=1$.

\begin{exm}
    Consideremos el producto de Sobolev
    \[
        \langle p,q\rangle_S=\int_{\mathbb S_1}p(z)\overline{q(z)}\,d\mathbf m(z)+p'(2)\overline{q'(2)},
        \qquad p,q\in\Poly,
    \]
    y el operador de multiplicación $\D p(z)=zp(z)$. Entonces
    \[
        \|\D p\|_S^2=\int_{\mathbb S_1}|zp(z)|^2\,d\mathbf m(z)+|(\D p)'(2)|^2=\|p\|_{L^2(\mathbf m)}^2+|p(2)+2p'(2)|^2.
    \]
    Para ver que $\D$ no es acotado, consideremos
    \[
        p_n(z) = \frac{n^2}{2^n} \left( z^n - \frac{n}{2(n+1)} z^{n+1} \right).
    \]
    Entonces
    \[
        p_n'(z) = \frac{n^2}{2^n} \left( n z^{n-1} - \frac{n}{2} z^n \right) \implies p_n'(2) = \frac{n^2}{2^n} \left( n 2^{n-1} - n 2^{n-1} \right) = 0.
    \]
    Como los monomios son ortonormales en $L^2(\mathbf m)$,
    \[
        \|p_n\|_{L^2(\mathbf m)}^2 = \frac{n^4}{4^n} \left( 1 + \frac{n^2}{4(n+1)^2} \right) \xrightarrow[n\to\infty]{} 0.
    \]
    En cambio,
    \[
        p_n(2) = \frac{n^2}{2^n} \left( 2^n - \frac{n}{2(n+1)} 2^{n+1} \right) = n^2 \left( 1 - \frac{n}{n+1} \right) = \frac{n^2}{n+1} \xrightarrow[n\to\infty]{} \infty.
    \]
    Por tanto,
    \[
        \|p_n\|_S^2 = \|p_n\|_{L^2(\mathbf m)}^2 + |p_n'(2)|^2 = \|p_n\|_{L^2(\mathbf m)}^2 \to 0,
    \]
    mientras que
    \[
        \|\D p_n\|_S^2 = \|p_n\|_{L^2(\mathbf m)}^2 + |p_n(2)|^2 \to \infty.
    \]
    Luego $\D$ no es acotado, es decir, $Q_0(\D)=\infty$.
    
    Sin embargo, la no acotación proviene del término $p(2)$. Si
    \[
        V = \{p \in \Poly : p(2) = 0\},
    \]
    entonces, para todo $p \in V$,
    \[
        \|\D p\|_S^2 = \|p\|_{L^2(\mathbf m)}^2 + |2p'(2)|^2 \le 4\left(\|p\|_{L^2(\mathbf m)}^2 + |p'(2)|^2\right) = 4\|p\|_S^2.
    \]
    Por tanto, $\D|_V$ es acotado y
    \[
        Q_1(\D) \le 2 < \infty.
    \]
\end{exm}

De este ejemplo conviene destacar que existen operadores lineales no acotados sobre $\Poly$ cuya restricción a subespacios ``grandes'' (en el sentido de que tienen codimensión 1) es acotada. Esto justifica estudiar la estabilización de la cadena de los $Q_k$ para encontrar subespacios de codimensión finita donde esos operadores sean acotados.

\section{Equivalencia con los números de Gelfand}

A continuación, probaremos que cuando consideramos operadores lineales acotados sobre $\Poly$, que admiten por tanto una extensión al espacio de Hilbert $\mathcal H$, entonces los índices $Q_k$ coinciden con los números de Gelfand.

\begin{thm}\label{thm:equivalencia_Qk_Gelfand}
    Sea $(\mathcal H,\langle\cdot,\cdot\rangle)$ un espacio de Hilbert tal que $\mathcal H=\overline{\Poly}$, y sea $T\in\mathcal L(\mathcal H)$ con $T(\Poly)\subset\Poly$. Entonces, para todo $k\ge0$,
    \begin{equation}\label{eq:igualdad_Qk_Gelfand}
        Q_k(T)=c_k(T),
    \end{equation}
    donde $Q_k(T)$ se calcula sobre $\Poly$ con la norma inducida por $\mathcal H$.
\end{thm}

\begin{proof}
    Sea $V\subset\Poly$ tal que
    \[
        \operatorname{codim}_{\Poly}(V)=m<k+1.
    \]
    Poniendo $M:=\overline V$, la Proposición~\ref{prop:relacion_codimensiones} da
    \[
        \operatorname{codim}_{\mathcal H}(M)\le m<k+1.
    \]
    Por tanto, $M$ es admisible en la definición de $c_k(T)$. Como $T$ es continuo y $V$ es denso en $M$,
    \[
        \|T|_M\|=\|T|_V\|.
    \]
    Luego
    \[
        c_k(T)\le \|T|_M\|=\|T|_V\|.
    \]
    Tomando ínfimo sobre esos $V$,
    \[
        c_k(T)\le Q_k(T).
    \]

    Sea ahora $M\subset\mathcal H$ cerrado con
    \[
        \operatorname{codim}_{\mathcal H}(M)=m<k+1.
    \]
    Entonces, por la Proposición~\ref{prop:relacion_codimensiones}, el subespacio
    \[
        V:=M\cap\Poly
    \]
    satisface
    \[
        \operatorname{codim}_{\Poly}(V)=m,
        \qquad
        \overline V=M,
    \]
    y
    \[
        \|T|_V\|=\|T|_M\|.
    \]
    Por tanto,
    \[
        Q_k(T)\le \|T|_V\|=\|T|_M\|.
    \]
    Tomando ínfimo sobre los subespacios $M$ admisibles en la definición de $c_k(T)$,
    \[
        Q_k(T)\le c_k(T).
    \]

    Las dos desigualdades prueban \eqref{eq:igualdad_Qk_Gelfand}.
\end{proof}

\begin{rmk}
    Esto justifica que el índice $Q_k$ es una extensión del índice de Gelfand para operadores no acotados, ya que coinciden cuando estos son acotados.
\end{rmk}
