\chapter{Codimensión, Dicotomía e índice \texorpdfstring{$Q_k$}{Qk}}\label{sec:codimension_qk}
Partimos de un producto interno $\langle\cdot, \cdot\rangle$. Denotamos por $\mathcal H$ la compleción de $\Poly$ respecto de la norma inducida por ese producto. En particular,
\[
    \mathcal H=\overline{\Poly},
\]
y $\Poly$ es un subespacio denso de $\mathcal H$. Siguiendo la Definición~\ref{def:codim_general}, cuando $M\subset\mathcal H$ es un subespacio cerrado escribiremos
\[
    \operatorname{codim}_{\mathcal H}(M)=\dim(\mathcal H/M)=\dim(M^\perp),
\]
mientras que para un subespacio vectorial $V\subset\Poly$ escribiremos
\[
    \operatorname{codim}_{\Poly}(V)=\dim(\Poly/V),
\]
entendiendo aquí la dimensión en sentido algebraico.

En espacios de Hilbert, los valores singulares están ligados a la estructura hilbertiana del espacio. Sin embargo, en contextos más generales no existe una única noción canónica que los sustituya. La teoría de los \(s\)-números, desarrollada por Pietsch \cite{Ullrich2024, Pietsch1987}, proporciona un marco para extender esta filosofía a operadores acotados entre espacios de Banach. En particular, nos apoyaremos en los números de Gelfand, que se definen a continuación.

\begin{defn}\label{def:numeros_gelfand}
    Sea $T\in\mathcal L(\mathcal H)$. Para cada $k\ge 0$, definimos el número de Gelfand de orden $k$ de $T$ como
    \begin{equation}\label{eq:gelfand_numbers}
        c_k(T)=\inf\{\|T|_M\|:M\subset \mathcal H,\ M\text{ cerrado},\ \operatorname{codim}_{\mathcal H}(M)<k+1\}.
    \end{equation}
    En particular, $c_0(T)=\|T\|$. \cite{Ullrich2024}
\end{defn}

\begin{rmk}
    Nuestra definición del número de Gelfand $c_0(T)$ corresponde al número $c_1(T)$, y en general $c_{k-1}(T)$ corresponde a $c_k(T)$. La usamos así por coherencia de notación al trabajar en espacios de polinomios.
\end{rmk}

\section{Restricciones lineales y codimensión finita}

Comenzamos fijando el lenguaje algebraico que usaremos para describir subespacios de codimensión finita. En un espacio vectorial, imponer una condición lineal equivale a tomar el núcleo de un funcional lineal; imponer varias condiciones lineales equivale a considerar una intersección de núcleos. En particular, los subespacios de codimensión finita pueden entenderse como aquellos determinados por un número finito de restricciones lineales independientes.

\begin{prop}\label{prop:finite-codim-kernels}
    Para un subespacio vectorial $V\subset \Poly$, son equivalentes:
    \begin{enumerate}
        \item $\operatorname{codim}_{\Poly}(V)<\infty$;
        \item existen $m\in\mathbb N\cup\{0\}$ y $L_1,\dots,L_m\in\Poly^*$ tales que
              \[
                    V=\bigcap_{j=1}^m \ker L_j.
                \]
              donde, si $m=0$, entendemos por convención que la intersección vacía es todo $\Poly$.
    \end{enumerate}
    Además,
    \[
        \operatorname{codim}_{\Poly}(V)
        =
        \min\Bigl\{m\in\mathbb N\cup\{0\}: \exists\,L_1,\dots,L_m\in\Poly^*\ \text{tales que}\ V=\bigcap_{j=1}^m\ker L_j\Bigr\},
    \]
    y más precisamente, si $V=\bigcap_{j=1}^m\ker L_j$, entonces
    \[
        \operatorname{codim}_{\Poly}(V)=\dim[L_1,\dots,L_m].
    \]
\end{prop}

\begin{proof}
    Supongamos que $\operatorname{codim}_{\Poly}(V)=n<\infty$. Entonces existe un subespacio complementario $F\subset\Poly$ tal que
    \[
        \Poly=V\oplus F,
        \qquad
        \dim(F)=n.
    \]
    Tomemos una base $u_1,\dots,u_n$ de $F$ y sea $u_1^*,\dots,u_n^*$ la base dual de $F^*$. Para cada $p\in\Poly$, escribiendo de manera única $p=v+f$ con $v\in V$ y $f\in F$, definimos
    \[
        L_j(p):=u_j^*(f),\qquad j=1,\dots,n.
    \]
    Entonces $L_j\in\Poly^*$ y
    \[
        V=\bigcap_{j=1}^n\ker L_j,
    \]
    pues $L_j(p)=0$ para todo $j$ si y solo si la componente $f$ es nula.

    Supongamos ahora que
    \[
        V=\bigcap_{j=1}^m\ker L_j.
    \]
    Sea
    \[
        r:=\dim[L_1,\dots,L_m].
    \]
    Reordenando si hace falta, podemos suponer que $L_1,\dots,L_r$ forman una base de ese subespacio. Entonces
    \[
        V=\bigcap_{i=1}^r\ker L_i.
    \]
    Consideremos la aplicación lineal
    \[
        \Psi:\Poly\to\mathbb C^r,
        \qquad
        \Psi(p)=(L_1(p),\dots,L_r(p)).
    \]
    Como $L_1,\dots,L_r$ son linealmente independientes, $\Psi$ es sobreyectiva. Además,
    \[
        \ker\Psi=V.
    \]
    Por el primer teorema de isomorfía,
    \[
        \Poly/V\cong\mathbb C^r,
    \]
    y por tanto
    \[
        \operatorname{codim}_{\Poly}(V)=r=
        \dim[L_1,\dots,L_m].
    \]
    La fórmula del mínimo se deduce de inmediato de esta identidad.
\end{proof}

A continuación damos una caracterización de la codimensión algebraica atendiendo a la ortogonalidad dada por el producto escalar.

\begin{prop}\label{prop:equiv_ortho_interseccion}
    Sea $\langle\cdot,\cdot\rangle$ un producto interno en $\Poly$, sean $f_1,\dots,f_m\in\Poly$ y sea $F=[f_1,\dots,f_m]$. Para $j=1,\dots,m$, definimos
    \[
        L_j(p):=\langle p,f_j\rangle,
        \qquad p\in\Poly.
    \]
    Entonces
    \[
        F^{\perp}_{\Poly}=\{p\in\Poly:\langle p,f_j\rangle=0,\ j=1,\dots,m\}=\bigcap_{j=1}^m\ker(L_j).
    \]
    Además,
    \[
        \operatorname{codim}_{\Poly}(F^{\perp}_{\Poly})
        =
        \dim F.
    \]
    En particular, si $f_1,\dots,f_m$ son linealmente independientes, entonces
    \[
        \operatorname{codim}_{\Poly}(F^{\perp}_{\Poly})=m.
    \]
\end{prop}

\begin{proof}
    Se tiene
    \[
        F^{\perp}_{\Poly}=\bigcap_{j=1}^m\ker(L_j)
    \]
    por definición.

    Sea ahora
    \[
        r:=\dim F.
    \]
    Reordenando si hace falta, podemos suponer que $f_1,\dots,f_r$ forman una base de ese subespacio. Entonces, para cada $j>r$, el vector $f_j$ es combinación lineal de $f_1,\dots,f_r$, y por sesquilinealidad del producto interno se sigue que $L_j$ es combinación lineal de $L_1,\dots,L_r$ y
    \[
        \bigcap_{i=1}^r \operatorname{ker}(L_i) \subset \operatorname{ker}(L_j).
    \]
    Por tanto,
    \[
        F^{\perp}_{\Poly}=\bigcap_{i=1}^r\ker(L_i).
    \]

    Si $\sum_{i=1}^r\alpha_iL_i=0$, tomando
    \(
    q=\sum_{i=1}^r\overline{\alpha_i}f_i
    \)
    se obtiene
    \[
        0=\sum_{i=1}^r\alpha_iL_i(q)=\langle q,q\rangle.
    \]
    Luego $q=0$, y como $f_1,\dots,f_r$ son linealmente independientes, $\alpha_1=\cdots=\alpha_r=0$. Por tanto, $L_1,\dots,L_r$ son linealmente independientes.

    Aplicando la Proposición~\ref{prop:finite-codim-kernels} a $L_1,\dots,L_r$, concluimos que
    \[
        \operatorname{codim}_{\Poly}(F^{\perp}_{\Poly})=r.
    \]
\end{proof}

La misma idea se extiende a intersecciones finitas de núcleos de funcionales acotados.

\begin{prop}\label{prop:finite-codim-hilbert-kernels}
    Sea $\mathcal H$ un espacio de Hilbert y sean $\varphi_1,\dots,\varphi_N$ funcionales lineales acotados y linealmente independientes. Si
    \[
        M:=\bigcap_{i=1}^N \ker(\varphi_i),
    \]
    entonces $M$ es cerrado y
    \[
        \operatorname{codim}_{\mathcal H}(M)=N.
    \]
\end{prop}

\begin{proof}
    Como cada $\ker(\varphi_i)$ es cerrado, la intersección finita $M$ también es cerrada.

    Por el teorema de Riesz, existen $f_1,\dots,f_N\in\mathcal H$ linealmente independientes tales que
    \[
        \varphi_i(x)=\langle x,f_i\rangle_{\mathcal H},
        \qquad x\in\mathcal H,
    \]
    para $i=1,\dots,N$. Sea
    \[
        F=[f_1, \dots, f_N].
    \]
    Entonces
    \[
        x\in M
        \iff \varphi_i(x)=0\ \text{para todo }i
        \iff \langle x,f_i\rangle_{\mathcal H}=0\ \text{para todo }i
        \iff x\perp F,
    \]
    de modo que $M=F^{\perp}$. Como los vectores $f_1,\dots,f_N$ son linealmente independientes, se tiene $\dim(F)=N$, y por la descomposición ortogonal $\mathcal H=F^{\perp}\oplus F$ concluimos que
    \[
        \operatorname{codim}_{\mathcal{H}}(M)=N.
    \]
\end{proof}

\begin{prop}\label{prop:closed-finite-codim-hilbert}
    Sea $M\subset\mathcal H$ un subespacio cerrado. Son equivalentes:
    \begin{enumerate}
        \item $\operatorname{codim}_{\mathcal H}(M)=m<\infty$;
        \item existen funcionales lineales acotados y linealmente independientes $\varphi_1,\dots,\varphi_m\in\mathcal H^*$ tales que
              \[
                  M=\bigcap_{j=1}^m \ker \varphi_j.
              \]
    \end{enumerate}
    Además, si $M=\bigcap_{j=1}^r\ker\varphi_j$ para ciertos funcionales lineales acotados $\varphi_1,\dots,\varphi_r$, entonces
    \[
        \operatorname{codim}_{\mathcal H}(M)=\dim[\varphi_1,\dots,\varphi_r].
    \]
\end{prop}

\begin{proof}
    Supongamos que $\operatorname{codim}_{\mathcal H}(M)=m<\infty$. Entonces $M^\perp$ es finito-dimensional de dimensión $m$. Tomamos una base ortonormal $f_1,\dots,f_m$ de $M^\perp$ y definimos
    \[
        \varphi_j(x):=\langle x,f_j\rangle_{\mathcal H},
        \qquad x\in\mathcal H,
    \]
    para $j=1,\dots,m$. Entonces cada $\varphi_j$ es lineal y acotado, y además
    \[
        M=(M^\perp)^\perp=\bigcap_{j=1}^m \ker\varphi_j.
    \]
    Si $\sum_{j=1}^m\alpha_j\varphi_j=0$, evaluando en $f_i$ obtenemos $\alpha_i=0$ para todo $i$, de modo que $\varphi_1,\dots,\varphi_m$ son linealmente independientes.

    Recíprocamente, si $M=\bigcap_{j=1}^m\ker\varphi_j$ con $\varphi_1,\dots,\varphi_m$ linealmente independientes, la Proposición~\ref{prop:finite-codim-hilbert-kernels} da
    \[
        \operatorname{codim}_{\mathcal H}(M)=m.
    \]

    Para la afirmación final, sea
    \[
        s:=\dim[\varphi_1,\dots,\varphi_r].
    \]
    Reordenando si hace falta, podemos suponer que $\varphi_1,\dots,\varphi_s$ forman una base de ese subespacio. Entonces
    \[
        M=\bigcap_{j=1}^s\ker\varphi_j,
    \]
    y la Proposición~\ref{prop:finite-codim-hilbert-kernels} implica
    \[
        \operatorname{codim}_{\mathcal H}(M)=s.
    \]
\end{proof}

\section{Dicotomía de la codimensión}

La sección anterior describe los subespacios de codimensión finita mediante restricciones lineales. Sin embargo, en nuestro problema esas restricciones viven inicialmente en \(\Poly\), mientras que la teoría clásica de los números de Gelfand se formula en la compleción hilbertiana \(\mathcal H\). Por tanto, es necesario entender qué ocurre con una condición lineal impuesta en \(\Poly\) cuando se toma la clausura del subespacio que define. 

\begin{prop}\label{thm:dicotomia_1}
    Sea $(\mathcal H,\langle\cdot,\cdot\rangle)$ un espacio de Hilbert tal que $\mathcal H=\overline{\Poly}$, y sea $L:\Poly\to\mathbb C$ un funcional lineal no nulo. Entonces ocurre exactamente una de las dos posibilidades siguientes:
    \begin{enumerate}
        \item si $L$ no es acotado respecto de la norma de $\mathcal H$, entonces
              \[
                  \overline{\ker L}=\mathcal H,
                  \qquad
                  \operatorname{codim}_{\mathcal H}(\overline{\ker L})=0;
              \]
        \item si $L$ es acotado respecto de la norma de $\mathcal H$, entonces
              \[
                  \operatorname{codim}_{\mathcal H}(\overline{\ker L})=1.
              \]
    \end{enumerate}
\end{prop}

\begin{proof}
    Supongamos que $L$ no es acotado. Fijado $p\in\Poly$, existe una sucesión $w_n\in\Poly$ tal que
    \[
        |L(w_n)|>n\|w_n\|.
    \]
    Definimos
    \[
        u_n:=\frac{w_n}{L(w_n)}.
    \]
    Entonces
    \[
        L(u_n)=1,
        \qquad
        \|u_n\|=\frac{\|w_n\|}{|L(w_n)|}<\frac1n,
    \]
    de modo que $\|u_n\|\to0$. Sea
    \[
        v_n:=p-L(p)u_n\in\ker L,
        \qquad
        \|v_n-p\|=|L(p)|\,\|u_n\|\to 0.
    \]
    Como $\Poly$ es denso en $\mathcal H$, se sigue que $\overline{\ker L}=\mathcal H$.

    Supongamos que $L$ es acotado. Entonces se extiende de manera única a un funcional lineal acotado $\widetilde L$ sobre $\mathcal H$. Como $\ker\widetilde L$ es cerrado y contiene a $\ker L$,
    \[
        \overline{\ker L}^{\,\mathcal H}\subset \ker\widetilde L.
    \]
    Recíprocamente, sea $x\in\ker\widetilde L$. Tomamos una sucesión $p_n\in\Poly$ tal que $p_n\to x$. Fijado $q\in\Poly$ con $L(q)\neq 0$, definimos
    \[
        v_n:=p_n-\frac{\widetilde L(p_n)}{L(q)}q.
    \]
    Entonces $v_n\in\ker L$ y, como $\widetilde L(x)=0$, se tiene
    \[
        v_n-p_n=-\frac{\widetilde L(p_n)}{L(q)}q\xrightarrow[n\to\infty]{}0,
    \]
    de modo que $v_n\to x$. Por tanto,
    \[
        \ker\widetilde L\subset \overline{\ker L}^{\,\mathcal H},
    \]
    y concluimos que
    \[
        \overline{\ker L}^{\,\mathcal H}=\ker\widetilde L.
    \]
    Para todo $x\in\mathcal H$,
    \[
        x=\left(x-\frac{\widetilde L(x)}{L(q)}q\right)+\frac{\widetilde L(x)}{L(q)}q,
    \]
    y el primer sumando pertenece a $\ker\widetilde L$. Por tanto,
    \[
        \mathcal H=\ker\widetilde L+[q].
    \]
    Como $q\notin\ker\widetilde L$, esa suma es directa. En consecuencia,
    \[
        \operatorname{codim}_{\mathcal H}(\overline{\ker L})=\operatorname{codim}_{\mathcal H}(\ker\widetilde L)=1.
    \]
\end{proof}

\begin{prop}\label{prop:relacion_codimensiones}
    Sea $\mathcal H=\overline{\Poly}$ con respecto a un producto interno.
    \begin{enumerate}
        \item Si $V\subset\Poly$ satisface $\operatorname{codim}_{\Poly}(V)=m<\infty$, entonces
              \[
                  \operatorname{codim}_{\mathcal H}(\overline V)\le m.
              \]
        \item Si $M\subset\mathcal H$ es cerrado y $\operatorname{codim}_{\mathcal H}(M)=m<\infty$, entonces
              \[
                  \overline{M\cap\Poly}=M
                  \qquad\text{y}\qquad
                  \operatorname{codim}_{\Poly}(M\cap\Poly)=m.
              \]
    \end{enumerate}
\end{prop}

\begin{proof}
    Para $(1)$, tomemos un subespacio complementario $F\subset\Poly$ con
    \[
        \Poly=V\oplus F,
        \qquad
        \dim(F)=m.
    \]
    Como $F$ es finito-dimensional, es cerrado en $\mathcal H$. Además, $\overline V+F$ es cerrado: si $x_n=v_n+f_n\in \overline V+F$ y $x_n\to x$, entonces $(x_n)$ es acotada; como $F$ es finito-dimensional, de $(f_n)$ extraemos una subsucesión $(f_{n_k})$ convergente a cierto $f\in F$, y por tanto
    \[
        v_{n_k}=x_{n_k}-f_{n_k}\to x-f.
    \]
    Como $\overline V$ es cerrado, $x-f\in\overline V$, luego $x\in\overline V+F$. Por otra parte,
    \[
        V+F\subset \overline V+F,
    \]
    así que
    \[
        \overline{V+F}\subset \overline V+F.
    \]
    Recíprocamente, si $x=v+f\in \overline V+F$ con $v\in\overline V$ y $f\in F$, tomando $v_n\in V$ con $v_n\to v$ se tiene
    \[
        v_n+f\in V+F,
        \qquad
        v_n+f\to x,
    \]
    y por tanto $x\in\overline{V+F}$. Luego
    \[
        \overline{V+F}=\overline V+F.
    \]
    Como $\Poly=V\oplus F$, se sigue que
    \[
        \mathcal H=\overline{\Poly}=\overline{V+F}=\overline V+F.
    \]
    La aplicación lineal
    \[
        \pi_F:F\to \mathcal H/\overline V,
        \qquad
        \pi_F(f)=f+\overline V,
    \]
    es sobreyectiva, pues dado $x+\overline V\in \mathcal H/\overline V$, como
    \[
        \mathcal H=\overline V+F,
    \]
    existen $v\in \overline V$ y $f\in F$ tales que $x=v+f$. Por tanto,
    \[
        x+\overline V=f+\overline V=\pi_F(f).
    \]
    Luego
    \[
        \dim(\mathcal H/\overline V)\le \dim(F)=m.
    \]

    Para $(2)$, como $M$ es cerrado y de codimensión $m$, la Proposición~\ref{prop:closed-finite-codim-hilbert} garantiza que existen $\varphi_1,\dots,\varphi_m\in\mathcal H^*$ linealmente independientes tales que
    \[
        M=\bigcap_{j=1}^m \ker \varphi_j,
    \]
    y, restringiendo a $\Poly$,
    \[
        M\cap\Poly=\bigcap_{j=1}^m \ker L_j,
    \]
    donde $L_j:=\varphi_j|_{\Poly}$ para $j=1,\dots,m$. Si una combinación lineal de los $L_j$ se anula en $\Poly$, también se anula en $\mathcal H$ por densidad, pues los $\varphi_j$ son continuos. Luego $L_1,\dots,L_m$ son linealmente independientes. Por la Proposición~\ref{prop:finite-codim-kernels},
    \[
        \operatorname{codim}_{\Poly}(M\cap\Poly)=m.
    \]
    Fijado $x\in M$, tomamos una sucesión $p_n\in\Poly$ tal que $p_n\to x$ y definimos
    \[
        a_n:=(\varphi_1(p_n),\dots,\varphi_m(p_n))\in\mathbb C^m.
    \]
    Como $x\in M$, se tiene $a_n\to 0$. Consideremos la aplicación lineal
    \[
        S:\Poly\to\mathbb C^m,
        \qquad
        S(p)=(L_1(p),\dots,L_m(p)).
    \]
    Como $L_1,\dots,L_m$ son linealmente independientes, $S$ es sobreyectiva. Elegimos $u_1,\dots,u_m\in\Poly$ tales que $S(u_i)=e_i$, donde $e_i$ es la base canónica de $\mathbb C^m$, y definimos
    \[
        R:\mathbb C^m\to\Poly,
        \qquad
        R(b_1,\dots,b_m):=\sum_{i=1}^m b_i u_i.
    \]
    Entonces $S\circ R=\mathrm{Id}_{\mathbb C^m}$. Para cada $n$, ponemos
    \[
        v_n:=p_n-R(a_n)\in\Poly.
    \]
    Como $S(v_n)=0$, se tiene $v_n\in M\cap\Poly$. Además,
    \[
        \|v_n-x\|\le \|p_n-x\|+\|R(a_n)\|\xrightarrow[n\to\infty]{}0,
    \]
    porque $p_n\to x$, $a_n\to0$ y $R$ es continua sobre el espacio finito-dimensional $\mathbb C^m$. Luego $x\in\overline{M\cap\Poly}$, y por tanto $\overline{M\cap\Poly}=M$.
\end{proof}

\begin{rmk}
    La desigualdad de la parte $(1)$ puede ser estricta. No hay aquí dos nociones incompatibles de codimensión, sino dos formas de medir un mismo subespacio en ambientes distintos. Si $V\subset\Poly$, la cantidad $\operatorname{codim}_{\Poly}(V)$ cuenta las restricciones lineales antes de clausurar; en cambio, $\operatorname{codim}_{\mathcal H}(\overline V)$ mide cuántas de esas restricciones sobreviven al pasar a la compleción hilbertiana. La Proposición~\ref{thm:dicotomia_1} muestra precisamente que una condición lineal puede colapsar por completo: un subespacio de codimensión uno en $\Poly$ puede tener clausura total en $\mathcal H$ y, por tanto, codimensión topológica cero.
\end{rmk}

\section{Índice \texorpdfstring{$Q_k$}{Qk}}

Una vez vista la dicotomía, podemos introducir el índice que será relevante para operadores lineales no necesariamente acotados sobre $\Poly$. Para un operador lineal $T:\Poly\to\Poly$ y un subespacio $V\subset\Poly$, escribiremos
\[
    \|T|_V\|:=\sup_{\substack{p\in V\\ p\ne0}}\frac{\|Tp\|}{\|p\|}\in[0,+\infty].
\]

\begin{defn}\label{def:indice_Qk}
    Sea $T:\Poly\to\Poly$ un operador lineal no necesariamente acotado definido sobre $(\Poly,\|\cdot\|)$, donde la norma está inducida por un producto interno. Para cada $k\ge0$, definimos
    \[
        Q_k(T)
        :=\inf\bigl\{\|T|_V\|:V\subset\Poly\ \ \operatorname{codim}_{\Poly}(V)<k+1\bigr\},
    \]
    donde $Q_k(T)\in [0,+\infty]$.
\end{defn}

\begin{rmk}
    En particular, para $k=0$ la condición
    \(
    \operatorname{codim}_{\Poly}(V)<1
    \)
    fuerza $V=\Poly$, y por tanto
    \[
        Q_0(T)=\|T\|,
    \]
    admitiendo el valor $+\infty$ cuando $T$ no es acotado.
\end{rmk}

Veamos una caracterización de $Q_k(T)$ en términos de subespacios de dimensión finita. Poder expresar $Q_k(T)$ como un supremo sobre subespacios de dimensión finita será útil más adelante.

\begin{thm}\label{thm:maxmin_Qk}
    Sea $T:\Poly\to\Poly$ un operador lineal, no necesariamente acotado. Entonces, para todo $k\ge0$,
    \[
        Q_k(T)
        =
        \sup_{\substack{E\subset\Poly\\ \dim E=k+1}}
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|Tp\|}{\|p\|},
    \]
    admitiendo el valor $+\infty$.
\end{thm}

\begin{proof}
    Denotemos
    \[
        \mu_k(T):=
        \sup_{\substack{E\subset\Poly\\ \dim E=k+1}}
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|Tp\|}{\|p\|}.
    \]

    Probemos primero que $\mu_k(T)\le Q_k(T)$. Sea $V\subset\Poly$ con
    \[
        \operatorname{codim}_{\Poly}(V)<k+1,
    \]
    y sea $E\subset\Poly$ con $\dim E=k+1$. Entonces
    \[
        E\cap V\ne\{0\}.
    \]
    En efecto, si $E\cap V=\{0\}$, la aplicación cociente $E\to\Poly/V$ sería inyectiva, lo que implicaría
    \[
        k+1=\dim E\le \dim(\Poly/V)=\operatorname{codim}_{\Poly}(V)<k+1,
    \]
    una contradicción. Tomando $0\ne p\in E\cap V$, obtenemos
    \[
        \inf_{\substack{q\in E\\ q\ne0}}
        \frac{\|Tq\|}{\|q\|}
        \le
        \frac{\|Tp\|}{\|p\|}
        \le
        \sup_{\substack{q\in V\\ q\ne0}}
        \frac{\|Tq\|}{\|q\|}.
    \]
    Tomando supremo en $E$ e ínfimo en $V$, se sigue que
    \[
        \mu_k(T)\le Q_k(T).
    \]

    Probemos ahora la desigualdad recíproca. Si $Q_k(T)=0$, no hay nada que demostrar. Sea
    \[
        0<a<Q_k(T),
    \]
    donde, si $Q_k(T)=+\infty$, tomamos $a>0$ arbitrario. Consideremos la forma hermítica
    \[
        B_a(p,q):=\langle Tp,Tq\rangle-a^2\langle p,q\rangle,
        \qquad p,q\in\Poly.
    \]

    Afirmamos que existe un subespacio $E\subset\Poly$ con $\dim E=k+1$ tal que
    \[
        B_a(p,p)>0,
        \qquad 0\ne p\in E.
    \]
    Supongamos lo contrario. Entonces todo subespacio sobre el que $B_a$ es definida positiva tiene dimensión a lo sumo $k$. Elegimos uno de dimensión máxima, digamos $F$, con
    \[
        m:=\dim F\le k.
    \]
    Definimos
    \[
        V:=\{p\in\Poly: B_a(p,f)=0\ \text{para todo }f\in F\}.
    \]
    Como $V$ está definido por $m$ condiciones lineales, se tiene
    \[
        \operatorname{codim}_{\Poly}(V)\le m\le k<k+1.
    \]
    Además, si existiera $p\in V$ con $B_a(p,p)>0$, entonces $F\oplus[p]$ sería un subespacio de dimensión $m+1$ sobre el cual $B_a$ sería definida positiva, contradiciendo la maximalidad de $F$. Por tanto,
    \[
        B_a(p,p)\le0,
        \qquad p\in V.
    \]
    Es decir,
    \[
        \|Tp\|\le a\|p\|,
        \qquad p\in V.
    \]
    En consecuencia,
    \[
        Q_k(T)\le \|T|_V\|\le a,
    \]
    lo cual contradice $a<Q_k(T)$. La afirmación queda probada.

    Así, para cada $a<Q_k(T)$ existe $E\subset\Poly$ con $\dim E=k+1$ tal que
    \[
        \frac{\|Tp\|}{\|p\|}>a,
        \qquad 0\ne p\in E.
    \]
    Por tanto,
    \[
        \mu_k(T)\ge a.
    \]
    Haciendo $a\uparrow Q_k(T)$, o tomando $a$ arbitrario si $Q_k(T)=+\infty$, concluimos que
    \[
        \mu_k(T)\ge Q_k(T).
    \]

    Junto con la primera desigualdad, esto prueba la igualdad.
\end{proof}

\begin{prop}\label{prop:propiedades_Qk}
    Sea $T:\Poly\to\Poly$ un operador lineal. Entonces
    \[
        \|T\|=Q_0(T)\ge Q_1(T)\ge \cdots \ge Q_k(T)\ge Q_{k+1}(T)\ge \cdots \ge 0,
    \]
    admitiendo el valor $+\infty$ cuando corresponda.
\end{prop}

\begin{proof}
    La identidad $Q_0(T)=\|T\|$ es inmediata de la definición.

    Para cada $k\ge0$, sea
    \[
        \mathcal{F}_k = \{ V \subset \Poly : \operatorname{codim}_{\Poly}(V) < k+1 \}.
    \]
    Entonces $\mathcal F_k\subset\mathcal F_{k+1}$, y por tanto
    \[
        Q_{k+1}(T) \le Q_k(T).
    \]
    La no negatividad es trivial.
\end{proof}

\begin{defn}
    Sea $T:\Poly\to\Poly$ un operador lineal no necesariamente acotado. Para $k\ge1$, diremos que la cadena 
    \[
        Q_0(T)\ge Q_1(T)\ge \cdots \ge Q_k(T)\ge Q_{k+1}(T)\ge \cdots \ge 0,
    \]
    estabiliza en $k$ si $Q_{k-1}(T)=\infty$ y $Q_k(T)<\infty$.
\end{defn}

A continuación fijamos un modelo elemental en el que la cadena se rompe por primera vez en $k=1$.

\begin{exm}
    Consideremos el producto de Sobolev
    \[
        \langle p,q\rangle_S=\int_{\mathbb S_1}p(z)\overline{q(z)}\,d\mathbf m(z)+p'(2)\overline{q'(2)},
        \qquad p,q\in\Poly,
    \]
    y el operador de multiplicación $\D p(z)=zp(z)$. Entonces
    \[
        \|\D p\|_S^2=\int_{\mathbb S_1}|zp(z)|^2\,d\mathbf m(z)+|(\D p)'(2)|^2=\|p\|_{L^2(\mathbf m)}^2+|p(2)+2p'(2)|^2.
    \]
    Para ver que $\D$ no es acotado, consideremos
    \[
        p_n(z) = \frac{n^2}{2^n} \left( z^n - \frac{n}{2(n+1)} z^{n+1} \right).
    \]
    Entonces
    \[
        p_n'(z) = \frac{n^2}{2^n} \left( n z^{n-1} - \frac{n}{2} z^n \right) \implies p_n'(2) = \frac{n^2}{2^n} \left( n 2^{n-1} - n 2^{n-1} \right) = 0.
    \]
    Como los monomios son ortonormales en $L^2(\mathbf m)$,
    \[
        \|p_n\|_{L^2(\mathbf m)}^2 = \frac{n^4}{4^n} \left( 1 + \frac{n^2}{4(n+1)^2} \right) \xrightarrow[n\to\infty]{} 0.
    \]
    En cambio,
    \[
        p_n(2) = \frac{n^2}{2^n} \left( 2^n - \frac{n}{2(n+1)} 2^{n+1} \right) = n^2 \left( 1 - \frac{n}{n+1} \right) = \frac{n^2}{n+1} \xrightarrow[n\to\infty]{} \infty.
    \]
    Por tanto,
    \[
        \|p_n\|_S^2 = \|p_n\|_{L^2(\mathbf m)}^2 + |p_n'(2)|^2 = \|p_n\|_{L^2(\mathbf m)}^2 \to 0,
    \]
    mientras que
    \[
        \|\D p_n\|_S^2 = \|p_n\|_{L^2(\mathbf m)}^2 + |p_n(2)|^2 \to \infty.
    \]
    Luego $\D$ no es acotado, es decir, $Q_0(\D)=\infty$.
    
    Sin embargo, la no acotación proviene del término $p(2)$. Si
    \[
        V = \{p \in \Poly : p(2) = 0\},
    \]
    entonces, para todo $p \in V$,
    \[
        \|\D p\|_S^2 = \|p\|_{L^2(\mathbf m)}^2 + |2p'(2)|^2 \le 4\left(\|p\|_{L^2(\mathbf m)}^2 + |p'(2)|^2\right) = 4\|p\|_S^2.
    \]
    Por tanto, $\D|_V$ es acotado y
    \[
        Q_1(\D) \le 2 < \infty.
    \]
\end{exm}

De este ejemplo conviene destacar que existen operadores lineales no acotados sobre $\Poly$ cuya restricción a subespacios ``grandes'' (en el sentido de que tienen codimensión 1) es acotada. Esto justifica estudiar la estabilización de la cadena de los $Q_k$ para encontrar subespacios de codimensión finita donde esos operadores sean acotados.

\section{Compatibilidad con los índices clásicos}

Por completitud del capítulo, podemos relacionar el índice $Q_k$ con los números de Gelfand de la extensión hilbertiana del operador.

\begin{thm}\label{thm:equivalencia_Qk_Gelfand}
    Sea $(\mathcal H,\langle\cdot,\cdot\rangle)$ un espacio de Hilbert tal que $\mathcal H=\overline{\Poly}$, y sea $T\in\mathcal L(\mathcal H)$ con $T(\Poly)\subset\Poly$. Entonces, para todo $k\ge0$,
    \begin{equation}\label{eq:igualdad_Qk_Gelfand}
        Q_k(T)=c_k(T),
    \end{equation}
    donde el índice $Q_k(T)$ se considera para $T|_{\Poly}$ con la norma inducida por $\mathcal{H}$.
\end{thm}

\begin{proof}
    Sea $V\subset\Poly$ tal que
    \[
        \operatorname{codim}_{\Poly}(V)=m<k+1.
    \]
    Poniendo $M:=\overline V$, la Proposición~\ref{prop:relacion_codimensiones} da
    \[
        \operatorname{codim}_{\mathcal H}(M)\le m<k+1.
    \]
    Por tanto, $M$ es admisible en la definición de $c_k(T)$. Como $T$ es continuo y $V$ es denso en $M$,
    \[
        \|T|_M\|=\|T|_V\|.
    \]
    Luego
    \[
        c_k(T)\le \|T|_M\|=\|T|_V\|.
    \]
    Tomando ínfimo sobre esos $V$,
    \[
        c_k(T)\le Q_k(T).
    \]

    Sea ahora $M\subset\mathcal H$ cerrado con
    \[
        \operatorname{codim}_{\mathcal H}(M)=m<k+1.
    \]
    Entonces, por la Proposición~\ref{prop:relacion_codimensiones}, el subespacio
    \[
        V:=M\cap\Poly
    \]
    satisface
    \[
        \operatorname{codim}_{\Poly}(V)=m,
        \qquad
        \overline V=M,
    \]
    y
    \[
        \|T|_V\|=\|T|_M\|.
    \]
    Por tanto,
    \[
        Q_k(T)\le \|T|_V\|=\|T|_M\|.
    \]
    Tomando ínfimo sobre los subespacios $M$ admisibles en la definición de $c_k(T)$,
    \[
        Q_k(T)\le c_k(T).
    \]

    Las dos desigualdades prueban \eqref{eq:igualdad_Qk_Gelfand}.
\end{proof}

\begin{rmk}
    Esto justifica que el índice $Q_k$ es una extensión del índice de Gelfand para operadores no acotados, ya que coinciden cuando estos son acotados.
\end{rmk}

De manera análoga usando la caracterización del índice $Q_k$, podemos hacer lo mismo para el número de Bernstein de la extensión hilbertiana del operador.

\begin{prop}\label{prop:Qk_Bernstein}
    Sea $T:\Poly\to\Poly$ un operador lineal acotado respecto de $\|\cdot\|$, y sea
    \[
        \widetilde T\in\mathcal L(\mathcal H),
        \qquad
        \mathcal H=\overline{\Poly},
    \]
    su extensión acotada. Entonces, para todo $k\ge0$,
    \[
        Q_k(T)=b_k(\widetilde T).
    \]
\end{prop}

\begin{proof}
    Por la caracterización max--min algebraica del Teorema~\ref{thm:maxmin_Qk},
    \[
        Q_k(T)
        =
        \sup_{\substack{E\subset\Poly\\ \dim E=k+1}}
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|Tp\|}{\|p\|}.
    \]
    Como $\widetilde T|_{\Poly}=T$, esto equivale a
    \[
        Q_k(T)
        =
        \sup_{\substack{E\subset\Poly\\ \dim E=k+1}}
        \inf_{\substack{p\in E\\ p\ne0}}
        \frac{\|\widetilde T p\|}{\|p\|}.
    \]

    Puesto que todo subespacio de $\Poly$ es también un subespacio de $\mathcal H$, se tiene inmediatamente
    \[
        Q_k(T)\le b_k(\widetilde T).
    \]

    Probemos la desigualdad contraria. Sea $F\subset\mathcal H$ un subespacio de dimensión $k+1$, y escribamos
    \[
        \alpha_F:=
        \inf_{\substack{x\in F\\ x\ne0}}
        \frac{\|\widetilde T x\|}{\|x\|}.
    \]
    Fijemos $\varepsilon>0$. Tomemos una base ortonormal
    \[
        f_1,\dots,f_{k+1}
    \]
    de $F$. Como $\Poly$ es denso en $\mathcal H$, podemos escoger
    \[
        p_1,\dots,p_{k+1}\in\Poly
    \]
    tan próximos a $f_1,\dots,f_{k+1}$ como queramos.

    Definimos $A:F\to\mathcal H$ por
    \[
        Af_j=p_j,
        \qquad j=1,\dots,k+1,
    \]
    y extendemos linealmente. Escogiendo las aproximaciones suficientemente buenas, podemos suponer que
    \[
        \|A-I_F\|<\delta
    \]
    con $0<\delta<1$. En particular, $A$ es inyectivo. Por tanto,
    \[
        E:=A(F)
    \]
    es un subespacio de $\Poly$ de dimensión $k+1$.

    Para todo $x\in F$ se tiene
    \[
        \|Ax\|\le (1+\delta)\|x\|,
    \]
    y además
    \[
        \|\widetilde T Ax\|
        \ge
        \|\widetilde T x\|-\|\widetilde T\|\,\|Ax-x\|
        \ge
        \bigl(\alpha_F-\|\widetilde T\|\delta\bigr)\|x\|.
    \]
    Por tanto, para todo $0\ne x\in F$,
    \[
        \frac{\|\widetilde T Ax\|}{\|Ax\|}
        \ge
        \frac{\alpha_F-\|\widetilde T\|\delta}{1+\delta}.
    \]
    Como el término de la derecha tiende a $\alpha_F$ cuando $\delta\to0$, podemos escoger las aproximaciones suficientemente buenas para que
    \[
        \frac{\alpha_F-\|\widetilde T\|\delta}{1+\delta}
        \ge
        \alpha_F-\varepsilon.
    \]
    Así,
    \[
        \inf_{\substack{y\in E\\ y\ne0}}
        \frac{\|\widetilde T y\|}{\|y\|}
        \ge
        \alpha_F-\varepsilon.
    \]
    Como $E\subset\Poly$ y $\dim E=k+1$, se deduce que
    \[
        Q_k(T)\ge \alpha_F-\varepsilon.
    \]
    Puesto que $F$ y $\varepsilon$ eran arbitrarios, obtenemos
    \[
        Q_k(T)\ge
        \sup_{\substack{F\subset\mathcal H\\ \dim F=k+1}}
        \inf_{\substack{x\in F\\ x\ne0}}
        \frac{\|\widetilde T x\|}{\|x\|}
        =
        b_k(\widetilde T).
    \]

    Juntando ambas desigualdades,
    \[
        Q_k(T)=b_k(\widetilde T).
    \]
\end{proof}

\begin{rmk}
    Cuando $T$ es acotado y estamos en un Hilbert, los números de Bernstein y los números de Gelfand coinciden, ya que son los valores singulares del operador. Por tanto, la Proposición~\ref{prop:Qk_Bernstein} y el Teorema~\ref{thm:equivalencia_Qk_Gelfand} afirman que el índice $Q_k$ es una extensión para operadores no acotados densamente definidos.
\end{rmk}
