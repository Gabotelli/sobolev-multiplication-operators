\chapter{Estabilización del operador de multiplicación}\label{sec:estabilizacion}

En el Capítulo~\ref{sec:codimension_qk} construimos el marco abstracto de los índices $Q_k$. En este capítulo lo aplicamos al operador de multiplicación
\[
    \D p(z)=zp(z),
\]
y estudiamos la cadena de índices de Gelfand
\[
     Q_0(\D) \geq Q_1(\D) \geq Q_2(\D) \geq \dots
\]
cuando la norma proviene de un producto de Sobolev discreto de la forma
\begin{equation}\label{eq:norma_sobolev_discreta_general}
    \langle p,q\rangle_S
    =
    \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
    +
    \sum_{j=1}^N p'(c_j)\overline{q'(c_j)},
    \qquad p,q\in\Poly.
\end{equation}

\section{Funcionales de evaluación}

Los funcionales de evaluación y derivación aparecerán de forma recurrente en el análisis. En particular, el funcional asociado al operador de multiplicación jugará un papel central en la detección de las direcciones inestables.

\begin{defn}\label{def:funcionales_evaluacion_ch3}
    Para cada $c\in\C$ definimos inicialmente sobre $\Poly$ los funcionales lineales
    \[
        \delta_c(p):=p(c),
        \qquad
        \delta'_c(p):=p'(c),
        \qquad p\in\Poly,
    \]
    y el funcional asociado al operador de multiplicación
    \begin{equation}\label{eq:def_psi_c}
        \psi_c(p):=(\D p)'(c)=p(c)+cp'(c),
        \qquad p\in\Poly.
    \end{equation}
    Si alguno de estos funcionales es continuo respecto a la norma considerada, entonces admite una extensión continua a la compleción del espacio. En ese caso denotaremos dicha extensión, cuando convenga, por $\widetilde\delta_c$, $\widetilde\delta'_c$ o $\widetilde\psi_c$.
\end{defn}

\begin{defn}\label{def:abpe_ch3}
    Sea $\mu$ una medida positiva finita en $\C$. Diremos que $a\in\C$ es un punto de evaluación acotada (\emph{bounded point evaluation}, BPE) para $L^2(\mu)$ si existe una constante $C>0$ tal que
    \begin{equation}\label{eq:def_bpe}
        |p(a)|^2\le C\int |p(z)|^2\,d\mu(z),
        \qquad p\in\Poly.
    \end{equation}
    Diremos que $a\in\C$ es un punto de evaluación acotada analítica (\emph{analytic bounded point evaluation}, ABPE) para $L^2(\mu)$ si existe un entorno abierto $U$ de $a$ y una constante $C>0$ tales que
    \[
        |p(z)|^2\le C\int |p(\zeta)|^2\,d\mu(\zeta),
        \qquad p\in\Poly,\ z\in U.
    \]
    Véase \cite{conway2020approximationmeanrationalfunctions,aleman2009nontangential}.
\end{defn}

\begin{defn}\label{def:delta_singular}
    Sea $\langle\cdot,\cdot\rangle_{\mu}$ un producto interno sobre $\Poly$. Diremos que un punto $c\in\C$ es \emph{$\delta$-singular} respecto a $\|\cdot\|_{\mu}$ si la restricción del funcional $\delta_c$ al subespacio $\ker(\delta'_c)$ no es acotada. Equivalente y más concretamente,
    \[
        \sup\Bigl\{\frac{|p(c)|}{\|p\|_{\mu}}:p\in\ker(\delta'_c)\setminus\{0\}\Bigr\}=+\infty.
    \]
    Esto equivale a la existencia de una sucesión $\{p_n\}_{n\ge1} $ con $p_n \in \ker(\delta'_c)$ tal que
    \[
        \|p_n\|_{\mu}\to0,
        \qquad
        p_n(c)=1
        \quad\text{para todo }n\ge1.
    \]
\end{defn}

\begin{lem}\label{lema:ABPE_derivada}
    Sea $R>0$ y sea $c\in\C$ con $|c|<R$. Entonces $c$ es un punto ABPE para $L^2(\mathbf{m}_R)$ y, en consecuencia, el funcional
    \[
        \delta'_c(p)=p'(c),
        \qquad p\in\Poly,
    \]
    es continuo respecto a la norma $L^2(\mathbf{m}_R)$.
\end{lem}

\begin{proof}
    En el caso de Lebesgue sobre la circunferencia, todo punto del disco abierto $\{|z|<R\}$ es BPE y de hecho ABPE \cite{miller1999bounded,aleman2009nontangential}. Tomamos $r>0$ tal que la circunferencia $\gamma_r=\{z:|z-c|=r\}$ quede contenida en el disco $\{|z|<R\}$. Por la fórmula de Cauchy para la derivada,
    \[
        p'(c)=\frac{1}{2\pi i}\oint_{\gamma_r}\frac{p(z)}{(z-c)^2}\,dz.
    \]
    Luego
    \[
        |p'(c)|\le \frac1r\max_{z\in\gamma_r}|p(z)|.
    \]
    Como todos los puntos de $\gamma_r$ son BPE, existe $M>0$ tal que
    \[
        |p(z)|\le M\|p\|_{L^2(\mathbf{m}_R)},
        \qquad z\in\gamma_r.
    \]
    Sustituyendo se obtiene
    \[
        |p'(c)|\le \frac Mr\|p\|_{L^2(\mathbf{m}_R)}.
    \]
    Esto prueba la continuidad de $\delta'_c$.
\end{proof}

\begin{prop}\label{prop:caracterizacion_psi_c}
    Sea $\mu$ una medida positiva finita en $\C$, sea $c\in\C$ y sea
    \[
        \psi_c(p)=p(c)+cp'(c),
        \qquad p\in\Poly.
    \]
    Entonces se verifican las siguientes implicaciones:
    \begin{enumerate}
        \item si $c$ es $\delta$-singular respecto a $\|\cdot\|_{\mu}$, entonces $\delta_c$ no es acotado sobre $\Poly$;
        \item si $c$ es $\delta$-singular respecto a $\|\cdot\|_{\mu}$, entonces $\psi_c$ no es acotado sobre $\Poly$;
    \end{enumerate}
\end{prop}

\begin{proof}
    La primera implicación es inmediata, pues la no acotación de $\delta_c$ sobre $\ker(\delta'_c)$ implica su no acotación sobre el espacio mayor $\Poly$.

    Para la segunda, si $c$ es $\delta$-singular, existe una sucesión $\{p_n\}_{n\ge1}$ con $p_n \in \ker(\delta'_c)$ tal que
    \[
        \|p_n\|_{\mu}\to0,
        \qquad
        p_n(c)=1.
    \]
    Como $p_n'(c)=0$, se tiene
    \[
        \psi_c(p_n)=p_n(c)=1,
    \]
    y por tanto
    \[
        \frac{|\psi_c(p_n)|}{\|p_n\|_{\mu}}=\frac1{\|p_n\|_{\mu}}\xrightarrow[n\to\infty]{}\infty.
    \]
\end{proof}

\begin{thm}\label{thm:equivalencia_lebesgue}
    Sea $R>0$ y sea $c\in\C$. Consideremos la medida de Lebesgue normalizada $\mathbf{m}_R$ sobre $\mathbb{S}_R$. Entonces son equivalentes:
    \begin{enumerate}
        \item $c$ es $\delta$-singular respecto a $\|\cdot\|_{L^2(\mathbf{m}_R)}$;
        \item $|c|\ge R$.
    \end{enumerate}
\end{thm}

\begin{proof}
    Probamos primero que $(2)\Rightarrow(1)$. Si $|c|>R$, definimos
    \[
        r_n(z):=\frac{(n+1)c z^n-nz^{n+1}}{c^{n+1}}.
    \]
    Entonces $r_n(c)=1$, $r_n'(c)=0$ y
    \[
        \|r_n\|_{L^2(\mathbf{m}_R)}^2
        =
        (n+1)^2\left(\frac{R}{|c|}\right)^{2n}
        +
        n^2\left(\frac{R}{|c|}\right)^{2n+2}
        \xrightarrow[n\to\infty]{}0.
    \]
    Por tanto $c$ es $\delta$-singular.

    Si $|c|=R$, tomamos
    \[
        \alpha_n:=\frac{\sum_{k=1}^n k}{\sum_{k=1}^n k^2}=\frac{3}{2n+1},
        \qquad
        P_n(z):=\sum_{k=0}^n (1-\alpha_n k)\left(\frac zc\right)^k.
    \]
    Entonces $P_n'(c)=0$ y
    \[
        P_n(c)=\frac{(n+1)(n+2)}{2(2n+1)}.
    \]
    Definiendo $s_n:=P_n/P_n(c)$ se obtiene $s_n(c)=1$, $s_n'(c)=0$ y
    \[
        \|s_n\|_{L^2(\mathbf{m}_R)}^2
        =
        \frac{\|P_n\|_{L^2(\mathbf{m}_R)}^2}{|P_n(c)|^2}
        =
        \frac1{P_n(c)}
        \xrightarrow[n\to\infty]{}0.
    \]
    De nuevo $c$ es $\delta$-singular.

    Probamos ahora que $(1)\Rightarrow(2)$ por contrarrecíproco. Si $|c|<R$, entonces $c$ es BPE para $L^2(\mathbf{m}_R)$ y, por el Lema~\ref{lema:ABPE_derivada}, tanto $\delta_c$ como $\delta'_c$ son funcionales acotados respecto a $\|\cdot\|_{L^2(\mathbf{m}_R)}$. En particular, la restricción de $\delta_c$ a $\ker(\delta'_c)$ es acotada, y por ello $c$ no es $\delta$-singular.
\end{proof}

\section{Caso estable}

Este caso es conocido y ha sido tratado por Gonzalo y Escribano \cite{EG25-2} para contextos mas generales. A continuación presentamos un resultado particular para el producto de Sobolev discreto \ref{eq:norma_sobolev_discreta_general} apoyandonos en el hecho de que los puntos interiores son ABPE para la parte continua.

\begin{prop}\label{prop:atomos_interiores_acotado}
    Sea $R>0$ y sean $c_1,\dots,c_N\in\C$ tales que
    \[
        |c_j|<R,
        \qquad j=1,\dots,N.
    \]
    Consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{j=1}^N p'(c_j)\overline{q'(c_j)},
        \qquad p,q\in\Poly,
    \]
    y sea $\D p(z)=zp(z)$. Entonces $\D$ es acotado respecto a $\|\cdot\|_S$. En particular,
    \[
        Q_0(\D)<\infty.
    \]
\end{prop}

\begin{proof}
    Como $|c_j|<R$ para todo $j$, cada punto $c_j$ es ABPE para $L^2(\mathbf{m}_R)$ y, por el Lema~\ref{lema:ABPE_derivada}, la evaluación y la derivada en $c_j$ se controlan por la norma base. En consecuencia, para cada $j$ existen constantes $A_j,B_j<\infty$ tales que
    \[
        |p(c_j)|\le A_j\|p\|_{L^2(\mathbf{m}_R)},
        \qquad
        |p'(c_j)|\le B_j\|p\|_{L^2(\mathbf{m}_R)},
        \qquad p\in\Poly.
    \]
    Sea ahora $p\in\Poly$. Como $|z|=R$ sobre $\mathbb{S}_R$,
    \[
        \|\D p\|_{L^2(\mathbf{m}_R)}=R\|p\|_{L^2(\mathbf{m}_R)}\le R\|p\|_S.
    \]
    Además, para cada $j$,
    \[
        |(\D p)'(c_j)|=|p(c_j)+c_jp'(c_j)|
        \le
        (A_j+|c_j|B_j)\|p\|_{L^2(\mathbf{m}_R)}
        \le
        (A_j+|c_j|B_j)\|p\|_S.
    \]
    Por tanto,
    \[
        \|\D p\|_S^2
        =
        R^2\|p\|_{L^2(\mathbf{m}_R)}^2
        +
        \sum_{j=1}^N |(\D p)'(c_j)|^2
        \le
        \left(R^2+\sum_{j=1}^N (A_j+|c_j|B_j)^2\right)\|p\|_S^2.
    \]
    Esto prueba la acotación de $\D$, y por definición $Q_0(\D)=\|\D\|_S<\infty$.
\end{proof}

\begin{exm}\label{exm:c0_raiz2}
    La cota general del resultado anterior no es óptima en todos los casos. Tomemos $R=1$ y consideremos el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{M_1} p(z)\overline{q(z)}\,d\mathbf m_1(z)
        +
        p'(0)\overline{q'(0)},
        \qquad p,q\in\Poly.
    \]
    Si $p(z)=\sum_{k=0}^n a_k z^k$, entonces $p'(0)=a_1$ y
    \begin{equation}\label{eq:norma_in_0}
        \|p\|_S^2=|a_0|^2+2|a_1|^2+\sum_{k=2}^n |a_k|^2.
    \end{equation}
    Además, para $\D p(z)=zp(z)$ se tiene $(\D p)'(0)=a_0$, luego
    \begin{equation}\label{eq:norma_out_0}
        \|\D p\|_S^2=2|a_0|^2+|a_1|^2+\sum_{k=2}^n |a_k|^2.
    \end{equation}
    Por tanto,
    \begin{equation}\label{eq:ratio_c0_raiz2}
        \frac{\|\D p\|_S^2}{\|p\|_S^2}
        =
        2-
        \frac{3|a_1|^2+\sum_{k\ge2}|a_k|^2}{\|p\|_S^2}
        \le2,
    \end{equation}
    y el máximo se alcanza en las funciones constantes. En consecuencia,
    \[
        \|\D\|_S=Q_0(\D)=\sqrt2.
    \]

    Además, si imponemos la restricción $p(0)=0$, entonces $a_0=0$ y las expresiones \eqref{eq:norma_in_0}--\eqref{eq:norma_out_0} dan
    \[
        \|\D p\|_S^2=|a_1|^2+\sum_{k=2}^n |a_k|^2\le \|p\|_S^2,
    \]
    de modo que sobre el subespacio $\ker(\delta_0)$ el operador actúa con norma a lo sumo $1$.
\end{exm}

\section{Caso inestable}

Notese que al abordar el caso estable conocido se ha impuesto que $|c_j|<R$ para todo $j$. Luego es natural preguntarse qué ocurre cuando alguno de los puntos $c_j$ cumple que $|c_j|\ge R$. Empecemos por un caso sencillo con un solo átomo.

\begin{thm}\label{thm:un_atomo_exterior}\label{prop:lebesgue_pre_umbral_Qk}
    Sea $R>0$, sea $c\in\C$ con $|c|\ge R$ y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Sea además $\D p(z)=zp(z)$ y sea
    \[
        \psi_c(p)=p(c)+cp'(c)=(\D p)'(c).
    \]
    Entonces
    \[
        Q_0(\D)=\infty,
        \qquad
        Q_1(\D)\le R<\infty.
    \]
\end{thm}

\begin{proof}
    Por el Teorema~\ref{thm:equivalencia_lebesgue}, existe una sucesión $(p_n)\subset\Poly$ tal que
    \[
        p_n(c)=1,
        \qquad
        p_n'(c)=0,
        \qquad
        \|p_n\|_{L^2(\mathbf{m}_R)}\xrightarrow[n\to\infty]{}0.
    \]
    Como la parte discreta de la norma de Sobolev es $|p_n'(c)|^2=0$, se tiene también
    \[
        \|p_n\|_S\xrightarrow[n\to\infty]{}0.
    \]
    Además,
    \[
        \psi_c(p_n)=p_n(c)+cp_n'(c)=1.
    \]
    Por tanto, $\psi_c$ no es acotado respecto a $\|\cdot\|_S$.

    Además,
    \[
        |\psi_c(p)|=|(\D p)'(c)|\le \|\D p\|_S,
        \qquad p\in\Poly.
    \]
    Si $\D$ fuese acotado, también lo sería $\psi_c$, contradicción. Luego $\D$ no es acotado y, por tanto,
    \[
        Q_0(\D)=\|\D\|_S=\infty.
    \]

    Para obtener una restricción estable, consideramos
    \[
        W:=\ker(\psi_c)=\{p\in\Poly:\psi_c(p)=0\}.
    \]
    Este subespacio satisface $\operatorname{codim}_{\Poly}(W)=1$. Si $p\in W$, entonces
    \[
        \|\D p\|_S^2
        =
        \int_{\mathbb{S}_R}|zp(z)|^2\,d\mathbf{m}_R(z)
        +
        |\psi_c(p)|^2
        =
        \int_{\mathbb{S}_R}|zp(z)|^2\,d\mathbf{m}_R(z)
        \le
        R^2\int_{\mathbb{S}_R}|p(z)|^2\,d\mathbf{m}_R(z)
        \le
        R^2\|p\|_S^2.
    \]
    Por consiguiente, $\|\D|_W\|_S\le R$. Al tomar el ínfimo sobre los subespacios de codimensión algebraica menor que $2$, se obtiene
    \[
        Q_1(\D)\le R.
    \]
\end{proof}

\begin{cor}\label{cor:dicotomia_un_atomo}
    Sea $R>0$, sea $c\in\C$ y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Entonces el operador $\D p(z)=zp(z)$ es acotado si y solo si $|c|<R$. Equivalentemente,
    \[
        |c|<R
        \iff
        Q_0(\D)<\infty,
    \]
    mientras que
    \[
        |c|\ge R
        \iff
        Q_0(\D)=\infty,
    \]
    y en este segundo caso además
    \[
        Q_1(\D)<\infty.
    \]
\end{cor}

\begin{proof}
    Si $|c|<R$, la Proposición~\ref{prop:atomos_interiores_acotado} aplicada con $N=1$ prueba que $\D$ es acotado y, por tanto, $Q_0(\D)<\infty$. Si $|c|\ge R$, el Teorema~\ref{thm:un_atomo_exterior} da $Q_0(\D)=\infty$ y $Q_1(\D)\le R<\infty$.
\end{proof}

Pasamos ahora al producto de Sobolev discreto \ref{eq:norma_sobolev_discreta_general} con un conjunto finito de átomos derivados. Sea $F\subset\C$ un conjunto finito y consideremos
\[
    \langle p,q\rangle_S
    =
    \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
    +
    \sum_{c\in F} p'(c)\overline{q'(c)},
    \qquad p,q\in\Poly.
\]

\begin{defn}\label{def:class_points_new}
    En el marco anterior definimos
    \[
        I_{\mathrm{in}}:=\{c\in F:|c|<R\},
        \qquad
        I_{\mathrm{out}}:=\{c\in F:|c|\ge R\},
    \]
    y escribimos
    \[
        m:=|I_{\mathrm{out}}|.
    \]
\end{defn}

La Proposición~\ref{prop:atomos_interiores_acotado} ya resuelve completamente el caso $m=0$. Por tanto, en los apartados que siguen supondremos cuando sea necesario que
\[
    I_{\mathrm{out}}\neq\varnothing,
    \qquad
    m\ge1.
\]

\begin{rmk}
	Por el Teorema~\ref{thm:equivalencia_lebesgue} y la Proposición~\ref{prop:caracterizacion_psi_c}, cada $c\in I_{\mathrm{out}}$ da lugar a un funcional $\psi_c$ no acotado, mientras que los puntos de $I_{\mathrm{in}}$ permanecen controlados por la parte continua.
\end{rmk}

\section{Subespacio estable}

Veamos que podemos construir un subespacio estable imponiendo exactamente las restricciones que anulan las contribuciones discretas del producto de Sobolev \ref{eq:norma_sobolev_discreta_general}

\begin{defn}\label{def:vout_ch3}
    Sea $F\subset\C$ un conjunto finito y consideremos los funcionales asociados al operador de multiplicación \ref{eq:def_psi_c} para $c\in I_{\mathrm{out}}$. Definimos el subespacio
    \begin{equation}\label{eq:def_vout}
        V_{\mathrm{out}}:=\bigcap_{c\in I_{\mathrm{out}}}\ker(\psi_c).
    \end{equation}
\end{defn}

\begin{lem}\label{lem:hermite_datos_finito}
    Sea $F\subset\C$ un conjunto finito. Entonces, para cualesquiera familias de datos $(\alpha_a)_{a\in F}$ y $(\beta_a)_{a\in F}$ en $\C$, existe un polinomio $h\in\Poly$ tal que
    \[
        h(a)=\alpha_a,
        \qquad
        h'(a)=\beta_a,
        \qquad a\in F.
    \]
    Véase \cite{Dav75}.
\end{lem}

\begin{lem}\label{lem:independencia_funcionales_psi}
    Sea $R>0$, sea $I_{\mathrm{in}}\subset\C$ el conjunto de atomos interiores y sea $\psi_c(p)$ el funcional asociado al operador de multiplicación \ref{eq:def_psi_c} para cada $c\in I_{\mathrm{out}}$. El sistema $\{\psi_c\}_{c\in I_{\mathrm{out}}}$ es linealmente independiente. En particular,
    \[
        \dim[\psi_c:c\in I_{\mathrm{out}}]=m,
        \qquad m:=|I_{\mathrm{out}}|.
    \]
\end{lem}

\begin{proof}
    Fijado $c\in I_{\mathrm{out}}$, aplicamos el Lema~\ref{lem:hermite_datos_finito} al conjunto $E=I_{\mathrm{out}}$ con datos
    \[
        \alpha_d=\delta_{dc},
        \qquad
        \beta_d=0,
        \qquad d\in I_{\mathrm{out}}.
    \]
    Obtenemos un polinomio $h_c\in\Poly$ tal que
    \[
        h_c(d)=\delta_{dc},
        \qquad
        h_c'(d)=0,
        \qquad d\in I_{\mathrm{out}}.
    \]
    Entonces
    \[
        \psi_d(h_c)=h_c(d)+dh_c'(d)=\delta_{dc},
        \qquad d\in I_{\mathrm{out}},
    \]
    y esto prueba la independencia lineal.
\end{proof}

\begin{lem}\label{lem:vout_codimension}
    Sea $R>0$, sea $I_{\mathrm{out}}\subset\C$ el conjunto de atomos interiores y consideremos el subespacio $V_{\mathrm{out}}$ \ref{eq:def_vout}.Entonces
    \[
        \operatorname{codim}_{\Poly}(V_{\mathrm{out}})=m,
        \qquad m:=|I_{\mathrm{out}}|.
    \]
\end{lem}

\begin{proof}
    Por definición, $V_{\mathrm{out}}$ es una intersección finita de núcleos de funcionales lineales. El Lema~\ref{lem:independencia_funcionales_psi} muestra que los funcionales $\{\psi_c\}_{c\in I_{\mathrm{out}}}$ son linealmente independientes y generan un espacio de dimensión $m$. La Proposición~\ref{prop:finite-codim-kernels} da entonces la igualdad anunciada.
\end{proof}

\begin{lem}\label{lem:vout_acotacion}
    Sea $R>0$, sea $F= I_{\mathrm{in}} \cup I_{\mathrm{out}}\subset\C$ finito, sea $V_{\mathrm{out}}$ el subespacio estable \ref{eq:def_vout} y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Entonces existe una constante $C_{\mathrm{out}}<\infty$ tal que
    \[
        \|\D p\|_S\le C_{\mathrm{out}}\|p\|_S,
        \qquad p\in V_{\mathrm{out}}.
    \]
    En particular, $\D|_{V_{\mathrm{out}}}$ es acotado. Si además $I_{\mathrm{in}}=\varnothing$, puede tomarse $C_{\mathrm{out}}=R$.
\end{lem}

\begin{proof}
    Sea $p\in V_{\mathrm{out}}$. Entonces, para todo $c\in I_{\mathrm{out}}$,
    \[
        (\D p)'(c)=\psi_c(p)=0.
    \]
    Es decir, todas las contribuciones discretas procedentes de $I_{\mathrm{out}}$ desaparecen en la norma de Sobolev de $\D p$.

    Si $c\in I_{\mathrm{in}}$, entonces $|c|<R$. Por el Lema~\ref{lema:ABPE_derivada} y la propiedad ABPE de los puntos interiores, existen constantes $A_c,B_c<\infty$ tales que
    \[
        |p(c)|\le A_c\|p\|_S,
        \qquad
        |p'(c)|\le B_c\|p\|_S.
    \]
    De aquí se deduce
    \[
        |(\D p)'(c)|=|p(c)+cp'(c)|\le (A_c+|c|B_c)\|p\|_S.
    \]

    Además, como $|z|=R$ en $\mathbb{S}_R$,
    \[
        \|zp\|_{L^2(\mathbf{m}_R)}\le R\|p\|_{L^2(\mathbf{m}_R)}\le R\|p\|_S.
    \]
    Por consiguiente,
    \[
        \|\D p\|_S^2
        =
        \|zp\|_{L^2(\mathbf{m}_R)}^2
        +
        \sum_{c\in I_{\mathrm{in}}}|(\D p)'(c)|^2
        \le
        \left(R^2+\sum_{c\in I_{\mathrm{in}}}(A_c+|c|B_c)^2\right)\|p\|_S^2.
    \]
    Si $I_{\mathrm{in}}=\varnothing$, la suma anterior desaparece y queda
    \[
        \|\D p\|_S^2\le R^2\|p\|_S^2,
    \]
    de modo que puede tomarse $C_{\mathrm{out}}=R$.
\end{proof}

\begin{prop}\label{prop:umbral_fase1}
    Sea $R>0$, sea $F= I_{\mathrm{in}} \cup I_{\mathrm{out}}\subset\C$ finito, sea $V_{\mathrm{out}}$ el subespacio estable \ref{eq:def_vout} y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Entonces
    \[
        Q_m(\D)<\infty.
    \]
    Además,
    \[
        Q_m(\D)\le C_{\mathrm{out}},
    \]
    donde $C_{\mathrm{out}}$ es la constante del Lema~\ref{lem:vout_acotacion}.
\end{prop}

\begin{proof}
    Por el Lema~\ref{lem:vout_codimension}, el subespacio $V_{\mathrm{out}}$ tiene codimensión exacta $m$. Por el Lema~\ref{lem:vout_acotacion}, la restricción $\D|_{V_{\mathrm{out}}}$ es acotada. La definición de $Q_m$ da inmediatamente
    \[
        Q_m(\D)\le \|\D|_{V_{\mathrm{out}}}\|_S\le C_{\mathrm{out}}<\infty.
    \]
\end{proof}

\begin{cor}\label{cor:sin_interiores}
    Sea $R>0$, sea $F= I_{\mathrm{in}} \cup I_{\mathrm{out}}\subset\C$ finito, sea $V_{\mathrm{out}}$ el subespacio estable \ref{eq:def_vout} y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Si $I_{\mathrm{in}}=\varnothing$, entonces
    \[
        Q_m(\D)\le R.
    \]
\end{cor}

\begin{proof}
    Si $I_{\mathrm{in}}=\varnothing$, el Lema~\ref{lem:vout_acotacion} da
    \[
        \|\D p\|_S\le R\|p\|_S,
        \qquad p\in V_{\mathrm{out}}.
    \]
    Como $V_{\mathrm{out}}$ tiene codimensión $m$, la definición de $Q_m$ implica
    \[
        Q_m(\D)\le \|\D|_{V_{\mathrm{out}}}\|_S\le R.
    \]
\end{proof}

\begin{thm}\label{thm:caso_mixto}
    Sea $R>0$, sea $F= I_{\mathrm{in}} \cup I_{\mathrm{out}}\subset\C$ finito, sea $V_{\mathrm{out}}$ el subespacio estable \ref{eq:def_vout} y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Supongamos además que
    \[
        I_{\mathrm{in}}\neq\varnothing,
        \qquad
        I_{\mathrm{out}}\neq\varnothing.
    \]
    Entonces se obtiene la cota superior
    \[
        Q_m(\D)^2\le \|\D|_{V_{\mathrm{out}}}\|_S^2
        \le
        \max\{R^2,\lambda_{\max}(\mathbf B,\mathbf A)\},
    \]
    donde $\mathbf A=(\langle e_j,e_i\rangle_S)_{i,j=1}^d$ y $\mathbf B=(\langle \D e_j,\D e_i\rangle_S)_{i,j=1}^d$ son las matrices de Gram respecto a una base $\{e_1,\dots,e_d\}$ del subespacio finito-dimensional generado por los representadores de Riesz de los funcionales acotados
    \[
        \ell_c(p):=p'(c),\qquad c\in F,
    \]
    y
    \[
        \eta_c(p):=(\D p)'(c)=p(c)+cp'(c),\qquad c\in I_{\mathrm{in}},
    \]
    en la completación de $V_{\mathrm{out}}$.
\end{thm}

\begin{proof}
    Sea $\mathcal H_{\mathrm{out}}$ la completación de $V_{\mathrm{out}}$ con la norma de Sobolev. Los funcionales $\ell_c$ son acotados sobre $\mathcal H_{\mathrm{out}}$ para todo $c\in F$, y los funcionales $\eta_c$ son acotados para todo $c\in I_{\mathrm{in}}$ por el control interior probado antes. Sea $E\subset\mathcal H_{\mathrm{out}}$ el subespacio finito-dimensional generado por sus representadores de Riesz.

    Dado $p\in\mathcal H_{\mathrm{out}}$, escribimos su descomposición ortogonal $p=x+y$ con $x\in E$ e $y\in E^\perp$. Entonces $y'(c)=0$ para todo $c\in F$ y $(\D y)'(c)=0$ para todo $c\in I_{\mathrm{in}}$. Además, como $y\in\mathcal H_{\mathrm{out}}$, también se tiene $(\D y)'(c)=0$ para todo $c\in I_{\mathrm{out}}$, pues en $\mathcal H_{\mathrm{out}}$ las restricciones definitorias de $V_{\mathrm{out}}$ se extienden por continuidad como funcionales nulos. Por tanto, las contribuciones discretas desaparecen tanto en $\|y\|_S$ como en $\|\D y\|_S$, y queda
    \[
        \|\D y\|_S^2=R^2\|y\|_{L^2(\mathbf{m}_R)}^2=R^2\|y\|_S^2.
    \]

    La ortogonalidad respecto a ambas formas cuadráticas da entonces
    \[
        \frac{\|\D p\|_S^2}{\|p\|_S^2}
        =
        \frac{\|\D x\|_S^2+R^2\|y\|_S^2}{\|x\|_S^2+\|y\|_S^2}
        \le
        \max\!\left\{R^2,\sup_{\substack{q\in E\\ q\ne0}}\frac{\|\D q\|_S^2}{\|q\|_S^2}\right\}.
    \]
    El segundo término es el mayor cociente de Rayleigh generalizado del par $(\mathbf B,\mathbf A)$ asociado a una base de $E$, esto es, $\lambda_{\max}(\mathbf B,\mathbf A)$. Se concluye así la cota superior anunciada para $\|\D|_{V_{\mathrm{out}}}\|_S^2$, y la desigualdad para $Q_m(\D)$ es inmediata por definición.
\end{proof}

\section{Necesidad de la codimensión \texorpdfstring{$m$}{m}}

Veamos ahora que la codimensión $m$ es necesaria para estabilizar el operador de multiplicación, luego no podemos construir un subespacio estable con codimensión menor que $m$. Pero antes debemos ver unos resultados que nos ayudarán a construir las sucesiones que usaremos para ver la necesidad de la codimensión $m$.

\begin{lem}\label{lem:paquete_local_exterior}
    Sea $R>0$ y sea $c\in\C$ con $|c|\ge R$. Entonces existe una sucesión $\{r_{c,n}\}_{n\ge1}$ con $r_{c,n}\in\Poly$ tal que
    \[
        r_{c,n}(c)=1,
        \qquad
        r'_{c,n}(c)=0,
        \qquad
        \|r_{c,n}\|_{L^2(\mathbf{m}_R)}\xrightarrow[n\to\infty]{}0.
    \]
\end{lem}

\begin{proof}
    Distinguimos dos casos.

    Si $|c|>R$, definimos
    \[
        r_{c,n}(z):=\frac{(n+1)c z^n-nz^{n+1}}{c^{n+1}}.
    \]
    Entonces
    \[
        r_{c,n}(c)=1,
        \qquad
        r'_{c,n}(c)=0.
    \]
    Además,
    \[
        \|r_{c,n}\|_{L^2(\mathbf{m}_R)}^2
        =
        (n+1)^2\left(\frac{R}{|c|}\right)^{2n}
        +
        n^2\left(\frac{R}{|c|}\right)^{2n+2}
        \xrightarrow[n\to\infty]{}0,
    \]
    porque $R/|c|<1$.

    Si $|c|=R$, tomamos
    \[
        \alpha_n:=\frac{\sum_{k=1}^n k}{\sum_{k=1}^n k^2}=\frac{3}{2n+1},
    \]
    y definimos
    \[
        P_{c,n}(z):=\sum_{k=0}^n (1-\alpha_n k)\left(\frac zc\right)^k.
    \]
    Entonces
    \[
        P'_{c,n}(c)=0,
        \qquad
        P_{c,n}(c)=\frac{(n+1)(n+2)}{2(2n+1)}.
    \]
    Definimos
    \[
        r_{c,n}:=\frac{P_{c,n}}{P_{c,n}(c)}.
    \]
    Así,
    \[
        r_{c,n}(c)=1,
        \qquad
        r'_{c,n}(c)=0.
    \]
    Además,
    \[
        \|r_{c,n}\|_{L^2(\mathbf{m}_R)}^2
        =
        \frac{\|P_{c,n}\|_{L^2(\mathbf{m}_R)}^2}{|P_{c,n}(c)|^2}
        =
        \frac1{P_{c,n}(c)}
        \xrightarrow[n\to\infty]{}0.
    \]
    Esto concluye la prueba.
\end{proof}

\begin{lem}\label{lem:paquetes_diagonales}
    Sea $R>0$, sea $F\subset\C$ finito y definamos
    \[
        I_{\mathrm{out}}:=\{c\in F:|c|\ge R\}.
    \]
    Para cada $c\in I_{\mathrm{out}}$ existe una sucesión $(u_{c,n})_{n\ge1}\subset\Poly$ tal que
    \[
        \|u_{c,n}\|_S\xrightarrow[n\to\infty]{}0,
    \]
    y además
    \[
        \psi_d(u_{c,n})=\delta_{dc},
        \qquad d\in I_{\mathrm{out}}.
    \]
\end{lem}

\begin{proof}
    Fijamos $c\in I_{\mathrm{out}}$. Aplicamos el Lema~\ref{lem:hermite_datos_finito} al conjunto finito $F$ y obtenemos un polinomio $H_c\in\Poly$ tal que
    \[
        H_c(c)=1,
        \qquad
        H'_c(c)=0,
    \]
    y, para todo $a\in F\setminus\{c\}$,
    \[
        H_c(a)=0,
        \qquad
        H'_c(a)=0.
    \]
    Sea $(r_{c,n})$ la sucesión dada por el Lema~\ref{lem:paquete_local_exterior} y definamos
    \[
        u_{c,n}:=H_c r_{c,n}.
    \]

    Si $d=c$, entonces
    \[
        u_{c,n}(c)=H_c(c)r_{c,n}(c)=1
    \]
    y
    \[
        u'_{c,n}(c)=H'_c(c)r_{c,n}(c)+H_c(c)r'_{c,n}(c)=0.
    \]
    Por tanto,
    \[
        \psi_c(u_{c,n})=u_{c,n}(c)+cu'_{c,n}(c)=1.
    \]
    Si $d\in I_{\mathrm{out}}\setminus\{c\}$, entonces $H_c(d)=H'_c(d)=0$, luego
    \[
        u_{c,n}(d)=0,
        \qquad
        u'_{c,n}(d)=0,
    \]
    y por tanto
    \[
        \psi_d(u_{c,n})=0.
    \]
    Hemos probado así que
    \[
        \psi_d(u_{c,n})=\delta_{dc},
        \qquad d\in I_{\mathrm{out}}.
    \]

    Queda justificar que $\|u_{c,n}\|_S\to0$. Para todo $a\in F$, la elección de $H_c$ y $r_{c,n}$ da
    \[
        u'_{c,n}(a)=H'_c(a)r_{c,n}(a)+H_c(a)r'_{c,n}(a)=0.
    \]
    En consecuencia, todas las contribuciones discretas de la norma de Sobolev de $u_{c,n}$ se anulan y
    \[
        \|u_{c,n}\|_S^2=\|u_{c,n}\|_{L^2(\mathbf{m}_R)}^2.
    \]
    Como $H_c$ es fijo,
    \[
        \|u_{c,n}\|_{L^2(\mathbf{m}_R)}
        =
        \|H_c r_{c,n}\|_{L^2(\mathbf{m}_R)}
        \le
        \|H_c\|_{L^\infty(\mathbb{S}_R)}\,\|r_{c,n}\|_{L^2(\mathbf{m}_R)}.
    \]
    Por el Lema~\ref{lem:paquete_local_exterior}, el último término tiende a cero. Luego
    \[
        \|u_{c,n}\|_S\xrightarrow[n\to\infty]{}0.
    \]
\end{proof}

\begin{lem}\label{lem:imposicion_restricciones}
    Sea $R>0$, sea $F\subset\C$ finito y definamos
    \[
        I_{\mathrm{out}}:=\{c\in F:|c|\ge R\},
        \qquad
        m:=|I_{\mathrm{out}}|\ge1.
    \]
    Sea además $V\subset\Poly$ un subespacio con
    \[
        \operatorname{codim}_{\Poly}(V)<m.
    \]
    Para cada $n\ge1$ definimos
    \[
        W_n:=[u_{c,n}:c\in I_{\mathrm{out}}],
    \]
    donde los paquetes diagonales $u_{c,n}$ son los del Lema~\ref{lem:paquetes_diagonales}. Entonces, para cada $n\ge1$, existe un polinomio no nulo
    \[
        p_n\in V\cap W_n
    \]
    de la forma
    \[
        p_n=\sum_{c\in I_{\mathrm{out}}}\alpha_c^{(n)}u_{c,n},
    \]
    con coeficientes normalizados por
    \[
        \sum_{c\in I_{\mathrm{out}}}|\alpha_c^{(n)}|^2=1.
    \]
\end{lem}

\begin{proof}
    Escribimos $\operatorname{codim}_{\Poly}(V)=r<m$. Existen entonces funcionales lineales independientes $L_1,\dots,L_r:\Poly\to\C$ tales que
    \[
        V=\bigcap_{i=1}^r\ker(L_i).
    \]
    Para cada $n$ consideramos la matriz
    \[
        A_n:=\bigl(L_i(u_{c,n})\bigr)_{\substack{1\le i\le r\\ c\in I_{\mathrm{out}}}}\in\C^{r\times m}.
    \]
    Como $r<m$, el núcleo de $A_n$ es no trivial. Elegimos
    \[
        \alpha^{(n)}=(\alpha_c^{(n)})_{c\in I_{\mathrm{out}}}\in\ker(A_n),
        \qquad
        \|\alpha^{(n)}\|_{\ell^2}=1,
    \]
    y definimos
    \[
        p_n:=\sum_{c\in I_{\mathrm{out}}}\alpha_c^{(n)}u_{c,n}.
    \]
    Entonces, para cada $i=1,\dots,r$,
    \[
        L_i(p_n)=\sum_{c\in I_{\mathrm{out}}}\alpha_c^{(n)}L_i(u_{c,n})=0,
    \]
    y por tanto $p_n\in V$. Además $p_n\in W_n$ por construcción.

    Resta ver que $p_n\neq0$. Si fuese $p_n=0$, entonces aplicando $\psi_d$ para un $d\in I_{\mathrm{out}}$ obtendríamos
    \[
        0=\psi_d(p_n)=\sum_{c\in I_{\mathrm{out}}}\alpha_c^{(n)}\psi_d(u_{c,n})=\alpha_d^{(n)}.
    \]
    Como esto vale para todo $d\in I_{\mathrm{out}}$, se seguiría $\alpha^{(n)}=0$, contradiciendo la normalización $\|\alpha^{(n)}\|_{\ell^2}=1$.
\end{proof}

Ahora procedemos a ver que la codimensión $m$ es necesaria para estabilizar el operador de multiplicación.

\begin{thm}\label{thm:umbral_fase2}\label{prop:umbral_fase2}
    Sea $R>0$, sea $F\subset\C$ un conjunto finito y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Sea
    \[
        I_{\mathrm{out}}:=\{c\in F:|c|\ge R\},
        \qquad
        m:=|I_{\mathrm{out}}|.
    \]
    Si $m\ge1$, entonces para todo subespacio $V\subset\Poly$ con
    \[
        \operatorname{codim}_{\Poly}(V)<m,
    \]
    la restricción $\D|_V$ no es acotada. En particular,
    \[
        Q_{m-1}(\D)=\infty.
    \]
\end{thm}

\begin{proof}
    Sea $V\subset\Poly$ con $\operatorname{codim}_{\Poly}(V)<m$. Para cada $n\ge1$ consideramos el subespacio
    \[
        W_n:=[u_{c,n}:c\in I_{\mathrm{out}}],
    \]
    donde los $u_{c,n}$ son los paquetes diagonales del Lema~\ref{lem:paquetes_diagonales}. Por el Lema~\ref{lem:imposicion_restricciones}, para cada $n$ existe un polinomio no nulo
    \[
        p_n=\sum_{c\in I_{\mathrm{out}}}\alpha_c^{(n)}u_{c,n}\in V\cap W_n
    \]
    con
    \[
        \sum_{c\in I_{\mathrm{out}}}|\alpha_c^{(n)}|^2=1.
    \]

    Por el Lema~\ref{lem:paquetes_diagonales}, para cada $d\in I_{\mathrm{out}}$ se tiene
    \[
        \psi_d(p_n)=\sum_{c\in I_{\mathrm{out}}}\alpha_c^{(n)}\psi_d(u_{c,n})=\alpha_d^{(n)}.
    \]
    En consecuencia,
    \[
        \sum_{d\in I_{\mathrm{out}}}|\psi_d(p_n)|^2
        =
        \sum_{d\in I_{\mathrm{out}}}|\alpha_d^{(n)}|^2
        =
        1.
    \]
    Como $\psi_d(p_n)=(\D p_n)'(d)$, la parte discreta de la norma de Sobolev de $\D p_n$ satisface
    \[
        \|\D p_n\|_S^2\ge \sum_{d\in I_{\mathrm{out}}}|(\D p_n)'(d)|^2=1.
    \]
    Por otra parte,
    \[
        \|p_n\|_S
        \le
        \sum_{c\in I_{\mathrm{out}}}|\alpha_c^{(n)}|\,\|u_{c,n}\|_S
        \le
        \sqrt m\max_{c\in I_{\mathrm{out}}}\|u_{c,n}\|_S
        \xrightarrow[n\to\infty]{}0
    \]
    por el Lema~\ref{lem:paquetes_diagonales}. Por tanto,
    \[
        \frac{\|\D p_n\|_S}{\|p_n\|_S}\xrightarrow[n\to\infty]{}\infty,
    \]
    y la restricción $\D|_V$ no puede ser acotada. Como $V$ era arbitrario entre los subespacios de codimensión menor que $m$, concluimos que $Q_{m-1}(\D)=\infty$.
\end{proof}

\section{Teorema de estabilización}

\begin{thm}[Estabilización]\label{thm:estabilizacion}
    Sea $R>0$, sea $F\subset\C$ finito y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Sea además
    \[
        I_{\mathrm{out}}:=\{c\in F:|c|\ge R\},
        \qquad
        m:=|I_{\mathrm{out}}|.
    \]
    Entonces se verifican las siguientes afirmaciones:
    \begin{enumerate}
        \item Si $m=0$, entonces $\D$ es acotado y
              \[
                  Q_0(\D)<\infty.
              \]
        \item Si $m\ge1$, entonces
              \[
                  Q_0(\D)=Q_1(\D)=\cdots=Q_{m-1}(\D)=\infty,
                  \qquad
                  Q_m(\D)<\infty.
              \]
    \end{enumerate}
\end{thm}

\begin{proof}
    Si $m=0$, todos los átomos pertenecen al interior de $\mathbb{S}_R$ y la Proposición~\ref{prop:atomos_interiores_acotado} prueba directamente que $\D$ es acotado, es decir, $Q_0(\D)<\infty$.

    Supongamos ahora que $m\ge1$. El Teorema~\ref{thm:umbral_fase2} da
    \[
        Q_{m-1}(\D)=\infty.
    \]
    Como la sucesión $\{Q_k(\D)\}_{k\ge0}$ es monótona no creciente por la Proposición~\ref{prop:propiedades_Qk}, se sigue que
    \[
        Q_0(\D)=Q_1(\D)=\cdots=Q_{m-1}(\D)=\infty.
    \]
    Por otra parte, la Proposición~\ref{prop:umbral_fase1} muestra que
    \[
        Q_m(\D)<\infty.
    \]
    Esto concluye la demostración.
\end{proof}

\section{Cotas generales para \texorpdfstring{$Q_k(\D)$}{Qk(D)}}

Las cotas de esta sección solo usan la parte continua sobre $\mathbb{S}_R$ y la estructura derivativa de la norma de Sobolev. Resultarán útiles en el paso matricial del capítulo siguiente.

\begin{prop}\label{prop:qk_general_cota_inferior_R}
    Sea $R>0$, sea $F\subset\C$ finito y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R}p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Entonces, para todo $k\ge0$,
    \[
        Q_k(\D)\ge R.
    \]
\end{prop}

\begin{proof}
    Sea
    \[
        E:=\{p\in\Poly:p'(c)=0\text{ para todo }c\in F\}.
    \]
    Entonces
    \[
        E=\ker T,
        \qquad
        T:\Poly\to\C^{|F|},
        \qquad
        T(p):=(p'(c))_{c\in F},
    \]
    y por tanto $E$ tiene codimensión finita. Si $W\subset\Poly$ tiene codimensión finita, entonces $W\cap E\neq\{0\}$.

    Fijado ahora $k\ge0$, sea $W\subset\Poly$ con
    \[
        \operatorname{codim}_{\Poly}(W)<k+1.
    \]
    Tomamos $0\neq p\in W\cap E$. Como $p'(c)=0$ para todo $c\in F$,
    \[
        \|p\|_S^2=\|p\|_{L^2(\mathbf{m}_R)}^2.
    \]
    Además,
    \[
        (\D p)'(c)=p(c),
        \qquad c\in F.
    \]
    Por consiguiente,
    \[
        \|\D p\|_S^2
        =
        R^2\|p\|_{L^2(\mathbf{m}_R)}^2
        +
        \sum_{c\in F}|p(c)|^2
        \ge
        R^2\|p\|_S^2.
    \]
    Luego
    \[
        \|\D|_W\|_S\ge \frac{\|\D p\|_S}{\|p\|_S}\ge R.
    \]
    Tomando ínfimo sobre los subespacios $W$ con $\operatorname{codim}_{\Poly}(W)<k+1$, obtenemos $Q_k(\D)\ge R$.
\end{proof}

\begin{prop}\label{prop:qN_general_cota_superior_R}
    Sea $R>0$, sea $F\subset\C$ finito, sea $N:=|F|$ y consideremos sobre $\Poly$ el producto escalar
    \[
        \langle p,q\rangle_S
        =
        \int_{\mathbb{S}_R}p(z)\overline{q(z)}\,d\mathbf{m}_R(z)
        +
        \sum_{c\in F} p'(c)\overline{q'(c)},
        \qquad p,q\in\Poly.
    \]
    Para cada $c\in F$, definamos
    \[
        \eta_c(p):=p(c)+cp'(c),
        \qquad p\in\Poly,
    \]
    y sea
    \[
        W_N:=\bigcap_{c\in F}\ker\eta_c.
    \]
    Entonces
    \[
        \operatorname{codim}_{\Poly}(W_N)\le N
        \qquad\text{y}\qquad
        Q_N(\D)\le R.
    \]
\end{prop}

\begin{proof}
    Consideremos la aplicación lineal
    \[
        \Lambda_N:\Poly\to\C^N,
        \qquad
        \Lambda_N(p):=(\eta_c(p))_{c\in F}.
    \]
    Entonces $W_N=\ker\Lambda_N$, y por tanto
    \[
        \operatorname{codim}_{\Poly}(W_N)=\operatorname{rg}(\Lambda_N)\le N.
    \]
    Si $p\in W_N$, entonces
    \[
        (\D p)'(c)=\eta_c(p)=0,
        \qquad c\in F.
    \]
    En consecuencia,
    \[
        \|\D p\|_S^2=R^2\|p\|_{L^2(\mathbf{m}_R)}^2\le R^2\|p\|_S^2.
    \]
    Luego
    \[
        \|\D|_{W_N}\|_S\le R.
    \]
    Como $W_N$ es admisible en la definición de $Q_N$, concluimos que
    \[
        Q_N(\D)\le R.
    \]
\end{proof}
