\chapter{Estabilización y régimen no acotado}\label{sec:estabilizacion}

En el Capítulo~\ref{sec:codimension_qk} construimos el marco abstracto: codimensión algebraica, subespacios admisibles, restricciones por funcionales lineales e índice $Q_k(T)$ para operadores arbitrarios en $\Poly$. Aquí fijamos
\[
\D p(z)=zp(z),
\]
y estudiamos la cadena $\{Q_k(\D)\}_{k\ge0}$ cuando la norma proviene de un producto de Sobolev discreto.

La pregunta estructural es \emph{cuándo aparece por primera vez un índice finito para $\D$}. La respuesta depende de la posición de los átomos discretos respecto al soporte continuo. El resultado central del capítulo afirma que la cadena de infinitos
\[
\infty=Q_0(\D)=Q_1(\D)=\cdots
\]
se rompe en el número de átomos exteriores o de frontera.

\section{Especialización al operador de multiplicación}

En este capítulo trabajaremos siempre con el operador de multiplicación y con productos de Sobolev concretos. La definición abstracta de $Q_k$ pasa así a un problema explícito sobre la geometría de los átomos discretos.

\begin{defn}\label{def:operador_multiplicacion}
Sea $\D: \Poly \to \Poly$ el operador lineal definido por
\begin{equation}\label{eq:operator_multiplication_ch3}
(\D p)(z) = z\,p(z).
\end{equation}
\end{defn}

Antes del resultado general conviene fijar dos modelos elementales: uno con $|c|\ge R$, donde la no acotación desaparece tras una sola restricción, y uno interior, donde el operador ya es acotado.

Para cada $c\in\C$ escribiremos desde ahora
\begin{equation}\label{eq:def_psi_c}
\psi_c(p):=(\D p)'(c)=p(c)+cp'(c),
\qquad p\in\Poly.
\end{equation}
Este funcional recoge la contribución discreta de la derivada de $\D p$ en el átomo $c$.

\begin{prop}\label{prop:lebesgue_pre_umbral_Qk}
Sea
\[
\langle p,q\rangle_S
=
\int_{\mathbb{S}_R} p(z)\overline{q(z)}\,d\mathbf m_{0;R}(z)
+
p'(c)\overline{q'(c)},
\qquad p,q\in\Poly,
\]
con $|c|\ge R$, y sea $\D p(z)=zp(z)$. Entonces
\[
Q_0(\D)=\infty,
\qquad
Q_1(\D)\le R.
\]
En particular, en la circunferencia unidad ($R=1$) y para $|c|\ge1$, se tiene $Q_0(\D)=\infty$ y $Q_1(\D)\le1$.
\end{prop}

\begin{proof}
Si $|c|>R$, tomamos
\[
Q_n(z):=(n+1)c\,z^n-nz^{n+1}.
\]
Entonces $Q_n'(c)=0$ y $Q_n(c)=c^{n+1}$. Además,
\[
\|Q_n\|_S^2=(n+1)^2|c|^2R^{2n}+n^2R^{2n+2},
\qquad
|\psi_c(Q_n)|^2=|c|^{2n+2},
\]
de modo que
\[
\frac{|\psi_c(Q_n)|^2}{\|Q_n\|_S^2}
=
\frac{|c|^{2n+2}}{(n+1)^2|c|^2R^{2n}+n^2R^{2n+2}}
\xrightarrow[n\to\infty]{}\infty.
\]
Si $|c|=R$, definimos
\[
\alpha_n:=\frac{\sum_{k=1}^n k}{\sum_{k=1}^n k^2}=\frac{3}{2n+1},
\qquad
P_n(z):=\sum_{k=0}^n (1-\alpha_n k)\left(\frac zc\right)^k.
\]
Entonces $P_n'(c)=0$ y
\[
\|P_n\|_S^2=\frac{(n+1)(n+2)}{2(2n+1)},
\qquad
\psi_c(P_n)=\frac{(n+1)(n+2)}{2(2n+1)},
\]
por lo que
\[
\frac{|\psi_c(P_n)|^2}{\|P_n\|_S^2}
=
\frac{(n+1)(n+2)}{2(2n+1)}
\xrightarrow[n\to\infty]{}\infty.
\]
En ambos casos $\psi_c$ es no acotado. Como
\[
|\psi_c(p)|=|(\D p)'(c)|\le \|\D p\|_S,
\qquad p\in\Poly,
\]
se sigue que $\D$ no es acotado y, por tanto, $Q_0(\D)=\|\D\|_S=\infty$.

Para $Q_1$, tomamos
\[
W:=\{p\in\Poly:\psi_c(p)=0\}.
\]
Este subespacio es admisible y satisface $\operatorname{codim}_{\Poly}(W)=1$.
Si $p\in W$, entonces
\[
\|\D p\|_S^2
=
\int_{\mathbb{S}_R}|zp(z)|^2\,d\mathbf m_{0;R}(z)
+
|\psi_c(p)|^2
=
\int_{\mathbb{S}_R}|zp(z)|^2\,d\mathbf m_{0;R}(z)
\le
R^2\int_{\mathbb{S}_R}|p(z)|^2\,d\mathbf m_{0;R}(z)
\le
R^2\|p\|_S^2.
\]
Luego $\|\D|_W\|\le R$, y al tomar ínfimo sobre subespacios admisibles con codimensión algebraica $<2$, se obtiene $Q_1(\D)\le R$.
\end{proof}

\begin{exm}\label{exm:q1_finito_modelo_exterior}
La Proposición~\ref{prop:lebesgue_pre_umbral_Qk} proporciona el modelo más simple donde la no acotación global desaparece al imponer una sola restricción lineal. En efecto, con un único átomo de derivada en un punto $c$ con $|c|\ge R$, el operador $\D$ satisface
\[
Q_0(\D)=\infty,
\qquad
Q_1(\D)\le R.
\]
La restricción relevante es precisamente
\[
W=\ker(\psi_c),
\qquad
\psi_c(p)=p(c)+cp'(c)=(\D p)'(c).
\]
Es decir: la singularidad está concentrada en un único funcional lineal, y anular ese funcional basta para recuperar una restricción acotada.
\end{exm}

\begin{exm}\label{exm:c0_raiz2}
Consideremos ahora el caso interior
\[
\langle p,q\rangle_S = \int_{\mathbb{S}_1} p(z)\overline{q(z)} \, \frac{|dz|}{2\pi} + p'(0)\overline{q'(0)},
\]
y sea $\D p(z)=zp(z)$. Entonces
\[
Q_0(\D)=\|\D\|_S=\sqrt2.
\]
\end{exm}

\begin{proof}
Si $p(z) = \sum_{k=0}^n a_k z^k$, entonces $p'(0) = a_1$ y se obtiene
\begin{equation} \label{eq:norma_in_0}
\|p\|_S^2 = |a_0|^2 + 2|a_1|^2 + \sum_{k=2}^n |a_k|^2.
\end{equation}
Además, para $\D p(z) = z p(z)$ se tiene $(\D p)'(0) = a_0$, luego
\begin{equation} \label{eq:norma_out_0}
\|\D p\|_S^2 = 2|a_0|^2 + |a_1|^2 + \sum_{k=2}^n |a_k|^2.
\end{equation}
Por tanto,
\begin{equation}\label{eq:ratio_c0_raiz2}
\frac{\|\D p\|_S^2}{\|p\|_S^2} = 2 - \frac{3|a_1|^2 + \sum_{k \ge 2} |a_k|^2}{\|p\|_S^2} \le 2,
\end{equation}
y el máximo se alcanza en las funciones constantes. En consecuencia,
\[
Q_0(\D)=\|\D\|_S=\sqrt2.
\]

Además, este ejemplo muestra de manera transparente qué parte del espacio concentra la pérdida de norma: si imponemos la restricción $p(0)=0$, entonces $a_0=0$ y las expresiones \eqref{eq:norma_in_0}--\eqref{eq:norma_out_0} dan
\[
\|\D p\|_S^2=|a_1|^2+\sum_{k=2}^n |a_k|^2\le \|p\|_S^2,
\]
de modo que sobre el subespacio $\ker(\delta_0)$ el operador ya actúa con norma a lo sumo $1$.
\end{proof}

\section{Interior, exterior y funcionales asociados}

Trabajaremos con la circunferencia
\[
\mathbb S_R:=\{z\in\C:|z|=R\},
\]
que es el soporte de la parte continua en el caso modelo de Lebesgue. Dado un átomo discreto situado en un punto $c\in\C$, hay tres configuraciones geométricas posibles:
\begin{itemize}
\item si $|c|<R$, el átomo está \emph{dentro} de la circunferencia;
\item si $|c|=R$, el átomo está \emph{en la frontera};
\item si $|c|>R$, el átomo está \emph{fuera} del soporte continuo.
\end{itemize}

La dicotomía relevante es inmediata: los átomos con $|c|\ge R$ introducen la pérdida de acotación del operador de multiplicación, mientras que los átomos interiores quedan controlados por la geometría del soporte continuo.

\begin{rmk}\label{rem:lectura_geometrica_exterior}
Puede pensarse que los átomos con $|c|\ge R$ introducen ``puntos de fuga'' para el operador $\D$: al multiplicar por $z$, la evaluación en esos puntos deja de estar gobernada por la parte continua de la norma. Esa es la razón geométrica por la que en el Teorema~\ref{thm:estabilizacion_Qm} aparecen exactamente $m$ direcciones inestables.
\end{rmk}

\subsection{Funcionales asociados y dicotomía BPE}

Pasamos ahora a la formulación técnica de la intuición anterior.

\begin{defn}\label{def:funcionales_restringidos}
Sea $c\in\C$. Definimos los funcionales lineales
\[
\delta_c:\Poly\to\C,
\qquad
\delta_c(p)=p(c),
\]
y
\[
\delta'_c:\Poly\to\C,
\qquad
\delta'_c(p)=p'(c).
\]
Definimos además
\[
N_c:=\ker(\delta'_c)=\{p\in\Poly:p'(c)=0\}.
\]
Como $\delta'_c$ no es el funcional nulo, se tiene
\[
\operatorname{codim}_{\Poly}(N_c)=1.
\]
\end{defn}

\begin{defn}\label{def:bpe_ch3}
Decimos que $a\in\C$ es un punto de evaluación acotada (\emph{bounded point evaluation}, BPE) para $L^2(\mu)$ si existe una constante $C>0$ tal que
\begin{equation}\label{eq:def_bpe}
|p(a)|^2\le C\int |p(z)|^2\,d\mu(z),
\qquad p\in\Poly.
\end{equation}
Equivalentemente, la evaluación puntual se extiende continuamente a $\overline{\Poly}^{L^2(\mu)}$.
\end{defn}

\begin{defn}\label{def:delta_singular}
Sea $\langle\cdot,\cdot\rangle_S$ un producto interno de Sobolev sobre $\Poly$.
Diremos que un punto $c\in\C$ es \emph{$\delta$-singular} respecto a
$\|\cdot\|_S$ si la restricción del funcional $\delta_c$ al subespacio
$N_c$ no es acotada, es decir, si
\[
\sup\Bigl\{\frac{|p(c)|}{\|p\|_S}:p\in N_c\setminus\{0\}\Bigr\}=+\infty.
\]
Equivalentemente, existe una sucesión $(p_n)\subset N_c$ tal que
\[
\|p_n\|_S\to0,
\qquad
|p_n(c)|=1
\quad\text{para todo }n\ge1.
\]
\end{defn}

La notación introducida en \eqref{eq:def_psi_c} será el puente entre la geometría del átomo y la acción de $\D$:
\[
\psi_c(p)=(\D p)'(c)=p(c)+cp'(c),
\qquad p\in\Poly.
\]

\begin{prop}\label{prop:acotacion_extension_continua}
Consideremos el caso de la medida de Lebesgue en la circunferencia de radio $R$, con producto de Sobolev discreto $\langle \cdot, \cdot \rangle_S$. Para los puntos interiores $c$ con $|c|<R$, el funcional de evaluación $\delta_c$ es continuo respecto a la norma de Sobolev $\|\cdot\|_S$. En consecuencia, se extiende de manera única a todo el espacio de Hilbert $\mathcal H$ y su núcleo es un subespacio cerrado.
\end{prop}

\begin{proof}
Para cualquier polinomio $p\in\Poly$ y un punto $c$ con $|c|<R$, la fórmula integral de Cauchy sobre $\mathbb S_R$ da
\[
p(c)=\frac{1}{2\pi i}\int_{\mathbb S_R}\frac{p(z)}{z-c}\,dz.
\]
Aplicando Cauchy--Schwarz obtenemos
\[
|p(c)|^2\le
\left(\frac{1}{2\pi}\int_{\mathbb S_R}\frac{1}{|z-c|^2}|dz|\right)
\left(\int_{\mathbb S_R}|p(z)|^2|dz|\right).
\]
Como $|c|<R$, la primera integral es finita. La segunda queda controlada por la parte continua de la norma de Sobolev. Por tanto existe una constante $M_c<\infty$ tal que
\[
|p(c)|^2\le M_c\|p\|_S^2.
\]
La extensión continua y el carácter cerrado del núcleo son consecuencias estándar del teorema de extensión de operadores lineales acotados en subespacios densos \cite{kreyszig1989introductory}.
\end{proof}

\begin{lem}\label{lema:ABPE_derivada}
Sea $\mu$ la medida de Lebesgue normalizada en la circunferencia $\mathbb S_R$. Si $c\in\C$ satisface $|c|<R$, entonces $c$ es un punto ABPE y, en consecuencia, el funcional
\[
\delta'_c(p)=p'(c)
\]
es continuo respecto a la norma $L^2(\mu)$ y, por tanto, respecto a $\|\cdot\|_S$.
\end{lem}

\begin{proof}
En el caso de Lebesgue, todo punto del disco abierto $|c|<R$ es BPE y de hecho ABPE \cite{miller1999bounded,aleman2009nontangential}. Tomamos $r>0$ tal que la circunferencia $\gamma_r=\{z:|z-c|=r\}$ quede contenida en el disco $|z|<R$. Por la fórmula de Cauchy para la derivada,
\[
p'(c)=\frac{1}{2\pi i}\oint_{\gamma_r}\frac{p(z)}{(z-c)^2}\,dz.
\]
Luego
\[
|p'(c)|\le \frac1r\max_{z\in\gamma_r}|p(z)|.
\]
Como todos los puntos de $\gamma_r$ son BPE, existe $M>0$ tal que
\[
|p(z)|\le M\|p\|_{L^2(\mu)}
\qquad (z\in\gamma_r).
\]
Sustituyendo,
\[
|p'(c)|\le \frac Mr\|p\|_{L^2(\mu)}\le \frac Mr\|p\|_S.
\]
Por tanto $\delta'_c$ es continuo.
\end{proof}

\begin{lem}\label{lem:no_acotacion_derivada_exterior_lebesgue}
Sea $\mu=\mathbf m_{0;R}$ la medida de Lebesgue normalizada en la circunferencia $\mathbb S_R$. Si $c\in\C$ satisface $|c|\ge R$, entonces el funcional
\[
\delta'_c:\Poly\to\C,
\qquad
\delta'_c(p)=p'(c),
\]
no es acotado respecto de la norma $L^2(\mu)$.
\end{lem}

\begin{proof}
Para $n\ge1$ consideramos
\[
p_n(z):=\left(\frac zR\right)^n.
\]
Entonces $\|p_n\|_{L^2(\mu)}=1$ y
\[
p_n'(z)=\frac nR\left(\frac zR\right)^{n-1},
\qquad
|p_n'(c)|=\frac nR\left(\frac{|c|}{R}\right)^{n-1}.
\]
Si $|c|>R$, el crecimiento es exponencial; si $|c|=R$, entonces $|p_n'(c)|=n/R\to\infty$. En ambos casos,
\[
\sup_{\|p\|_{L^2(\mu)}=1}|p'(c)|=\infty.
\]
Luego $\delta'_c$ no es acotado.
\end{proof}

\begin{prop}\label{prop:no_acotacion_modelo_exterior}
Sea
\[
\langle p,q\rangle_S
=
\int_{\mathbb S_R}p(z)\overline{q(z)}\,d\mathbf m_{0;R}(z)
+
p'(c)\overline{q'(c)},
\qquad p,q\in\Poly,
\]
con $|c|\ge R$, y sea $\D p(z)=zp(z)$. Entonces el funcional
\[
\psi_c(p)=p(c)+cp'(c)
\]
es no acotado respecto de $\|\cdot\|_S$. En particular, $\D$ no es acotado y
\[
Q_0(\D)=\|\D\|_S=\infty.
\]
\end{prop}

\begin{proof}
\emph{Caso 1: $|c|>R$.}
Definimos, para $n\ge1$,
\[
Q_n(z):=(n+1)c\,z^n-nz^{n+1}.
\]
Entonces
\[
Q_n'(c)=0,
\qquad
Q_n(c)=c^{n+1}.
\]
Como los monomios $z^n$ y $z^{n+1}$ son ortogonales respecto a $\mathbf m_{0;R}$,
\[
\|Q_n\|_S^2=(n+1)^2|c|^2R^{2n}+n^2R^{2n+2}.
\]
Además,
\[
|\psi_c(Q_n)|^2=|Q_n(c)|^2=|c|^{2n+2}.
\]
Por tanto,
\[
\frac{|\psi_c(Q_n)|^2}{\|Q_n\|_S^2}
=
\frac{|c|^{2n+2}}{(n+1)^2|c|^2R^{2n}+n^2R^{2n+2}}
\sim
\frac1{n^2}\left(\frac{|c|}{R}\right)^{2n}\xrightarrow[n\to\infty]{}\infty.
\]

\emph{Caso 2: $|c|=R$.}
Para $n\ge1$ definimos
\[
\alpha_n:=\frac{\sum_{k=1}^n k}{\sum_{k=1}^n k^2}=\frac{3}{2n+1},
\qquad
P_n(z):=\sum_{k=0}^n (1-\alpha_n k)\left(\frac zc\right)^k.
\]
Entonces $cP_n'(c)=0$, luego $P_n'(c)=0$, y
\[
\|P_n\|_S^2=\sum_{k=0}^n |1-\alpha_n k|^2=\frac{(n+1)(n+2)}{2(2n+1)}.
\]
Además,
\[
\psi_c(P_n)=P_n(c)+cP_n'(c)=\frac{(n+1)(n+2)}{2(2n+1)}.
\]
Así,
\[
\frac{|\psi_c(P_n)|^2}{\|P_n\|_S^2}
=
\frac{(n+1)(n+2)}{2(2n+1)}\xrightarrow[n\to\infty]{}\infty.
\]

Si $\D$ fuera acotado, existiría $M>0$ tal que
\[
|\psi_c(p)|=|(\D p)'(c)|\le \|\D p\|_S\le M\|p\|_S,
\qquad p\in\Poly,
\]
lo que contradice la no acotación demostrada. Luego $\D$ no es acotado y $Q_0(\D)=\infty$.
\end{proof}

\begin{prop}\label{prop:caracterizacion_psi_c}
Sea $c\in\C$ y consideremos
\[
\psi_c(p)=p(c)+cp'(c).
\]
Entonces:
\begin{enumerate}
\item si $c$ es $\delta$-singular respecto a $\|\cdot\|_S$, entonces $\delta_c$ no es acotado en $\Poly$;
\item si $c$ es $\delta$-singular respecto a $\|\cdot\|_S$, entonces $\psi_c$ no es acotado respecto a $\|\cdot\|_S$;
\item si $\delta'_c$ es acotado y $\delta_c$ no es acotado, entonces $c$ es $\delta$-singular;
\item si $\delta'_c$ es acotado y $\psi_c$ no es acotado, entonces $c$ es $\delta$-singular.
\end{enumerate}
\end{prop}

\begin{proof}
La primera afirmación es inmediata, pues la no acotación de $\delta_c|_{N_c}$ implica la no acotación de $\delta_c$ sobre el espacio mayor $\Poly$.

Para la segunda, si $c$ es $\delta$-singular, existe una sucesión $(p_n)\subset N_c$ tal que
\[
\|p_n\|_S\to0,
\qquad
|p_n(c)|=1.
\]
Como $p_n'(c)=0$,
\[
\psi_c(p_n)=p_n(c),
\]
y entonces
\[
\frac{|\psi_c(p_n)|}{\|p_n\|_S}=\frac1{\|p_n\|_S}\xrightarrow[n\to\infty]{}\infty.
\]

Para la tercera, tomamos $(p_n)\subset\Poly$ con $\|p_n\|_S\le1$ y $|p_n(c)|\to\infty$. Definimos
\[
q_n(z):=p_n(z)-p_n'(c)(z-c).
\]
Entonces $q_n'(c)=0$, es decir, $q_n\in N_c$, y $q_n(c)=p_n(c)$. Como $\delta'_c$ es acotado, existe $A_c>0$ con $|p_n'(c)|\le A_c$, y por tanto
\[
\|q_n\|_S\le 1+A_c\|z-c\|_S.
\]
La sucesión $(q_n)$ queda uniformemente acotada en norma, pertenece a $N_c$ y satisface $|q_n(c)|\to\infty$. Luego $c$ es $\delta$-singular.

Para la cuarta, si $\delta'_c$ es acotado y $\psi_c$ no es acotado, entonces
\[
\delta_c=\psi_c-c\delta'_c
\]
también es no acotado. Aplicando la tercera implicación se concluye que $c$ es $\delta$-singular.
\end{proof}

\begin{thm}\label{thm:equivalencia_lebesgue}
Sea $\mu=\mathbf m_{0;R}$ la medida de Lebesgue normalizada en la circunferencia $\mathbb S_R$, y consideremos el producto de Sobolev discreto
\[
\langle p,q\rangle_S
=
\int_{\mathbb S_R}p(z)\overline{q(z)}\,d\mathbf m_{0;R}(z)
+
\sum_{k=1}^N p'(c_k)\overline{q'(c_k)}.
\]
Para cualquier $c\in\C$, son equivalentes:
\begin{enumerate}
\item $c$ es $\delta$-singular respecto a $\|\cdot\|_S$;
\item $|c|\ge R$;
\item $c$ no es BPE para $L^2(\mu)$.
\end{enumerate}
\end{thm}

\begin{proof}
\textbf{(2)$\iff$(3).} Es la caracterización clásica de los BPE para la medida de Lebesgue en circunferencia: los puntos BPE son exactamente los del disco abierto $|c|<R$ \cite{duren1970theory,szego1975orthogonal}.

\textbf{(2)$\Rightarrow$(1).} Si $|c|\ge R$, en la prueba de la Proposición~\ref{prop:no_acotacion_modelo_exterior} se construyen explícitamente polinomios con derivada nula en $c$ y crecimiento descontrolado en la evaluación en $c$. Más precisamente:
\begin{itemize}
\item si $|c|>R$, los polinomios $Q_n$ satisfacen $Q_n'(c)=0$ y
\[
\frac{|Q_n(c)|^2}{\|Q_n\|_S^2}
=
\frac{|c|^{2n+2}}{(n+1)^2|c|^2R^{2n}+n^2R^{2n+2}}
\xrightarrow[n\to\infty]{}\infty;
\]
\item si $|c|=R$, los polinomios $P_n$ satisfacen $P_n'(c)=0$ y
\[
\frac{|P_n(c)|^2}{\|P_n\|_S^2}
=
\frac{(n+1)(n+2)}{2(2n+1)}
\xrightarrow[n\to\infty]{}\infty.
\]
\end{itemize}
En ambos casos, por la Definición~\ref{def:delta_singular}, el punto $c$ es $\delta$-singular.

\textbf{(1)$\Rightarrow$(2).} Probamos el contrarrecíproco. Si $|c|<R$, entonces $c$ es BPE y, por el Lema~\ref{lema:ABPE_derivada}, tanto $\delta_c$ como $\delta'_c$ son funcionales acotados respecto a $\|\cdot\|_S$. En particular, $\delta_c$ es acotado al restringirse a $N_c$, luego $c$ no es $\delta$-singular.
\end{proof}

\begin{cor}\label{cor:lectura_tecnica_exterioridad}
En el caso de Lebesgue en la circunferencia $\mathbb S_R$, para todo $c\in\C$ son equivalentes:
\begin{enumerate}
\item $|c|\ge R$;
\item $c$ no es BPE para $L^2(\mathbf m_{0;R})$;
\item $\psi_c$ no es acotado respecto a la norma de Sobolev correspondiente.
\end{enumerate}
\end{cor}

\begin{proof}
La equivalencia entre (1) y (2) es el Teorema~\ref{thm:equivalencia_lebesgue}. Si $|c|\ge R$, la Proposición~\ref{prop:no_acotacion_modelo_exterior} da la no acotación de $\psi_c$, de modo que (1)$\Rightarrow$(3). Si $|c|<R$, por la Proposición~\ref{prop:acotacion_extension_continua} y el Lema~\ref{lema:ABPE_derivada} los funcionales $\delta_c$ y $\delta'_c$ son acotados; por la identidad $\psi_c=\delta_c+c\delta'_c$, también lo es $\psi_c$. Luego no puede darse (3). Esto prueba (3)$\Rightarrow$(1).
\end{proof}

\subsection{Clasificación geométrica}

La equivalencia anterior traduce la intuición geométrica en un criterio operativo: a partir de ahora ya no distinguiremos los átomos por una metáfora visual, sino por si generan o no una dirección inestable para $\D$.

\begin{defn}\label{def:class_points_new}
Sea $\mu=\mathbf m_{0;R}$ la medida de Lebesgue normalizada en la circunferencia $\mathbb S_R$, y consideremos el conjunto finito de puntos donde aparecen las masas discretas de derivada. Definimos
\[
\begin{aligned}
I_{\mathrm{out}}&:=\{c\in\C: c\text{ soporta una masa discreta de derivada},\ |c|\ge R\},\\
I_{\mathrm{in}}&:=\{c\in\C: c\text{ soporta una masa discreta de derivada},\ |c|<R\}.
\end{aligned}
\]
Si conviene separar la frontera del exterior estricto, definimos
\[
I_{\mathrm{out}}^*:=\{c\in I_{\mathrm{out}}:|c|>R\}.
\]
Finalmente,
\[
m:=|I_{\mathrm{out}}|.
\]
\end{defn}

\begin{rmk}\label{rem:compatibilidad_class_points}
Por el Corolario~\ref{cor:lectura_tecnica_exterioridad}, en el caso de Lebesgue la partición geométrica recién definida coincide exactamente con la dicotomía analítica
\[
 c\in I_{\mathrm{out}}
\iff
\psi_c\text{ no es acotado}
\iff
c\text{ no es BPE}.
\]
Es decir: en este modelo, los puntos de $I_{\mathrm{out}}$ son exactamente los que generan funcionales inestables, mientras que los puntos de $I_{\mathrm{in}}$ quedan en el régimen estable.
\end{rmk}

\section{El subespacio estable canónico}

Con la clasificación anterior ya puede formularse el objeto central del capítulo. La idea estructural es neutralizar exactamente las contribuciones asociadas a los puntos de $I_{\mathrm{out}}$, y solo esas. Eso conduce al subespacio canónico sobre el cual el operador recupera acotación.

Para cada $c\in I_{\mathrm{out}}$ escribiremos, por abreviatura,
\[
\psi_c(p):=(\D p)'(c)=p(c)+cp'(c).
\]

\begin{defn}\label{def:vout_ch3}
Definimos
\begin{equation}\label{eq:def_vout}
V_{\mathrm{out}}:=\bigcap_{c\in I_{\mathrm{out}}}\ker(\psi_c).
\end{equation}
Es el subespacio de polinomios en los que todas las contribuciones de $(\D p)'$ asociadas a puntos de $I_{\mathrm{out}}$ quedan anuladas.
\end{defn}

\begin{lem}\label{lem:independencia_funcionales_psi}
El sistema $\{\psi_c\}_{c\in I_{\mathrm{out}}}$ es linealmente independiente. En particular,
\[
\dim\operatorname{span}\{\psi_c:c\in I_{\mathrm{out}}\}=m.
\]
\end{lem}

\begin{proof}
Fijado $c\in I_{\mathrm{out}}$, por interpolación de Hermite existe un polinomio $h_c\in\Poly$ tal que
\[
h_c(c)=1,
\qquad
h_c'(c)=0,
\]
y, para todo $d\in I_{\mathrm{out}}$ con $d\ne c$,
\[
h_c(d)=0,
\qquad
h_c'(d)=0.
\]
Entonces
\[
\psi_d(h_c)=h_c(d)+dh_c'(d)=\delta_{dc}.
\]
Esto prueba la independencia lineal.
\end{proof}

\begin{lem}\label{lem:vout_codimension}
El subespacio $V_{\mathrm{out}}$ es admisible y satisface
\[
\operatorname{codim}_{\Poly}(V_{\mathrm{out}})=m.
\]
\end{lem}

\begin{proof}
Por definición, $V_{\mathrm{out}}$ es intersección finita de núcleos de funcionales lineales, luego es admisible. Por el Lema~\ref{lem:independencia_funcionales_psi}, los funcionales $\{\psi_c\}_{c\in I_{\mathrm{out}}}$ son linealmente independientes y generan un espacio de dimensión $m$. La Proposición~\ref{prop:finite-codim-kernels} da entonces la igualdad anunciada.
\end{proof}

\begin{lem}\label{lem:vout_acotacion}
Existe una constante $C_{\mathrm{out}}<\infty$ tal que
\[
\|\D p\|_S\le C_{\mathrm{out}}\|p\|_S,
\qquad p\in V_{\mathrm{out}}.
\]
\end{lem}

\begin{proof}
Sea $p\in V_{\mathrm{out}}$. Entonces, para todo $c\in I_{\mathrm{out}}$,
\[
(\D p)'(c)=\psi_c(p)=0.
\]
Es decir, todas las contribuciones discretas procedentes de puntos de $I_{\mathrm{out}}$ desaparecen en la norma de Sobolev de $\D p$.

Si $c\in I_{\mathrm{in}}$, entonces $|c|<R$. Por la Proposición~\ref{prop:acotacion_extension_continua} y el Lema~\ref{lema:ABPE_derivada}, existen constantes $A_c,B_c<\infty$ tales que
\[
|p(c)|\le A_c\|p\|_S,
\qquad
|p'(c)|\le B_c\|p\|_S.
\]
Luego
\[
|(\D p)'(c)|=|p(c)+cp'(c)|\le (A_c+|c|B_c)\|p\|_S.
\]

Además, como $|z|=R$ en el soporte continuo,
\[
\|zp\|_{L^2(\mathbf m_{0;R})}\le R\|p\|_{L^2(\mathbf m_{0;R})}\le R\|p\|_S.
\]
Por consiguiente,
\[
\|\D p\|_S^2
=
\|zp\|_{L^2(\mathbf m_{0;R})}^2
+
\sum_{c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}}|(\D p)'(c)|^2
\]
\[
=
\|zp\|_{L^2(\mathbf m_{0;R})}^2
+
\sum_{c\in I_{\mathrm{in}}}|(\D p)'(c)|^2
\]
\[
\le
\left(R^2+\sum_{c\in I_{\mathrm{in}}}(A_c+|c|B_c)^2\right)\|p\|_S^2.
\]
Esto prueba la acotación de $\D|_{V_{\mathrm{out}}}$.
\end{proof}

\begin{prop}\label{prop:umbral_fase1}
En el marco anterior,
\[
Q_m(\D)<\infty.
\]
Más precisamente,
\[
Q_m(\D)\le \|\D|_{V_{\mathrm{out}}}\|_S\le C_{\mathrm{out}}.
\]
\end{prop}

\begin{proof}
Por el Lema~\ref{lem:vout_codimension}, $V_{\mathrm{out}}$ es admisible y tiene codimensión exacta $m$. Por el Lema~\ref{lem:vout_acotacion}, la restricción $\D|_{V_{\mathrm{out}}}$ es acotada. La definición de $Q_m$ da inmediatamente
\[
Q_m(\D)\le \|\D|_{V_{\mathrm{out}}}\|_S<\infty.
\]
\end{proof}

\section{Necesidad de codimensión \texorpdfstring{$m$}{m}}

Para ver que la codimensión $m$ es necesaria, basta probar que ninguna codimensión estrictamente menor neutraliza a la vez las $m$ direcciones inestables.

\begin{lem}\label{lem:normalizacion_funcional_no_acotado}
Sea $X$ un espacio normado complejo y sea $\varphi:X\to\C$ un funcional lineal no acotado. Entonces existe una sucesión $(x_n)_{n\ge1}\subset X$ tal que
\[
\|x_n\|_X\to0,
\qquad
\varphi(x_n)=1
\quad\text{para todo }n.
\]
\end{lem}

\begin{proof}
Como $\varphi$ no es acotado, para cada $n\ge1$ existe $y_n\in X$ tal que
\[
\|y_n\|_X\le1,
\qquad
|\varphi(y_n)|\ge n.
\]
Definimos
\[
x_n:=\frac{y_n}{\varphi(y_n)}.
\]
Entonces $\varphi(x_n)=1$ y
\[
\|x_n\|_X=\frac{\|y_n\|_X}{|\varphi(y_n)|}\le \frac1n\xrightarrow[n\to\infty]{}0.
\]
\end{proof}

\begin{lem}\label{lem:paquetes_diagonales}
Para cada $c\in I_{\mathrm{out}}$ y cada $n\ge1$, existe un polinomio $u_{c,n}\in\Poly$ tal que
\[
\|u_{c,n}\|_S\xrightarrow[n\to\infty]{}0,
\qquad
\psi_d(u_{c,n})=\delta_{dc},
\quad \forall\,c,d\in I_{\mathrm{out}}.
\]
\end{lem}

\begin{proof}
Como cada $\psi_c$ es no acotado respecto de $\|\cdot\|_S$ (Corolario~\ref{cor:lectura_tecnica_exterioridad}), el Lema~\ref{lem:normalizacion_funcional_no_acotado} da, para todo $n\ge1$, un polinomio $x_{c,n}\in\Poly$ tal que
\[
\|x_{c,n}\|_S<\frac1n,
\qquad
\psi_c(x_{c,n})=1.
\]

Fijado $c\in I_{\mathrm{out}}$, elegimos un polinomio $g_c$ por interpolación de Hermite de modo que
\[
g_c(c)=1,
\qquad
g_c'(c)=0,
\]
y para todo $d\in I_{\mathrm{out}}$ con $d\ne c$,
\[
g_c(d)=0,
\qquad
g_c'(d)=0.
\]
Definimos
\[
u_{c,n}:=g_cx_{c,n}.
\]

La multiplicación por $g_c$ es acotada respecto a $\|\cdot\|_S$. En el soporte continuo basta controlar con $\|g_c\|_{L^\infty(|z|=R)}$. En la parte discreta, por la regla del producto,
\[
(g_cp)'(d)=g_c'(d)p(d)+g_c(d)p'(d).
\]
Si $d\in I_{\mathrm{out}}$, entonces $g_c'(d)=0$ y $(g_cp)'(d)=g_c(d)p'(d)$, que se controla con la parte discreta de $\|p\|_S$. Si $d\in I_{\mathrm{in}}$, existe una constante $C_{\mathrm{int},d}$ tal que
\[
|p(d)| \le C_{\mathrm{int},d}\|p\|_{L^2(|z|=R)} \le C_{\mathrm{int},d}\|p\|_S,
\]
y $p'(d)$ se controla directamente por $\|p\|_S$. En consecuencia, existe $C_c<\infty$ tal que
\[
\|g_cp\|_S\le C_c\|p\|_S,
\qquad p\in\Poly,
\]
y en particular
\[
\|u_{c,n}\|_S\le C_c\|x_{c,n}\|_S<\frac{C_c}{n}\xrightarrow[n\to\infty]{}0.
\]

Por último, para $d\in I_{\mathrm{out}}$,
\[
\psi_d(u_{c,n})=u_{c,n}(d)+du'_{c,n}(d)=\delta_{dc},
\]
por las condiciones de interpolación impuestas a $g_c$. Esto concluye la prueba.
\end{proof}

\begin{lem}\label{lem:imposicion_restricciones}
Sea $V\subset\Poly$ un subespacio admisible con $\operatorname{codim}_{\Poly}(V)=r<m$. Entonces existe una sucesión $(p_n)\subset V$ tal que $\|p_n\|_S\to0$ y cada $p_n$ es combinación lineal de los paquetes $\{u_{c,n}\}_{c\in I_{\mathrm{out}}}$ con coeficientes de norma unitaria en $\ell^2(I_{\mathrm{out}})$.
\end{lem}

\begin{proof}
Por admisibilidad, existen funcionales lineales independientes $\lambda_1,\dots,\lambda_r:\Poly\to\C$ tales que
\[
V=\bigcap_{i=1}^r\ker(\lambda_i).
\]
Para cada $n$ consideramos la matriz
\[
A_n:=\bigl(\lambda_i(u_{c,n})\bigr)_{\substack{1\le i\le r\\ c\in I_{\mathrm{out}}}}\in\C^{r\times m}.
\]
Enumeremos temporalmente $I_{\mathrm{out}}=\{c^{(1)},\dots,c^{(m)}\}$ para identificar $\C^{I_{\mathrm{out}}}$ con $\C^m$. Como $r<m$, el núcleo de $A_n$ es no trivial. Elegimos
\[
\beta^{(n)}=(\beta_c^{(n)})_{c\in I_{\mathrm{out}}}\in\ker(A_n),
\qquad
\|\beta^{(n)}\|_{\ell^2}=1,
\]
y definimos
\[
p_n:=\sum_{c\in I_{\mathrm{out}}}\beta_c^{(n)}u_{c,n}.
\]
Entonces $\lambda_i(p_n)=0$ para todo $i$, luego $p_n\in V$. Además,
\[
\|p_n\|_S
\le
\sum_{c\in I_{\mathrm{out}}}|\beta_c^{(n)}|\,\|u_{c,n}\|_S
\le
\sqrt m\max_{c\in I_{\mathrm{out}}}\|u_{c,n}\|_S\xrightarrow[n\to\infty]{}0.
\]
\end{proof}

\begin{prop}\label{prop:umbral_fase2}
Para todo subespacio admisible $V\subset\Poly$ con $\operatorname{codim}_{\Poly}(V)<m$, la restricción $\D|_V$ no es acotada. En particular,
\[
Q_{m-1}(\D)=\infty.
\]
\end{prop}

\begin{proof}
Fijemos un subespacio admisible $V\subset\Poly$ con $\operatorname{codim}_{\Poly}(V)=r<m$. Por el Lema~\ref{lem:imposicion_restricciones}, existe una sucesión $(p_n)\subset V$ tal que $\|p_n\|_S\to0$ y
\[
p_n=\sum_{c\in I_{\mathrm{out}}}\beta_c^{(n)}u_{c,n},
\qquad
\|\beta^{(n)}\|_{\ell^2}=1.
\]

Definimos el funcional
\[
\psi^{(n)}:=\sum_{c\in I_{\mathrm{out}}}\overline{\beta_c^{(n)}}\,\psi_c.
\]
La diagonalidad de los paquetes da
\[
\psi^{(n)}(p_n)
=
\sum_{c,d\in I_{\mathrm{out}}}\overline{\beta_c^{(n)}}\beta_d^{(n)}\psi_c(u_{d,n})
=
\sum_{c\in I_{\mathrm{out}}}|\beta_c^{(n)}|^2=1.
\]

Por Cauchy--Schwarz,
\[
|\psi^{(n)}(p)|
\le
\left(\sum_{c\in I_{\mathrm{out}}}|\beta_c^{(n)}|^2\right)^{1/2}
\left(\sum_{c\in I_{\mathrm{out}}}|\psi_c(p)|^2\right)^{1/2}
\le
\|\D p\|_S,
\qquad p\in\Poly,
\]
ya que los términos $|\psi_c(p)|^2=|(\D p)'(c)|^2$ aparecen en la suma discreta de $\|\D p\|_S^2$. Aplicado a $p_n$,
\[
1=|\psi^{(n)}(p_n)|\le \|\D p_n\|_S.
\]
Como $\|p_n\|_S\to0$, se sigue que
\[
\frac{\|\D p_n\|_S}{\|p_n\|_S}\xrightarrow[n\to\infty]{}\infty.
\]
Por tanto $\D|_V$ no es acotada. Como $V$ era arbitrario entre los subespacios admisibles de codimensión $<m$, concluimos que $Q_{m-1}(\D)=\infty$.
\end{proof}

\section{Teorema de estabilización}

\begin{thm}[Teorema de estabilización]\label{thm:estabilizacion_Qm}
Sea
\[
\langle p,q\rangle_S
=
\int_{\mathbb S_R}p(z)\overline{q(z)}\,d\mathbf m_{0;R}(z)
+
\sum_{c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}} p'(c)\overline{q'(c)},
\qquad p,q\in\Poly,
\]
y sea $m=|I_{\mathrm{out}}|$. Entonces
\[
Q_j(\D)=\infty
\qquad \text{para todo } 0\le j \le m-1,
\]
y el límite en el índice $m$ colapsa siendo finito (y en particular, menor o igual a $R$ si $I_{\mathrm{in}}=\varnothing$):
\begin{equation}\label{eq:colapso_estabilizacion}
Q_m(\D)<\infty.
\end{equation}
Es decir: la cadena de infinitos se rompe exactamente en $m$.
\end{thm}

\begin{proof}
Por la Proposición~\ref{prop:umbral_fase2},
\[
Q_{m-1}(\D)=\infty.
\]
Como la sucesión $\{Q_k(\D)\}_{k\ge0}$ es monótona no creciente por la Proposición~\ref{prop:propiedades_Qk}, se deduce que para todo $0\le j<m$,
\[
Q_j(\D)\ge Q_{m-1}(\D)=\infty,
\]
luego $Q_j(\D)=\infty$.

Por otra parte, la Proposición~\ref{prop:umbral_fase1} da
\[
Q_m(\D)\le \|\D|_{V_{\mathrm{out}}}\|_S<\infty.
\]
Esto prueba la ruptura exacta en $m$.
\end{proof}

\section{Refinamientos del umbral}

\subsection{Caso sin átomos interiores}

Cuando no hay átomos interiores, el subespacio estable canónico no solo da finitud: da el valor exacto del umbral.

\begin{thm}\label{thm:umbral_puramente_exterior}
Sea
\[
\langle p,q\rangle_S
=
\int_{\mathbb S_R} p(z)\overline{q(z)}\,d\mathbf m_{0;R}(z)
+
\sum_{c\in I_{\mathrm{out}}} p'(c)\overline{q'(c)},
\]
y supongamos que $I_{\mathrm{in}}=\varnothing$. Sea $m=|I_{\mathrm{out}}|$. Entonces
\[
Q_j(\D)=\infty\ \text{ para }\ 0\le j\le m-1,
\qquad
Q_m(\D)\le R.
\]
En particular, el primer índice finito del operador de multiplicación es exactamente $m$.
\end{thm}

\begin{proof}
El Teorema~\ref{thm:estabilizacion_Qm} ya da $Q_j(\D)=\infty$ para $0\le j<m$. Como en este caso no hay átomos interiores, la estimación del Lema~\ref{lem:vout_acotacion} se reduce a
\[
\|\D p\|_S\le R\|p\|_S,
\qquad p\in V_{\mathrm{out}}.
\]
Por tanto,
\[
Q_m(\D)\le \|\D|_{V_{\mathrm{out}}}\|_S\le R.
\]
\end{proof}

\begin{thm}\label{thm:exactitud_Qm_R}
Bajo las hipótesis del Teorema~\ref{thm:umbral_puramente_exterior}, se tiene
\[
Q_{m-1}(\D)=\infty
\qquad\text{y}\qquad
Q_m(\D)=R.
\]
En particular,
\[
\infty=Q_0(\D)=\dots=Q_{m-1}(\D)>Q_m(\D)=R\ge Q_{m+1}(\D)\ge\dots\ge0.
\]
\end{thm}

\begin{proof}
La explosión $Q_{m-1}(\D)=\infty$ es el contenido del Teorema~\ref{thm:umbral_puramente_exterior}. La igualdad exacta $Q_m(\D)=R$ se obtiene combinando las dos estimaciones siguientes.

\emph{Cota superior.} El Teorema~\ref{thm:umbral_puramente_exterior} ya da $Q_m(\D)\le R$.

\emph{Cota inferior.} Sea $V\subset\Poly$ un subespacio admisible con $\operatorname{codim}_{\Poly}(V)\le m$. Enumeremos temporalmente $I_{\mathrm{out}}=\{c^{(1)},\dots,c^{(m)}\}$ y consideremos la aplicación lineal
\[
T:\Poly\to\C^m,
\qquad
T(p)=\bigl(p'(c^{(1)}),\dots,p'(c^{(m)})\bigr).
\]
Su núcleo $W=\ker(T)$ tiene codimensión exacta $m$. Como $V$ tiene codimensión a lo sumo $m$, la intersección $V\cap W$ es no trivial. Para cualquier $p\in V\cap W$ se cumple $p'(c)=0$ para todo $c\in I_{\mathrm{out}}$, luego
\[
\|p\|_S=\|p\|_{L^2},
\qquad
\|\D p\|_S\ge \|zp\|_{L^2}=R\|p\|_{L^2}=R\|p\|_S.
\]
En consecuencia, $\|\D|_V\|\ge R$. Tomando ínfimo sobre todos esos subespacios $V$ se obtiene $Q_m(\D)\ge R$.

Ambas desigualdades implican la identidad deseada.
\end{proof}

\begin{rmk}\label{rem:frontera_Qm}
Cuando alguno de los átomos satisface $c\in I_{\mathrm{out}}\setminus I_{\mathrm{out}}^*$, el funcional $\psi_c$ sigue siendo no acotado por la Proposición~\ref{prop:no_acotacion_modelo_exterior}. Por tanto, la demostración anterior no distingue entre átomos estrictamente exteriores y átomos situados exactamente en la frontera: ambos pertenecen al mismo régimen de estabilización algebraica, aunque el Capítulo~\ref{sec:ceros_acotacion} los separará después en la geometría fina de los ceros.
\end{rmk}

\subsection{Caso mixto}

En el caso mixto, los átomos interiores no cambian el índice de estabilización, pero sí deforman la constante con la que $\D$ actúa sobre el subespacio estable canónico. Ese matiz es el que prepara la lectura matricial del capítulo siguiente.

\begin{thm}\label{thm:umbral_caso_mixto}
Sea
\[
\langle p,q\rangle_S
=
\int_{\mathbb S_R} p(z)\overline{q(z)}\,d\mathbf m_{0;R}(z)
+
\sum_{c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}} p'(c)\overline{q'(c)},
\]
donde los conjuntos de puntos $I_{\mathrm{in}}$ e $I_{\mathrm{out}}$ son ambos no vacíos, y sea $m=|I_{\mathrm{out}}|$. Entonces
\[
Q_j(\D)=\infty
\qquad \text{para todo } 0\le j\le m-1.
\]
Además, en el índice $m$ colapsa, y existen constantes $A_c,B_c<\infty$ para $c\in I_{\mathrm{in}}$ tales que
\begin{equation}\label{eq:colapso_mixto}
Q_m(\D)
\le
\left(R^2+\sum_{c\in I_{\mathrm{in}}}(A_c+|c|B_c)^2\right)^{1/2}
<\infty.
\end{equation}
\end{thm}

\begin{proof}
La parte $Q_j(\D)=\infty$ para $0\le j<m$ es exactamente el contenido del Teorema~\ref{thm:estabilizacion_Qm}. La cota para $Q_m(\D)$ es la estimación obtenida en la prueba del Lema~\ref{lem:vout_acotacion}, aplicada al subespacio $V_{\mathrm{out}}$.
\end{proof}

\begin{thm}\label{thm:caracterizacion_finita_vout_mixto}
Bajo las hipótesis del Teorema~\ref{thm:umbral_caso_mixto}, sea
\[
V_{\mathrm{out}}:=\bigcap_{c\in I_{\mathrm{out}}}\ker(\psi_c),
\qquad
\psi_c(p)=(\D p)'(c),
\]
y sea $\mathcal H_{\mathrm{out}}$ la completación de $V_{\mathrm{out}}$ para la norma de Sobolev. Para cada $c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}$ definimos
\[
\ell_c(p):=p'(c),
\qquad p\in V_{\mathrm{out}},
\]
y, para $c\in I_{\mathrm{in}}$,
\[
\eta_c(p):=(\D p)'(c)=p(c)+cp'(c),
\qquad p\in V_{\mathrm{out}}.
\]
Entonces:
\begin{enumerate}
\item cada $\ell_c$ y cada $\eta_c$ se extienden continuamente a $\mathcal H_{\mathrm{out}}$;
\item si $u_c,v_c\in\mathcal H_{\mathrm{out}}$ son los representadores de Riesz de $\ell_c$ y $\eta_c$, y si
\[
E:=\operatorname{span}\Bigl(\{u_c:c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}\}\cup\{v_c:c\in I_{\mathrm{in}}\}\Bigr),
\]
entonces
\[
\|\D|_{V_{\mathrm{out}}}\|_S^2
=
\max\!\left\{R^2,\ \sup_{\substack{p\in E\\p\ne0}}\frac{\|\D p\|_S^2}{\|p\|_S^2}\right\}.
\]
En particular, si $e_1,\dots,e_d$ es una base de $E$ y
\[
\mathbf A=(\langle e_j,e_i\rangle_S)_{i,j=1}^d,
\qquad
\mathbf B=(\langle \D e_j,\D e_i\rangle_S)_{i,j=1}^d,
\]
entonces
\[
\|\D|_{V_{\mathrm{out}}}\|_S^2
=
\max\{R^2,\lambda_{\max}(\mathbf B,\mathbf A)\},
\]
donde $\lambda_{\max}(\mathbf B,\mathbf A)$ denota el máximo cociente de Rayleigh generalizado.
\end{enumerate}
\end{thm}

\begin{proof}
Por el Lema~\ref{lem:vout_acotacion}, la restricción $\D|_{V_{\mathrm{out}}}$ es acotada y se extiende a $\mathcal H_{\mathrm{out}}$.

Para cada $c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}$, la cota
\[
|\ell_c(p)|=|p'(c)|\le \|p\|_S,
\qquad p\in V_{\mathrm{out}},
\]
es inmediata por la propia definición de la norma de Sobolev. Si $c\in I_{\mathrm{in}}$, el Lema~\ref{lem:vout_acotacion} proporciona constantes $A_c,B_c<\infty$ con
\[
|\eta_c(p)|=|p(c)+cp'(c)|\le (A_c+|c|B_c)\|p\|_S,
\qquad p\in V_{\mathrm{out}}.
\]
Luego todos esos funcionales se extienden continuamente, y el teorema de Riesz produce los vectores $u_c$ y $v_c$.

Sea ahora $p\in\mathcal H_{\mathrm{out}}$ y escribamos su descomposición ortogonal respecto de $E$:
\[
p=x+y,
\qquad
x\in E,
\quad
y\in E^\perp.
\]
Como $y$ es ortogonal a todos los representadores de Riesz, se tiene
\[
\ell_c(y)=0\quad(c\in I_{\mathrm{in}}\cup I_{\mathrm{out}}),
\qquad
\eta_c(y)=0\quad(c\in I_{\mathrm{in}}).
\]
En particular,
\[
\|y\|_S^2=\|y\|_{L^2(\mathbf m_{0;R})}^2.
\]
Además, por paso al límite desde $V_{\mathrm{out}}$, también desaparecen las contribuciones discretas exteriores de $\D y$, y por tanto
\[
\|\D y\|_S^2=R^2\|y\|_{L^2(\mathbf m_{0;R})}^2=R^2\|y\|_S^2.
\]

Como además $\langle x,y\rangle_S=0$ y las derivadas puntuales de $y$ se anulan, se obtiene
\[
\|p\|_S^2=\|x\|_S^2+\|y\|_S^2,
\]
y análogamente
\[
\|\D p\|_S^2=\|\D x\|_S^2+R^2\|y\|_S^2.
\]
Dividiendo ambas identidades,
\[
\frac{\|\D p\|_S^2}{\|p\|_S^2}
\le
\max\!\left\{R^2,\sup_{\substack{q\in E\\q\ne0}}\frac{\|\D q\|_S^2}{\|q\|_S^2}\right\}.
\]
La desigualdad opuesta es inmediata porque tanto $E$ como vectores no nulos de $E^\perp$ pertenecen al dominio variacional. Esto prueba la segunda identidad.

La formulación matricial es ahora el principio de Rayleigh--Ritz en el espacio finito-dimensional $E$: si $x=\sum_{j=1}^d v_je_j$, entonces
\[
\|x\|_S^2=v^*\mathbf Av,
\qquad
\|\D x\|_S^2=v^*\mathbf Bv,
\]
y por tanto
\[
\sup_{\substack{x\in E\\x\ne0}}\frac{\|\D x\|_S^2}{\|x\|_S^2}=\lambda_{\max}(\mathbf B,\mathbf A).
\]
Sustituyendo en la identidad anterior se obtiene la caracterización anunciada.
\end{proof}

\begin{rmk}\label{rem:q_m_mixto_abierto_finito_dimensional}
El Teorema~\ref{thm:caracterizacion_finita_vout_mixto} identifica exactamente la constante canónica
\[
C_{\mathrm{out}}:=\|\D|_{V_{\mathrm{out}}}\|_S,
\]
y en particular da la cota superior
\[
Q_m(\D)\le C_{\mathrm{out}}.
\]
Sin embargo, esto no determina todavía el valor exacto de $Q_m(\D)$, porque en la definición de $Q_m$ el ínfimo se toma sobre \emph{todos} los subespacios admisibles de codimensión $\le m$, no solo sobre el subespacio canónico $V_{\mathrm{out}}$. El problema exacto del caso mixto, por tanto, sigue abierto en esta memoria. Lo importante para lo que sigue es que la posible desviación respecto de $R$ queda ya confinada a un espacio finito-dimensional explícito.
\end{rmk}

El teorema de estabilización resuelve la pregunta central de este capítulo: el operador de multiplicación es globalmente no acotado, pero la inestabilidad relevante se concentra en exactamente $m$ direcciones, una por cada punto de $I_{\mathrm{out}}$. Al anular esas $m$ direcciones, el operador recupera acotación; cuando no hay átomos interiores, además, el umbral exacto es $R$.

La estructura algebraica complementaria del subespacio canónico queda recogida en el Apéndice~\ref{ap:estructura_algebraica_vout}. Para la progresión principal del trabajo, lo decisivo es otra cosa: la obstrucción ya quedó localizada en codimensión finita. Por eso el paso al Capítulo~\ref{sec:matricial} es inmediato. Allí veremos cómo esa ruptura se manifiesta en las truncaciones matriciales de $\D$ y en sus valores singulares.
